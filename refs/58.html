가상 예시 · 기관명은 자리표시
빗물이 기왓골을 타고 내려와 암막새 끝에 맺히는 움직임 3초, 한 번. 멈춘 화면에는 젖은 골과 맺힌 물방울이 남습니다.

가져갈 장치 3

표지 하나를 위한 장식이 아니라, 다른 계획안 표지에도 옮겨 쓸 수 있는 방법만 골랐습니다.

원근으로 물러나는 반복 사물

같은 기와가 여러 장이지만 카메라를 처마 선을 따라 비스듬히 두어, 가장 가까운 수막새 한 장만 크게 읽히고 나머지는 리듬이 됩니다. 복제해 늘어놓은 화면이 되지 않습니다.

응용: 사물함 한 줄, 책상 열, 통학버스 행렬처럼 반복되는 학교 사물의 맨 앞 하나를 주인공으로.

멈춘 뒤에도 남는 움직임의 흔적

빗방울은 3초 뒤 사라지지만 지나간 골은 젖은 자국으로 어둡게 남고, 둘째 물방울은 암막새 끝에 맺혀 멈춥니다. 인쇄해도 "물은 골로 모여 끝에서 떨어진다"는 뜻이 남습니다.

응용: 이동 경로를 보여 주는 표지(통학로, 급식 동선)에서 움직임 대신 지나간 자국을 남기기.

기관명 옆 인장 자리에 수호 상징

해태를 붉은 인면에 흰 선(백문)으로 새겨, 문서에 도장이 찍히는 자리에 둡니다. 화면의 유일한 강조색이 이 한 곳이라, 기와 회청과 한지 사이에서 바로 눈에 들어옵니다.

응용: 상징 동물·꽃·문양을 기관 줄 옆 작은 인장으로. 본문 장표에서는 쪽 번호 옆 작은 인장으로 이어 쓰기.

흐름 설계 메모

수막새는 기와 줄의 끝을 막아 빗물과 바람이 스미지 않게 하는 기와이고, 해태는 옳고 그름을 가려 궁궐 문을 지키는 상상 속 동물입니다. 둘 다 "지킨다"는 일을 하므로 학교 재난안전 계획에 맞췄습니다.

계획안 구성은유이 표지 뒤 본문 장표 아이디어
Ⅰ 추진 배경·현황기왓장 한 장 한 장 = 학교 시설 하나하나. 소성 색이 조금씩 다른 것처럼 학교마다 노후도가 다름핵심 수치: 30년 이상 노후 건물 비율, 최근 3년 학교 안전사고 건수(수치 타일 + 설명 카드)
Ⅱ 목표 및 지표처마 끝까지 빈틈없이 = 사각지대 0비교: 현재값 → 목표값(예: 안전점검 사각지대 37곳 → 0곳), 측정 방법 함께 적기
Ⅲ 세부 추진과제기왓골 = 위험이 모이는 경로, 막새 = 경로 끝을 막는 장치현안→해결방안: 왼쪽에 위험이 모이는 곳(통학로·과학실·노후 전기설비), 오른쪽에 끝단 대책(보행로 분리, 약품 이력제, 전기 안전진단)
Ⅳ 추진체계해태 = 점검하고 가리는 눈. 기관 인장 자리에 둔 것과 같은 뜻조직·체계: 교육감 → 재난안전대책본부 → 시설·급식·체험학습 점검반 → 학교 안전책임관
Ⅴ 일정 및 예산기와는 처마 끝 줄부터 깔고 위로 올라감 = 가장 취약한 끝단부터 먼저일정: 1분기 취약 학교 우선 정비 → 2~3분기 확대 → 4분기 전수 점검(간트 + 마일스톤)

본문은 평면 요소에 H 계열 값을 입혀 이어 갑니다(한지 바탕, 먹 글자, 기와 회청 농담 다섯 단계, 주홍은 한 곳). 움직임은 이 표지 한 장만 씁니다.

재현 체크

빌린 대표 장치와, 실물에서 확인해 넣은 관찰 디테일입니다.

  • 연화문 수막새: 단판 연판 8엽(끝이 살짝 들린 판단), 연판 사이 가는 간판, 가운데 솟은 자방과 연자 1+8, 구획선, 테두리 구슬무늬(연주문) 24알
  • 세월의 흔적: 위쪽 테가 이 빠지듯 떨어져 나간 자리(깨진 면은 더 밝은 흙색), 연판 하나가 닳은 것, 오목한 곳에 남은 흰 회 자국, 기와마다 다른 소성 색
  • 조선 암막새: 아래로 처진 역삼각형 드림새, 테두리 띠, 가운데 연꽃 봉오리와 양쪽 당초 줄기
  • 지붕 구조: 오목한 암키와 골과 둥근 수키와 등이 번갈아 생기는 기왓골, 위 장이 아래 장을 덮는 이음 턱
  • 해태: 이마의 뿔, 목띠의 방울, 다리의 비늘, 곱슬 갈기와 세 갈래 꼬리, 넓적다리 소용돌이(석수 조각의 관례)를 전각 선으로
  • 뺀 것: 와구토(막새가 없는 수키와 끝을 막는 흙)는 막새가 이미 막고 있어 그리지 않음. 해태의 화재 방지설은 전승일 뿐 문헌 근거가 없다는 문화재청 설명이 있어, "옳고 그름을 가려 지킨다"는 뜻만 씀

  function vnoise(x,y,s){var xi=Math.floor(x),yi=Math.floor(y),xf=x-xi,yf=y-yi;var u=xf*xf*(3-2*xf),v=yf*yf*(3-2*yf);
    var a=hash(xi,yi,s),b=hash(xi+1,yi,s),c=hash(xi,yi+1,s),d=hash(xi+1,yi+1,s);return a+(b-a)*u+(c-a)*v+(a-b-c+d)*u*v;}
  function fbm(x,y,s,o){var t=0,a=.5,f=1;for(var i=0;i<(o||4);i++){t+=a*vnoise(x*f,y*f,s+i*17);f*=2.03;a*=.5;}return t;}
  function clamp(x,a,b){return x<a?a:x>b?b:x;}
  function sstep(a,b,x){var t=clamp((x-a)/(b-a),0,1);return t*t*(3-2*t);}

  /* ── 기와 단면(가로 x): 암키와 골(오목) + 수키와 등(반원) ── */
  var RS=0.22;                                   /* 수키와 반지름 */
  function chan(x){var c=Math.floor(x)+0.5,t=(x-c)/0.5;return -0.09*(1-t*t);}
  function ridge(x){var k=Math.round(x),d=x-k;return Math.abs(d)<RS? -0.02+Math.sqrt(RS*RS-d*d) : -9;}
  function prof(x){return Math.max(chan(x),ridge(x));}
  var SLOPE=28*Math.PI/180, SA=Math.sin(SLOPE), CA=Math.cos(SLOPE);
  function roofP(x,s){return [x, prof(x)+s*SA, -s*CA];}
  var TILT=12*Math.PI/180;                      /* 막새 면이 아래로 숙은 각 */
  var FN=[0,-Math.sin(TILT),Math.cos(TILT)], FUP=[0,Math.cos(TILT),Math.sin(TILT)];

  /* ── 카메라 ── */
  var E=[-2.0,1.05,1.55], T=[2.6,-0.12,-0.75];
  function sub(a,b){return [a[0]-b[0],a[1]-b[1],a[2]-b[2]];}
  function dot(a,b){return a[0]*b[0]+a[1]*b[1]+a[2]*b[2];}
  function cross(a,b){return [a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0]];}
  function norm(a){var l=Math.hypot(a[0],a[1],a[2]);return [a[0]/l,a[1]/l,a[2]/l];}
  function add(a,b,k){return [a[0]+b[0]*k,a[1]+b[1]*k,a[2]+b[2]*k];}
  var F=norm(sub(T,E)), Rt=norm(cross(F,[0,1,0])), Up=cross(Rt,F);
  var SC=1, OX=0, OY=0;
  function proj(p){var d=sub(p,E),z=dot(d,F);return [OX+SC*dot(d,Rt)/z, OY-SC*dot(d,Up)/z, z];}
  /* 주인공 수막새(x=0)가 그림 구역의 정해진 자리·크기에 오도록 맞춘다 */
  (function fit(){var c=[0,-0.02,0.03];var a=proj(c),b=proj(add(c,[1,0,0],0.25));
    SC=(H*0.15)/Math.hypot(b[0]-a[0],b[1]-a[1]); a=proj(c); OX=W*0.27-a[0]; OY=H*0.60-a[1];})();

  var LIGHT=norm([-0.55,0.72,0.42]);

  /* ── 막새 면 부조: 높이맵 → 법선 → 음영 ── */
  function shadeTex(N,Hm,Al,seed,base,opt){
    var cv=document.createElement('canvas');cv.width=cv.height=N;var g=cv.getContext('2d');var im=g.createImageData(N,N),px=im.data;
    /* 가볍게 흐리기(마모된 부조), 넓게 흐린 값과 차이 = 오목한 곳 */
    function blur(src,r){var tmp=new Float32Array(N*N),out=new Float32Array(N*N),i,j,k,acc,w=2*r+1;
      for(j=0;j<N;j++){acc=0;for(k=-r;k<=r;k++)acc+=src[j*N+clamp(k,0,N-1)];for(i=0;i<N;i++){tmp[j*N+i]=acc/w;acc+=src[j*N+clamp(i+r+1,0,N-1)]-src[j*N+clamp(i-r,0,N-1)];}}
      for(i=0;i<N;i++){acc=0;for(k=-r;k<=r;k++)acc+=tmp[clamp(k,0,N-1)*N+i];for(j=0;j<N;j++){out[j*N+i]=acc/w;acc+=tmp[clamp(j+r+1,0,N-1)*N+i]-tmp[clamp(j-r,0,N-1)*N+i];}}return out;}
    var Hs=blur(blur(Hm,Math.max(1,Math.round(N/320))),Math.max(1,Math.round(N/320)));
    var Hb=blur(blur(Hm,Math.round(N/30)),Math.round(N/30));
    var L=[-0.5,-0.66,0.56];var l=Math.hypot(L[0],L[1],L[2]);L=[L[0]/l,L[1]/l,L[2]/l];
    var hs=opt.hscale*N/2;
    for(var j=1;j<N-1;j++)for(var i=1;i<N-1;i++){var id=j*N+i;var a=Al[id];if(a<=0){px[id*4+3]=0;continue;}
      var dx=(Hs[id+1]-Hs[id-1])*hs*0.5, dy=(Hs[id+N]-Hs[id-N])*hs*0.5;
      var nx=-dx,ny=-dy,nz=1,nl=Math.hypot(nx,ny,nz);nx/=nl;ny/=nl;nz/=nl;
      var df=Math.max(0,nx*L[0]+ny*L[1]+nz*L[2]);
      var rz=2*df*nz-L[2];var sp=Math.pow(Math.max(0,rz),18)*0.10;
      var cav=Math.max(0,Hb[id]-Hs[id]);
      var u=i/N,v=j/N;
      var mot=(fbm(u*7,v*7,seed,4)-0.5)*0.16;
      var sh=0.40+0.70*df-cav*1.25+mot;
      var r=base[0]*sh,gg=base[1]*sh,b=base[2]*sh;
      /* 오목한 곳에 남은 흰 회 자국 */
      var lime=sstep(0.08,0.2,cav)*sstep(0.55,0.75,fbm(u*11,v*11,seed+5,3))*0.42;
      r+=(206-r)*lime;gg+=(206-gg)*lime;b+=(200-b)*lime;
      /* 깨진 면은 밝은 생흙색 */
      if(opt.frac&&opt.frac[id]>0){var f=opt.frac[id];r+=(150-r)*f*0.6;gg+=(146-gg)*f*0.6;b+=(140-b)*f*0.6;}
      var gr=(hash(i,j,seed+9)-0.5)*14; if(hash(i*3,j*7,seed+3)>0.993)gr-=40;
      px[id*4]=clamp(r+gr+sp*255,0,255);px[id*4+1]=clamp(gg+gr+sp*255,0,255);px[id*4+2]=clamp(b+gr+sp*250,0,255);px[id*4+3]=a*255;}
    g.putImageData(im,0,0);return cv;}

  /* 수막새: 테(주연부) 구슬무늬 24 · 연판 8 · 간판 · 자방과 연자 1+8 · 이 빠진 테 */
  function makeSumak(N,seed){
    var Hm=new Float32Array(N*N),Al=new Float32Array(N*N),frac=new Float32Array(N*N);
    var chipA=-0.95+hash(seed,1,2)*0.3, worn=Math.floor(hash(seed,3,4)*8);
    for(var j=0;j<N;j++)for(var i=0;i<N;i++){var id=j*N+i;
      var u=(i+.5)/N*2-1,v=(j+.5)/N*2-1,r=Math.hypot(u,v),th=Math.atan2(v,u);
      var edge=0.985+0.012*(fbm(th*3+5,1,seed,2)-0.5);
      var dA=Math.atan2(Math.sin(th-chipA),Math.cos(th-chipA));
      var chipR=1.2; if(Math.abs(dA)<0.32){chipR=0.885+0.06*fbm(th*9,2,seed+1,3)+0.25*Math.pow(Math.abs(dA)/0.32,3);}
      var rim=Math.min(edge,chipR);
      Al[id]=clamp((rim-r)*N*0.5,0,1); if(Al[id]<=0)continue;
      var h=0.10;
      /* 테: 안쪽 벽 0.835~0.86, 바깥 모서리 둥글게 */
      if(r>0.835) h=0.10+0.40*sstep(0.835,0.865,r);
      if(r>rim-0.03) h-=0.18*sstep(rim-0.03,rim,r);
      /* 구슬무늬 */
      var st=Math.PI*2/24, ba=Math.round(th/st)*st, bx=0.917*Math.cos(ba), by=0.917*Math.sin(ba), bd=Math.hypot(u-bx,v-by);
      if(bd<0.033) h+=0.16*Math.sqrt(1-(bd/0.033)*(bd/0.033));
      /* 구획선 */
      h=Math.max(h,0.10+0.14*Math.exp(-Math.pow((r-0.80)/0.009,2)));
      /* 연판 8 */
      var pst=Math.PI/4, pi=Math.round((th-pst/2)/pst), pa=pi*pst+pst/2, ph=th-pa, t=(r-0.295)/(0.775-0.295);
      if(t>0&&t<1){var d=r*Math.sin(ph), hw=0.172*Math.pow(Math.sin(Math.PI*(0.08+0.92*t)),0.48)*(1-0.12*t);
        if(Math.abs(d)<hw&&Math.cos(ph)>0){var q=d/hw,dome=Math.sqrt(1-q*q);
          var hp=0.12+0.30*dome*Math.pow(Math.sin(Math.PI*(0.05+0.95*t)),0.3);
          hp-=0.07*Math.exp(-Math.pow((t-0.70)/0.08,2))*dome;          /* 판단 앞 오목 */
          hp+=0.17*Math.exp(-Math.pow((t-0.885)/0.055,2))*(1-q*q);     /* 판단 반전(살짝 들린 끝) */
          hp+=0.03*Math.exp(-Math.pow(d/0.010,2))*t;                    /* 가운데 능선 */
          if(((pi%8)+8)%8===worn) hp=0.10+(hp-0.10)*(0.82-0.25*sstep(0.6,1,t));
          h=Math.max(h,hp);}}
      /* 간판(연판 사이 가는 능선 + 끝 삼각) */
      var ma=Math.round(th/pst)*pst, dm=r*Math.sin(th-ma);
      if(r>0.42&&r<0.79&&Math.cos(th-ma)>0){var w=0.006+0.022*sstep(0.70,0.785,r)*(1-sstep(0.785,0.79,r));h=Math.max(h,0.10+0.15*Math.exp(-Math.pow(dm/Math.max(w,0.006),2)));}
      /* 자방 + 연자 */
      if(r<0.30){var hz=0.36+0.04*(1-r*r/0.09);hz*=sstep(0.30,0.265,r);hz+=0.05*Math.exp(-Math.pow((r-0.255)/0.012,2));
        var sd=Math.hypot(u,v); if(sd<0.045) hz+=0.12*Math.sqrt(1-sd*sd/0.002025);
        var sa=Math.PI/4, s0=Math.round(th/sa)*sa, sx=0.158*Math.cos(s0), sy=0.158*Math.sin(s0), s1=Math.hypot(u-sx,v-sy);
        if(s1<0.04) hz+=0.11*Math.sqrt(1-s1*s1/0.0016);
        h=Math.max(h,hz);}
      /* 마모 */
      h-=0.05*fbm(u*5,v*5,seed+7,4);
      if(chipR<1.1&&r>chipR-0.05){frac[id]=sstep(chipR-0.05,chipR,r);h-=0.12*frac[id]*fbm(u*20,v*20,seed,2);}
      Hm[id]=h;}
    return shadeTex(N,Hm,Al,seed,[108,117,128],{hscale:0.13,frac:frac});}

  /* 암막새: 조선식 역삼각 드림새, 테두리 + 가운데 연꽃 봉오리 + 양쪽 당초 */
  function amakY(xo){var t=xo/0.5;var top=-0.09*(1-t*t)-0.012;var ua=Math.min(1,Math.abs(xo)/0.42);return [top, top-(0.07+0.16*Math.pow(1-ua,1.25))];}
  var amakCurves=(function(){var cs=[];[-1,1].forEach(function(sg){var pts=[];
      for(var p=0;p<=1;p+=0.02){var x=sg*(0.055+0.20*p);var yy=amakY(x);var mid=(yy[0]+yy[1])/2+0.012;pts.push([x,mid+0.018*Math.sin(p*Math.PI*1.6)]);}
      var e=pts[pts.length-1],cx=e[0]-sg*0.0,cy=e[1]+0.028, a0=Math.PI/2;
      for(var q=0;q<=1;q+=0.02){var ang=a0+sg*q*Math.PI*2.1, rr=0.028*(1-0.75*q);pts.push([cx+sg*rr*Math.cos(ang-Math.PI/2)*-1*0+rr*Math.sin(ang)*sg*-1, cy-rr*Math.cos(ang)]);}
      var tail=[];for(var p2=0;p2<=1;p2+=0.05){var x2=sg*(0.27+0.10*p2);var y2=amakY(x2);tail.push([x2,(y2[0]+y2[1])/2+0.006]);}
      cs.push(pts);cs.push(tail);});return cs;})();
  function makeAmak(N,seed){
    var Hm=new Float32Array(N*N),Al=new Float32Array(N*N),S=0.42;
    for(var j=0;j<N;j++)for(var i=0;i<N;i++){var id=j*N+i;var xo=((i+.5)/N*2-1)*S, y=-((j+.5)/N*2-1)*S;
      if(Math.abs(xo)>0.42)continue;var yy=amakY(xo);var top=yy[0],bot=yy[1];
      var dt=top-y, db=y-bot; var ua=Math.min(1,Math.abs(xo)/0.42); var sl=0.16*1.25*Math.pow(Math.max(1e-3,1-ua),0.25)/0.42; db/=Math.sqrt(1+sl*sl);
      var a=clamp(Math.min(dt,db)*N/(2*S),0,1); if(a<=0)continue; Al[id]=a;
      var m=Math.min(dt,db), h=0.10;
      if(m<0.03) h=0.10+0.32*sstep(0.03,0.022,m)-0.12*sstep(0.006,0,m);
      /* 당초 줄기 */
      var best=1;for(var c=0;c<amakCurves.length;c++){var P=amakCurves[c];for(var k=0;k<P.length;k+=1){var dd=Math.hypot(xo-P[k][0],y-P[k][1]);if(dd<best)best=dd;}}
      h=Math.max(h,0.10+0.26*Math.exp(-Math.pow(best/0.0085,2)));
      /* 가운데 연꽃 봉오리(세 잎) */
      var cy=(amakY(0)[0]+amakY(0)[1])/2+0.01;
      function petal(cx0,cy0,ang,a1,b1){var dx=xo-cx0,dy=y-cy0,ca=Math.cos(ang),sa=Math.sin(ang);var px1=dx*ca+dy*sa,py1=-dx*sa+dy*ca;var q=(px1*px1)/(a1*a1)+(py1*py1)/(b1*b1);return q<1?0.10+0.26*Math.sqrt(1-q):0;}
      h=Math.max(h,petal(0,cy+0.012,Math.PI/2,0.05,0.022),petal(-0.03,cy-0.002,Math.PI/2+0.75,0.042,0.018),petal(0.03,cy-0.002,Math.PI/2-0.75,0.042,0.018));
      h-=0.05*fbm(xo*14,y*14,seed+3,4);
      Hm[id]=h;}
    return shadeTex(N,Hm,Al,seed,[110,118,129],{hscale:0.10});}

  /* 실루엣(두께 표현용) */
  function silhouette(tex,col){var c=document.createElement('canvas');c.width=tex.width;c.height=tex.height;var g=c.getContext('2d');g.drawImage(tex,0,0);g.globalCompositeOperation='source-in';g.fillStyle=col;g.fillRect(0,0,c.width,c.height);return c;}

  /* 면을 3D 평면에 붙이기(작은 면이라 아핀으로 근사) */

  /* 처마 길이: 그림 구역 안에서 끝나도록 마지막 수키와 줄을 고른다 */
  var X0=-0.235, S1=5.2, X1=(function(){for(var k=1;k<20;k++){var p=proj([k,-0.02,0.03]);if(p[0]>W*0.97)return k-1+0.235;}return 8.235;})();
  var wetMask=document.createElement('canvas');wetMask.width=W;wetMask.height=H;
