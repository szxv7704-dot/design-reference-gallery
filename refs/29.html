매일 운행
접시 테 유약 = 활동

오른쪽 범례를 누르면 레일이 돌아 그 색 접시가 맨 앞에 와서 멈춥니다. 이 동작이 아래 ‘회전 레일 목차’ 장치입니다.

가져갈 장치

회전초밥 가게의 레일과 접시 색 규칙을 빌렸습니다. 사물을 그대로 옮기지 않고, 순환하는 레일과 색 코드 두 가지만 발표 문법으로 씁니다.

회전 레일 목차

목차 항목을 순환 레일 위 접시로 올려 둡니다. 섹션이 바뀔 때마다 레일이 돌아 다음 접시가 앞자리로 와서 멈춥니다. 순서가 있으면서도 해마다 다시 도는 사업을 보여 주기 좋습니다.

응용: 분기별 점검 회의 목차, 연간 순환 행사, 반복 운영되는 연수 과정 소개

접시 색 범례

가게에서 접시 색이 가격을 뜻하듯 색 하나가 영역 하나를 뜻하게 정합니다. 표지에서 정한 4색을 안쪽 장표의 표·도표·아이콘 테두리까지 그대로 이어 써서 범례를 다시 읽지 않게 합니다.

응용: 대상 학교급 구분, 예산 구간, 사업 유형별 색 코딩

접시 계산 결산

식사를 마치면 쌓인 접시를 색별로 세어 계산합니다. 평가 장표에서 영역별 참여 실적을 접시 더미 높이로 보여 주면 막대그래프보다 장면이 남습니다.

응용: 프로그램별 참여 인원, 영역별 집행 실적, 만족도 응답 수 비교

흐름 설계 메모

계획안 목차를 레일 한 바퀴로 짭니다. 마지막 평가가 다음 해 운영으로 다시 이어지는 순환 구조라서 회전 레일과 뜻이 맞습니다.

계획안 구성은유안쪽 장표 아이디어
Ⅰ. 추진 배경레일을 까는 이유오후 돌봄 수요가 몰리는 시간대를 가게 앞 대기 줄 길이로 표현
Ⅱ. 운영 방침레일 운행 규칙같은 시간·같은 교실·담임과 연계라는 세 원칙을 레일 위 표지판 3개로
Ⅲ. 프로그램 구성접시 색 4가지놀이·간식·쉼·특성화 활동 비율을 레일 위 접시 개수로 보여 주는 구성비 장표
Ⅳ. 하루 운영레일 한 바퀴14시~19시 일과를 레일 둘레에 시각 눈금으로 두고 접시가 시간 순으로 지나가게
Ⅴ. 인력·공간레일 안쪽 조리대방과후 전담사·강사·담임이 조리대 안에서 접시를 올리는 역할 분담도
Ⅵ. 안전·위생접시 덮개와 위생 기준급간식 위생·귀가 안전 점검표를 투명 덮개 아래 체크리스트로
Ⅶ. 평가·환류접시 계산영역별 참여 실적을 색별 접시 더미로, 다음 해 메뉴 개선안으로 연결

재현 체크

  • 빌린 대표 장치: ① 조리대를 감싸고 도는 스테인리스 타원 레일과 초승달 판 컨베이어 ② 색이 곧 구분인 접시 ③ 쌓인 접시로 하는 계산 ④ 매장 안 운행 전광판
  • 표지에서는 ①·②만 쓰고 ③은 안쪽 장표 아이디어로, ④는 운행 시간 한 줄로만 남김
  • 조사 출처: nippon.com “Kaiten-zushi”, Japan Travel, Guide to Conveyor Belt Sushi

이미지 저작자·라이선스

  • 사진 사용 없음. 레일·접시·놀잇감은 모두 canvas로 그린 자체 렌더링(좌상단 광원, 금속 측면 음영, 도자기 접시 하이라이트)
  • 실존 매장·브랜드의 로고·명칭 사용 없음
  • 계획안 내용은 가상 예시

팔레트

폰트

  • 제목·본문: Noto Sans KR 900/700/400 (대체: Apple SD Gothic Neo, 맑은 고딕)
  • 전광판 숫자: Orbit (대체: Nanum Gothic Coding, Consolas)

PPT 이식

  • 레일: 둥근 사각형(모서리 최대)을 ‘3차원 회전 → 원근감 위’로 눕히고 테두리 굵게 + 금속 그라데이션. 안쪽에 원목 색 둥근 사각형
  • 접시: 흰 타원 + 색 테두리 타원 두 겹 + 아래 회색 타원(두께), 그림자 ‘오프셋 아래’. 놀잇감은 3D 아이콘이나 실물 누끼 사진으로 통일
  • 회전 레일 목차: 접시 위치만 바꾼 슬라이드를 복제해 ‘모핑’ 전환 → 접시 이름을 !!plate1 식으로 맞추면 레일을 따라 미끄러지듯 이동
  • 계속 도는 움직임: ‘이동 경로 → 사용자 지정 경로’로 레일 둘레를 그리고 ‘반복: 슬라이드 끝까지’

// 평면 좌표(x: 좌우, z: 앞(+)·뒤(-))로 경기장형 레일을 정의하고 간단한 원근 투영으로 그린다.
(function(){
  const cv=document.getElementById('rail'); let g=cv.getContext('2d'); const realG=g;
  const W=1920,H=1080, CX=1180, CY=800;
  const L=1100, R=330, BAND=72, GUARD=16, SKIRT=48;
  const P=2*L+2*Math.PI*R;
  const CAT=['#5E9479','#C98F38','#4D6B96','#A9544E']; // 놀이·간식·쉼·특성화
  const ORDER=[0,1,3,2,0,3,1,2,0,1,2,3,1,0];          // 레일 위 접시 순서(색 섞기)
  const N=ORDER.length, GAP=P/N;
  const reduce=matchMedia('(prefers-reduced-motion: reduce)').matches;

  const k=z=>1+z*0.00045;                // 앞일수록 크게
  function proj(x,z,h){const s=k(z);return [CX+x*s, CY+z*0.40*s-(h||0)*s*0.92, s];}
  // 중심선 위 s 위치의 점과 바깥 법선
  function pt(s){
    s=((s%P)+P)%P;
    if(s<L) return {x:-L/2+s,z:R,nx:0,nz:1};
    s-=L; if(s<Math.PI*R){const a=s/R;return {x:L/2+R*Math.sin(a),z:R*Math.cos(a),nx:Math.sin(a),nz:Math.cos(a)};}
    s-=Math.PI*R; if(s<L) return {x:L/2-s,z:-R,nx:0,nz:-1};
    s-=L; const a=s/R; return {x:-L/2-R*Math.sin(a),z:-R*Math.cos(a),nx:-Math.sin(a),nz:-Math.cos(a)};
  }
  const STEPS=360;
  function ring(d,h){const out=[];for(let i=0;i<=STEPS;i++){const p=pt(i/STEPS*P);out.push(proj(p.x+p.nx*d,p.z+p.nz*d,h));}return out;}
  function pathRing(a,b){g.beginPath();a.forEach((q,i)=>i?g.lineTo(q[0],q[1]):g.moveTo(q[0],q[1]));
    if(b){for(let i=b.length-1;i>=0;i--)g.lineTo(b[i][0],b[i][1]);} g.closePath();}

  // 정적인 부분(조리대·레일 몸통)을 미리 그려 둔다
  const base=document.createElement('canvas');base.width=W;base.height=H;
  function paintBase(ctx){
    // 바탕: 아주 옅은 바닥 그라데이션
    const bgG=g.createLinearGradient(0,0,0,H);bgG.addColorStop(0,'#FFFFFF');bgG.addColorStop(.55,'#FBFAF7');bgG.addColorStop(1,'#F1EEE8');
    g.fillStyle=bgG;g.fillRect(0,0,W,H);
    // 바닥 그림자
    const so=ring(BAND+GUARD+40,-SKIRT);
    g.save();g.filter='blur(28px)';g.fillStyle='rgba(60,50,40,.22)';pathRing(so.map(q=>[q[0]+18,q[1]+26]));g.fill();g.restore();
    // 바깥 스테인리스 치마(옆면): 앞쪽을 향한 면만
    const dO=BAND+GUARD, top=ring(dO,0), bot=ring(dO,-SKIRT);
    for(let i=0;i<STEPS;i++){
      const p=pt((i+.5)/STEPS*P); if(p.nz<-0.05) continue;
      const lum=0.72+0.18*(-p.nx)+0.08*p.nz; // 좌상단 광원
      const t=top[i],t2=top[i+1],b=bot[i],b2=bot[i+1];
      const gr=g.createLinearGradient(0,t[1],0,b[1]);
      const v=Math.round(255*Math.min(1,lum));
      gr.addColorStop(0,`rgb(${v},${v+2>255?255:v+2},${v+5>255?255:v+5})`);
      gr.addColorStop(.18,`rgb(${v-38},${v-34},${v-30})`);
      gr.addColorStop(1,`rgb(${v-70},${v-66},${v-60})`);
      g.fillStyle=gr;g.beginPath();g.moveTo(t[0],t[1]);g.lineTo(t2[0],t2[1]);g.lineTo(b2[0],b2[1]+.5);g.lineTo(b[0]-.5,b[1]+.5);g.closePath();g.fill();
    }
    // 바닥 경계선
    g.strokeStyle='rgba(40,46,54,.35)';g.lineWidth=2;g.beginPath();
    for(let i=0;i<=STEPS;i++){const p=pt(i/STEPS*P);const q=bot[i];if(p.nz<-0.05){g.moveTo(q[0],q[1]);continue;}g.lineTo(q[0],q[1]);}g.stroke();
    // 바깥 가드 윗면(금속)
    const oG=ring(BAND+GUARD,0), oB=ring(BAND,0), iB=ring(-BAND,0), iG=ring(-BAND-GUARD,0);
    let mg=g.createLinearGradient(0,600,0,1000);mg.addColorStop(0,'#E9EDF1');mg.addColorStop(.5,'#D3D9DF');mg.addColorStop(1,'#F4F6F8');
    g.fillStyle=mg;pathRing(oG,oB);g.fill();
    // 컨베이어 판(밝은 회백)
    const bG=g.createLinearGradient(0,620,0,1000);bG.addColorStop(0,'#E6EAEE');bG.addColorStop(1,'#F5F7F9');
    g.fillStyle=bG;pathRing(oB,iB);g.fill();
    // 안쪽 가드
    g.fillStyle=mg;pathRing(iB,iG);g.fill();
    // 안쪽 조리대(원목)
    const wG=g.createLinearGradient(CX-700,700,CX+700,900);wG.addColorStop(0,'#F0E0C3');wG.addColorStop(.5,'#E8D3AE');wG.addColorStop(1,'#EBD8B6');
    g.fillStyle=wG;pathRing(iG);g.fill();
    // 원목 결: 옅은 가로 결선
    g.save();pathRing(iG);g.clip();
    for(let i=0;i<46;i++){const y=700+i*5.2+Math.sin(i*1.7)*2;g.strokeStyle=`rgba(150,105,55,${0.05+0.05*((i*7)%3)/2})`;g.lineWidth=1+((i*5)%3)*.5;
      g.beginPath();for(let x=CX-760;x<=CX+760;x+=40){g.lineTo(x,y+Math.sin(x*.004+i)*3);}g.stroke();}
    g.restore();
    // 가장자리 하이라이트·그늘선
    g.lineWidth=2.2;
    g.strokeStyle='rgba(255,255,255,.95)';pathRing(oG);g.stroke();
    g.strokeStyle='rgba(90,100,112,.45)';g.lineWidth=1.6;pathRing(oB);g.stroke();
    g.strokeStyle='rgba(90,100,112,.45)';pathRing(iB);g.stroke();
    g.strokeStyle='rgba(255,255,255,.9)';g.lineWidth=1.6;pathRing(iG);g.stroke();
    g.strokeStyle='rgba(120,90,50,.35)';g.lineWidth=1.2;pathRing(ring(-BAND-GUARD-2,0));g.stroke();
  }
  // ---- 접시와 놀잇감 ----
  function plate(x,y,s,col,lift,glow){
    const c=g, r=58*s, ry=r*0.40, th=9*s;
    y-=lift*s;
    // 벨트 위 그림자
    c.save();c.globalAlpha=Math.max(.25,1-lift/40);
    const sg=c.createRadialGradient(x+10*s,y+lift*s+th+6*s,4,x+10*s,y+lift*s+th+6*s,r*1.15);
    sg.addColorStop(0,'rgba(40,50,60,.32)');sg.addColorStop(1,'rgba(40,50,60,0)');
    c.fillStyle=sg;c.beginPath();c.ellipse(x+10*s,y+lift*s+th+6*s,r*1.15,ry*1.25,0,0,7);c.fill();c.restore();
    if(glow){c.save();c.strokeStyle=col;c.globalAlpha=.45;c.lineWidth=3*s;c.beginPath();c.ellipse(x,y+th/2,r*1.18,ry*1.22,0,0,7);c.stroke();c.restore();}
    // 굽(아래 두께)
    const side=c.createLinearGradient(x-r,0,x+r,0);side.addColorStop(0,'#F4F5F6');side.addColorStop(.6,'#D9DDE1');side.addColorStop(1,'#BFC5CB');
    c.fillStyle=side;c.beginPath();c.ellipse(x,y+th,r*.98,ry*.98,0,0,Math.PI);c.lineTo(x-r,y);c.ellipse(x,y,r,ry,0,Math.PI,0,true);c.closePath();c.fill();
    // 윗면(도자기)
    const tg=c.createRadialGradient(x-r*.4,y-ry*.6,2,x,y,r*1.05);tg.addColorStop(0,'#FFFFFF');tg.addColorStop(.7,'#F6F7F8');tg.addColorStop(1,'#E2E6EA');
    c.fillStyle=tg;c.beginPath();c.ellipse(x,y,r,ry,0,0,7);c.fill();
    // 색 테두리 띠
    c.strokeStyle=col;c.lineWidth=r*.13;c.beginPath();c.ellipse(x,y,r*.86,ry*.86,0,0,7);c.stroke();
    // 띠 위 광택
    c.strokeStyle='rgba(255,255,255,.55)';c.lineWidth=r*.035;c.beginPath();c.ellipse(x,y,r*.86,ry*.86,0,Math.PI*1.05,Math.PI*1.55);c.stroke();
    // 안쪽 오목
    const wg=c.createRadialGradient(x+r*.1,y+ry*.15,2,x,y,r*.62);wg.addColorStop(0,'rgba(255,255,255,0)');wg.addColorStop(1,'rgba(150,160,170,.18)');
    c.fillStyle=wg;c.beginPath();c.ellipse(x,y,r*.62,ry*.62,0,0,7);c.fill();
    return y; // 윗면 중심 y
  }
  function cube(c,x,y,a,top,left,right){ // x,y: 윗면 중심, a: 한 변
    const hx=a*.62,hy=a*.25,h=a*.72;
    c.fillStyle=top;c.beginPath();c.moveTo(x,y-hy);c.lineTo(x+hx,y);c.lineTo(x,y+hy);c.lineTo(x-hx,y);c.closePath();c.fill();
    c.fillStyle=left;c.beginPath();c.moveTo(x-hx,y);c.lineTo(x,y+hy);c.lineTo(x,y+hy+h);c.lineTo(x-hx,y+h);c.closePath();c.fill();
    c.fillStyle=right;c.beginPath();c.moveTo(x+hx,y);c.lineTo(x,y+hy);c.lineTo(x,y+hy+h);c.lineTo(x+hx,y+h);c.closePath();c.fill();
    c.strokeStyle='rgba(255,255,255,.55)';c.lineWidth=1;c.beginPath();c.moveTo(x-hx,y);c.lineTo(x,y-hy);c.lineTo(x+hx,y);c.stroke();
  }
  function item(cat,x,y,s){
    const c=g;
    if(cat===0){ // 놀이: 원목 블록 쌓기
      const a=30*s;
      cube(c,x-a*.55,y-a*.55,a,'#F2DDB6','#D8BC8C','#BC9C6A');
      cube(c,x+a*.55,y-a*.55,a,'#C7DACB','#93B39E','#6F927C');
      cube(c,x,y-a*1.45,a,'#F2DDB6','#D8BC8C','#BC9C6A');
      // 원목 결과 모서리 마모
      c.strokeStyle='rgba(140,100,55,.28)';c.lineWidth=.8*s;
      for(const [bx,by] of [[x-a*.55,y-a*.55],[x,y-a*1.45]]) for(let k=1;k<4;k++){c.beginPath();c.moveTo(bx-a*.62+k*a*.12,by-k*a*.04);c.quadraticCurveTo(bx,by+a*.1-k*a*.05,bx+a*.62-k*a*.1,by+k*a*.03);c.stroke();}
    } else if(cat===1){ // 간식: 귤
      const rr=24*s, cy=y-rr*.85;
      // 귤: 위아래가 눌린 몸통, 껍질 기공, 꼭지(별 모양 받침), 잎맥 있는 잎
      const og=c.createLinearGradient(x-rr,cy-rr,x+rr,cy+rr);
      og.addColorStop(0,'#F3B25E');og.addColorStop(.6,'#E08A2E');og.addColorStop(1,'#B9661D');
      c.fillStyle=og;c.beginPath();c.ellipse(x,cy,rr*1.08,rr*.88,0,0,7);c.fill();
      c.save();c.beginPath();c.ellipse(x,cy,rr*1.08,rr*.88,0,0,7);c.clip();
      for(let k=0;k<70;k++){const a=k*2.399, d=Math.sqrt(k/70);const px=x+Math.cos(a)*d*rr, py=cy+Math.sin(a)*d*rr*.82;
        c.fillStyle=k%3?'rgba(150,72,10,.18)':'rgba(255,220,160,.25)';c.beginPath();c.arc(px,py,.9*s,0,7);c.fill();}
      const sh=c.createLinearGradient(x,cy-rr,x,cy+rr);sh.addColorStop(0,'rgba(255,240,210,.22)');sh.addColorStop(1,'rgba(90,40,0,.18)');
      c.fillStyle=sh;c.fillRect(x-rr*1.2,cy-rr,rr*2.4,rr*2);c.restore();
      c.fillStyle='#5C6B3A';c.beginPath();
      for(let k=0;k<10;k++){const a=k*Math.PI/5-Math.PI/2, rad=k%2?rr*.07:rr*.17;const px=x+rr*.05+Math.cos(a)*rad, py=cy-rr*.84+Math.sin(a)*rad*.45;k?c.lineTo(px,py):c.moveTo(px,py);}c.closePath();c.fill();
      c.fillStyle='#6E8F52';c.beginPath();c.moveTo(x+rr*.08,cy-rr*.9);c.quadraticCurveTo(x+rr*.5,cy-rr*1.35,x+rr*1.0,cy-rr*1.05);c.quadraticCurveTo(x+rr*.5,cy-rr*.82,x+rr*.08,cy-rr*.9);c.fill();
      c.strokeStyle='rgba(235,240,215,.7)';c.lineWidth=.8*s;c.beginPath();c.moveTo(x+rr*.12,cy-rr*.92);c.quadraticCurveTo(x+rr*.5,cy-rr*1.1,x+rr*.95,cy-rr*1.04);c.stroke();
    } else if(cat===2){ // 쉼: 작은 베개
      const w=46*s,h=22*s,cy=y-h*.75;
      const pg=c.createLinearGradient(x-w,cy-h,x+w*.6,cy+h);pg.addColorStop(0,'#F1F3F6');pg.addColorStop(.55,'#C9D3E0');pg.addColorStop(1,'#8C9DB5');
      c.fillStyle=pg;c.beginPath();
      c.moveTo(x-w*.8,cy-h*.6);
      c.quadraticCurveTo(x,cy-h*1.25,x+w*.8,cy-h*.6);
      c.quadraticCurveTo(x+w*.98,cy,x+w*.8,cy+h*.6);
      c.quadraticCurveTo(x,cy+h*1.15,x-w*.8,cy+h*.6);
      c.quadraticCurveTo(x-w*.98,cy,x-w*.8,cy-h*.6);c.fill();
      c.strokeStyle='rgba(255,255,255,.7)';c.lineWidth=1.5*s;c.beginPath();c.moveTo(x-w*.55,cy-h*.62);c.quadraticCurveTo(x-w*.1,cy-h*.9,x+w*.35,cy-h*.72);c.stroke();
      // 가장자리 박음질과 가운데 단추
      c.setLineDash([2.2*s,2.2*s]);c.strokeStyle='rgba(77,107,150,.55)';c.lineWidth=.9*s;c.beginPath();
      c.moveTo(x-w*.7,cy-h*.45);c.quadraticCurveTo(x,cy-h*1.02,x+w*.7,cy-h*.45);c.quadraticCurveTo(x+w*.84,cy,x+w*.7,cy+h*.45);
      c.quadraticCurveTo(x,cy+h*.95,x-w*.7,cy+h*.45);c.quadraticCurveTo(x-w*.84,cy,x-w*.7,cy-h*.45);c.stroke();c.setLineDash([]);
      c.fillStyle='#4D6B96';c.beginPath();c.arc(x+w*.05,cy+h*.05,2.4*s,0,7);c.fill();
    } else { // 특성화 활동: 작은 북
      const rx=24*s, ry=rx*.38, h=24*s, top=y-h-ry*.3;
      const sg=c.createLinearGradient(x-rx,0,x+rx,0);sg.addColorStop(0,'#D38A83');sg.addColorStop(.45,'#A9544E');sg.addColorStop(1,'#78352F');
      c.fillStyle=sg;c.beginPath();c.ellipse(x,top+h,rx,ry,0,0,Math.PI);c.lineTo(x-rx,top);c.ellipse(x,top,rx,ry,0,Math.PI,0,true);c.closePath();c.fill();
      // 줄 매듭(지그재그)
      c.strokeStyle='rgba(255,236,200,.95)';c.lineWidth=1.6*s;c.beginPath();
      for(let i=0;i<=8;i++){const a=Math.PI-i*Math.PI/8;const xx=x+Math.cos(a)*rx*.98;const yy=(i%2?top+h-ry*.2:top+ry*.5)+Math.sin(a)*ry*.95;i?c.lineTo(xx,yy):c.moveTo(xx,yy);}c.stroke();
      // 금색 테
      c.strokeStyle='#B9955A';c.lineWidth=2.4*s;c.beginPath();c.ellipse(x,top+h,rx,ry,0,0,Math.PI);c.stroke();
      const hg=c.createRadialGradient(x-rx*.3,top-ry*.3,1,x,top,rx);hg.addColorStop(0,'#FFFFFF');hg.addColorStop(1,'#E9E4DA');
      c.fillStyle=hg;c.beginPath();c.ellipse(x,top,rx,ry,0,0,7);c.fill();
      c.strokeStyle='#B9955A';c.lineWidth=2.4*s;c.stroke();
    }
  }

  // 정적인 바탕을 오프스크린 캔버스에 한 번만 그린다
  function runBase(){
    const bctx=base.getContext('2d');
    g = bctx; paintBase(bctx); g = realG;
  }

  // ---- 상태 ----
  let off=0, last=performance.now(), speed=reduce?0:36; // px/초
  let target=null, sel=-1, selUntil=0, lift=[];
  for(let i=0;i<N;i++) lift[i]=0;
  const FRONT=L/2; // 앞쪽 직선 한가운데

  function frame(now){
    const dt=Math.min(.05,(now-last)/1000); last=now;
    if(target){
      const d=target.to-off, step=Math.sign(d)*Math.max(Math.min(Math.abs(d),Math.abs(d)*dt*3.2),Math.min(Math.abs(d),40*dt));
      off+=step; if(Math.abs(target.to-off)<.5){off=target.to;target=null;selUntil=now+(reduce?1e9:2600);}
    } else if(now>selUntil){ sel=-1; off+=speed*dt; }
    for(let i=0;i<N;i++){const want=(i===sel&&!target)?22:0; lift[i]+= (want-lift[i])*Math.min(1,dt*8); if(reduce) lift[i]=want;}
    draw();
    if(!reduce||target) requestAnimationFrame(frame); else running=false;
  }
  let running=true;
  function kick(){ if(!running){running=true;last=performance.now();requestAnimationFrame(frame);} }

  function draw(){
    const c=g;
    c.drawImage(base,0,0);
    // 컨베이어 판 이음매(초승달 판): 레일과 함께 이동
    const SL=40;
    c.save();
    for(let s=(off%SL);s<P;s+=SL){
      const p=pt(s);
      const a=proj(p.x+p.nx*(BAND-2),p.z+p.nz*(BAND-2),0), b=proj(p.x-p.nx*(BAND-2),p.z-p.nz*(BAND-2),0);
      // 바깥쪽으로 휘는 초승달 선
      const p2=pt(s+9), m=proj(p2.x,p2.z,0);
      c.strokeStyle='rgba(120,132,146,.38)';c.lineWidth=1.6*a[2];
      c.beginPath();c.moveTo(a[0],a[1]);c.quadraticCurveTo(m[0],m[1],b[0],b[1]);c.stroke();
      c.strokeStyle='rgba(255,255,255,.8)';c.lineWidth=1*a[2];
      c.beginPath();c.moveTo(a[0]+1.5,a[1]+1);c.quadraticCurveTo(m[0]+1.5,m[1]+1,b[0]+1.5,b[1]+1);c.stroke();
    }
    c.restore();
    // 접시: 먼 것부터
    const list=[];
    for(let i=0;i<N;i++){const p=pt(off+i*GAP);list.push({i,p});}
    list.sort((A,B)=>A.p.z-B.p.z);
    for(const o of list){
      const q=proj(o.p.x,o.p.z,6);
      const cat=ORDER[o.i];
      const top=plate(q[0],q[1],q[2],CAT[cat],lift[o.i],o.i===sel);
      item(cat,q[0],top,q[2]);
    }
  }

  // 범례 칩: 누르면 그 색 접시 중 가장 가까운 것이 앞자리로
  document.querySelectorAll('.chip').forEach(btn=>btn.addEventListener('click',()=>{
    const cat=+btn.dataset.cat; let best=null;
    for(let i=0;i<N;i++){ if(ORDER[i]!==cat) continue;
      const cur=((off+i*GAP)%P+P)%P; let d=FRONT-cur; d=((d%P)+P)%P; // 앞으로만 돈다
      if(!best||d<best.d) best={i,d}; }
    sel=best.i; target={to:off+best.d};
    if(reduce){off=target.to;target=null;}
    selUntil=Infinity; kick(); if(reduce){draw();}
  }));

  function start(){ runBase(); draw(); requestAnimationFrame(t=>{last=t;frame(t);}); }
  if(document.fonts&&document.fonts.ready) document.fonts.ready.then(start); else start();
})();
</script>
