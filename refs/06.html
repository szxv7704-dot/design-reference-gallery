완성 상태

이 움직임이 보여 주는 것

계획안 본문에는 세부 과제가 01부터 12까지 한 줄로 적혀 있습니다. 카드 딜러가 패를 테이블에 던지듯 그 순서 그대로 펼쳐 놓고, 영역 틀이 자리를 편 뒤 같은 성격의 카드만 한 더미씩 모입니다.

모이는 순간 회색 카드가 영역 색으로 바뀌고, 카드 번호는 그대로 남습니다. 그래서 멈춘 화면에서도 “01·04·07·10이 한 영역”처럼 원래 순서와 분류가 동시에 읽힙니다.

계획안의 다른 부분에 쓰기

  • 대상별 지원 유형: 지원 사례 카드를 “유아 / 초등 / 특수” 세 더미로 모으기
  • 예산 편성 내역: 사업 카드를 “인건비 / 운영비 / 시설비”로 묶고 영역 머리에 합계를 붙이기
  • 기존안 대 개선안: 과제를 “유지 / 통합 / 신설” 세 더미로 가르며 달라진 점 보여 주기

<script>
(function () {
  /* ── 단계 목록: 한 줄 = 클릭 한 번. 순서를 바꾸거나 줄을 지우면 재생 순서가 바뀝니다 ──
     act: deal(카드 던지기) / lanes(영역 틀 펼치기) / sort(해당 영역으로 모으기) / count(개수 표시) */
  var RD_STEPS = [
    { act: 'deal',  label: '계획안 순서대로 과제 12개가 던져진다' },
    { act: 'lanes', label: '묶을 영역 3개가 자리를 편다' },
    { act: 'sort', area: 'a', label: '수업·교육과정 과제 4개가 모인다' },
    { act: 'sort', area: 'b', label: '인력·전문성 과제 4개가 모인다' },
    { act: 'sort', area: 'c', label: '인식·문화 과제 4개가 모인다' },
    { act: 'count', label: '영역마다 과제 수가 붙는다' }
  ];
  /* 흩어진 자리: [가로 0~1, 세로 0~1, 기울기(도)] — data-no 1번부터 순서대로, 표 영역 기준 */
  var RD_SCATTER = [
    [.13, .17, -5], [.38, .21, 3], [.62, .15, -2], [.87, .20, 6],
    [.12, .50, 4], [.37, .47, -4], [.63, .52, 5], [.88, .48, -3],
    [.14, .83, -3], [.38, .80, 6], [.62, .84, -6], [.86, .81, 2]
  ];

  function init(root) {
    if (root.dataset.rdReady) return;
    root.dataset.rdReady = '1';
    var table = root.querySelector('.rd-sort__table');
    var cards = Array.prototype.slice.call(root.querySelectorAll('.rd-sort__card'));
    var heads = root.querySelectorAll('.rd-sort__head');
    var beds = root.querySelectorAll('.rd-sort__bed');
    var counts = root.querySelectorAll('.rd-sort__count');
    var caption = root.querySelector('[data-rd="caption"]');
    var btnStep = root.querySelector('[data-rd="step"]');
    var reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
    var N = RD_STEPS.length, cur = N, runId = 0;

    cards.sort(function (x, y) { return x.dataset.no - y.dataset.no; });

    function v(name) { return getComputedStyle(root).getPropertyValue(name).trim(); }
    function ms(name) { return parseFloat(v(name)) || 0; }

    /* 요소를 to 상태로 두고, 시간이 있으면 from→to로 움직인다 */
    function tween(el, from, to, dur, delay, easing) {
      for (var k in to) el.style[k] = to[k];
      if (!dur || reduce || !el.animate) return;
      el.animate([from, to], { duration: dur, delay: delay || 0, easing: easing || 'ease-out', fill: 'backwards' });
    }
    /* 카드의 원래(최종) 중심 좌표: 표 영역 기준, 변형 무시 */
    function home(el) {
      var x = 0, y = 0, n = el;
      while (n && n !== table) { x += n.offsetLeft; y += n.offsetTop; n = n.offsetParent; }
      return { x: x + el.offsetWidth / 2, y: y + el.offsetHeight / 2 };
    }
    function scatterT(card) {
      var p = RD_SCATTER[(card.dataset.no - 1) % RD_SCATTER.length], h = home(card);
      return 'translate(' + (p[0] * table.clientWidth - h.x) + 'px,' + (p[1] * table.clientHeight - h.y) + 'px) rotate(' + p[2] + 'deg) scale(' + v('--rd-scatter-scale') + ')';
    }
    function deckT(card) {
      var p = RD_SCATTER[(card.dataset.no - 1) % RD_SCATTER.length], h = home(card);
      return 'translate(' + (table.clientWidth / 2 - h.x) + 'px,' + (table.clientHeight * 1.45 - h.y) + 'px) rotate(' + (p[2] * 4) + 'deg) scale(' + v('--rd-scatter-scale') + ')';
    }
    var HOME = 'translate(0px,0px) rotate(0deg) scale(1)';

    function reset() {
      cards.forEach(function (c) { c.classList.add('is-loose'); c.style.transitionDelay = '0ms'; c.style.opacity = '0'; c.style.transform = deckT(c); });
      heads.forEach(function (h) { h.style.clipPath = 'inset(0 100% 0 0)'; });
      beds.forEach(function (b) { b.style.opacity = '0'; });
      counts.forEach(function (c) { c.style.opacity = '0'; c.style.transform = 'scale(.3)'; });
    }

    /* 단계 i를 실행하고 걸리는 시간(ms)을 돌려준다. anim=false면 즉시 적용 */
    function doStep(i, anim) {
      var s = RD_STEPS[i], st = ms('--rd-stagger-ms'), bounce = v('--rd-bounce');
      if (s.act === 'deal') {
        var d = anim ? ms('--rd-deal-ms') : 0;
        cards.forEach(function (c, j) {
          tween(c, { opacity: 0, transform: deckT(c) }, { opacity: 1, transform: scatterT(c) }, d, j * st, bounce);
        });
        return d + (cards.length - 1) * st;
      }
      if (s.act === 'lanes') {
        var d2 = anim ? 420 : 0;
        heads.forEach(function (h, j) { tween(h, { clipPath: 'inset(0 100% 0 0)' }, { clipPath: 'inset(0 0% 0 0)' }, d2, j * 110, 'cubic-bezier(.6,0,.2,1)'); });
        beds.forEach(function (b, j) { tween(b, { opacity: 0 }, { opacity: 1 }, d2, 80 + j * 110); });
        return d2 + 2 * 110;
      }
      if (s.act === 'sort') {
        var d3 = anim ? ms('--rd-sort-ms') : 0, group = root.querySelectorAll('[data-area="' + s.area + '"] .rd-sort__card');
        Array.prototype.forEach.call(group, function (c, j) {
          var from = { opacity: 1, transform: c.style.transform };
          c.style.transitionDelay = (anim ? j * st + d3 * .35 : 0) + 'ms';
          c.classList.remove('is-loose');
          tween(c, from, { opacity: 1, transform: HOME }, d3, j * st, bounce);
        });
        return d3 + (group.length - 1) * st;
      }
      if (s.act === 'count') {
        var d4 = anim ? 420 : 0;
        counts.forEach(function (c, j) { tween(c, { opacity: 0, transform: 'scale(.3)' }, { opacity: 1, transform: 'scale(1)' }, d4, j * 90, bounce); });
        return d4 + 2 * 90;
      }
      return 0;
    }

    function cancelAll() {
      root.querySelectorAll('.rd-sort__card, .rd-sort__head, .rd-sort__bed, .rd-sort__count').forEach(function (el) {
        if (el.getAnimations) el.getAnimations().forEach(function (a) { a.cancel(); });
      });
    }
    /* 0~n단계까지 즉시 적용 */
    function go(n) {
      cancelAll();
      root.classList.add('rd-sort--instant');
      reset();
      for (var i = 0; i < n; i++) doStep(i, false);
      void root.offsetWidth;
      root.classList.remove('rd-sort--instant');
      cur = n; label();
    }
    function label() {
      caption.textContent = cur === 0 ? '빈 장표' : (cur + '/' + N + ' · ' + RD_STEPS[cur - 1].label);
      btnStep.textContent = cur >= N ? '한 단계씩 (처음부터)' : '한 단계씩 (' + (cur + 1) + '/' + N + ')';
    }
    function next() {
      runId++;
      go(cur >= N ? 0 : cur);
      doStep(cur, true); cur++; label();
    }
    function playAll() {
      var id = ++runId;
      if (reduce) { go(N); return; }
      go(0);
      var i = 0;
      (function tick() {
        if (id !== runId || i >= N) return;
        var t = doStep(i, true); i++; cur = i; label();
        setTimeout(tick, t + ms('--rd-gap-ms'));
      })();
    }

    root.querySelector('[data-rd="replay"]').addEventListener('click', playAll);
    btnStep.addEventListener('click', next);
    root.querySelector('.rd-sort__slide').addEventListener('keydown', function (e) {
      if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'PageDown') { e.preventDefault(); next(); }
    });
    /* 크기가 바뀌면 현재 단계를 다시 맞춘다 */
    var lastW = table.clientWidth;
    if (window.ResizeObserver) new ResizeObserver(function () {
      if (Math.abs(table.clientWidth - lastW) < 1) return;
      lastW = table.clientWidth; runId++; go(cur);
    }).observe(table);

    if (root.dataset.autoplay === 'true' && !reduce) playAll(); else go(N);
  }

  function boot() { document.querySelectorAll('.rd-sort').forEach(init); }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', boot); else boot();
})();
</script>
  • 내용: 영역 이름은 .rd-sort__head의 글자, 과제는 <li class="rd-sort__card" data-no="…">. data-no는 계획안 기재 순서이고 흩어질 때 이 순서로 놓입니다.
  • 색·타이밍: 맨 위 .rd-sort { --rd-a … --rd-bounce } 변수만 고치면 됩니다. 튕김이 과하면 1.35를 1.1로.
  • 클릭 순서: 스크립트 맨 위 RD_STEPS 배열 한 줄이 클릭 한 번입니다. 영역을 한꺼번에 모으려면 sort 세 줄을 지우고 순서를 바꿔 보세요.
  • 과제 수가 다를 때: 카드 개수에 맞춰 RD_SCATTER 좌표를 늘리거나 줄이고, 레인당 카드 수가 4가 아니면 grid-template-rows: repeat(4, 1fr)의 4를 바꿉니다.
  • 자동 재생 끄기: data-autoplay="false"로 두면 완성 상태로 시작하고 버튼으로만 재생합니다.

장표 네 장 만드는 법

① 장표 1: 최종 배치를 먼저 만든 뒤 카드 12장을 흩어진 자리로 옮기고 5° 안팎으로 기울이고 74%로 줄입니다. 채우기는 흰색, 번호는 회색. 클릭 1·2 애니메이션은 이 장에만 넣습니다.

② 선택 창(Alt+F10)에서 카드 이름을 !!과제01…!!과제12로 바꿉니다. 이름 앞 !!가 붙으면 모핑이 장이 달라도 같은 개체로 짝을 맞춥니다.

③ 장표 1을 복제해 장표 2에서는 수업·교육과정 4장만 제자리·원래 크기·기울기 0°·영역 색으로 바꿉니다. 장표 3, 4도 같은 방식으로 한 영역씩 더 옮깁니다. 장표 2~4에 화면 전환-모핑을 적용합니다.

똑같이 안 되는 부분과 근사 방법

모일 때 튕김: 모핑에는 오버슈트가 없습니다. 모핑 뒤 해당 카드 4장에 강조-크게/작게(105%, 자동 반대로, 0.2초)를 “이전 효과 다음에”로 붙이면 착지감이 비슷해집니다.
카드 사이 시간차: 모핑은 4장이 동시에 움직입니다. 시간차가 꼭 필요하면 모핑 대신 카드마다 이동 경로(선)+강조-회전+강조-크게/작게를 “이전 효과와 함께”로 묶고 지연을 0.06초씩 늘립니다. 손이 많이 가므로 기본은 모핑을 권합니다.
