01동작이 설명하는 정보 구조

계획안의 추진 체계는 교육청 → 지원 영역 4개 → 영역별 세부 과제 3개의 3단 구조입니다. 장표는 전체 도면을 먼저 보여 주고, 클릭마다 한 영역으로 카메라가 들어가 접혀 있던 세부 설명을 펼칩니다.

확대 중에도 나머지 영역은 흐릿하게 남고, 오른쪽 아래 키플랜(건축 도면의 위치 안내도)에 지금 보는 범위가 표시됩니다. 영역에서 영역으로 옮길 때는 한 번 전체로 빠졌다가 들어가므로 두 영역의 상대 위치가 움직임으로 읽힙니다. 마지막엔 전체 도면으로 돌아와 멈춥니다.

오른쪽 위 배율 표시(×1.0 → ×2.1)와 영역 기호(A~D, 세부 A-1~D-3)는 장식이 아니라 지금 어느 깊이, 어느 위치에 있는지를 알려 주는 정보입니다.

  • 조직도·업무 분장과 → 팀 → 담당자. 팀 단위로 들어가 담당 업무를 펼침
  • 예산 편성 구조정책사업 → 단위사업 → 세부사업. 금액표 대신 위치로 설명
  • 권역별 현황시·군 지도 → 권역 → 기관 목록. 키플랜이 그대로 지도가 됨

<script>
(function () {
  // 단계 목록: 이 배열만 고치면 순서·라벨이 바뀝니다. zone은 data-key 값, null이면 전체 도면
  const ZM_STEPS = [
    { zone: null, label: '전체 도면' },
    { zone: 'A',  label: '상세 A · 놀이 환경' },
    { zone: 'B',  label: '상세 B · 교사 역량' },
    { zone: 'C',  label: '상세 C · 기록·평가' },
    { zone: 'D',  label: '상세 D · 가정·지역 연계' },
    { zone: null, label: '전체 도면 · 네 영역 위치 확인' }
  ];
  const FIT = 0.84; // 확대 시 영역이 화면을 채우는 비율(0~1)

  document.querySelectorAll('[data-zm]').forEach(function (root) {
    if (root.dataset.zmReady) return;
    root.dataset.zmReady = '1';

    const slide = root.querySelector('.zm-slide');
    const view  = root.querySelector('.zm-view');
    const world = root.querySelector('.zm-world');
    const zones = Array.from(root.querySelectorAll('.zm-zone'));
    const key   = root.querySelector('.zm-key');
    const meterX  = root.querySelector('.zm-meter__x');
    const meterLb = root.querySelector('.zm-meter__lb');
    const status  = root.querySelector('.zm-ctrl__st');
    const reduce  = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    const ms = function (name) { return reduce ? 0 : parseFloat(getComputedStyle(root).getPropertyValue(name)) || 0; };

    // 영역 위치(월드 % 기준)를 읽어 키플랜을 그림
    function rectOf(el) {
      return { x: el.offsetLeft / world.offsetWidth * 100, y: el.offsetTop / world.offsetHeight * 100,
               w: el.offsetWidth / world.offsetWidth * 100, h: el.offsetHeight / world.offsetHeight * 100 };
    }
    key.insertAdjacentHTML('beforeend', '<div class="zm-key__h"></div>');
    const keyZones = {};
    zones.forEach(function (z) {
      const d = document.createElement('div');
      d.className = 'zm-key__z';
      d.style.setProperty('--c', getComputedStyle(z).getPropertyValue('--c'));
      key.appendChild(d); keyZones[z.dataset.key] = d;
    });
    const frame = document.createElement('div');
    frame.className = 'zm-key__f'; key.appendChild(frame);
    function layoutKey() {
      zones.forEach(function (z) {
        const r = rectOf(z), d = keyZones[z.dataset.key];
        d.style.left = r.x + '%'; d.style.top = r.y + '%'; d.style.width = r.w + '%'; d.style.height = r.h + '%';
      });
    }

    // 카메라 계산: 영역을 화면 가운데 놓는 scale·translate(% 단위라 크기와 무관)
    function camFor(zoneKey) {
      if (!zoneKey) return { s: 1, tx: 0, ty: 0 };
      const z = zones.find(function (el) { return el.dataset.key === zoneKey; });
      const r = rectOf(z);
      const s = Math.min(FIT * 100 / r.w, FIT * 100 / r.h);
      return { s: s, tx: 50 - s * (r.x + r.w / 2), ty: 50 - s * (r.y + r.h / 2) };
    }
    const tf = function (c) { return 'translate(' + c.tx + '%,' + c.ty + '%) scale(' + c.s + ')'; };

    let idx = ZM_STEPS.length - 1;          // 처음엔 완성 상태(마지막 단계 = 전체 도면)
    let cam = camFor(null);
    let timers = [];
    const clearTimers = function () { timers.forEach(clearTimeout); timers = []; };

    function go(i) {
      idx = i;
      const step = ZM_STEPS[i];
      const next = camFor(step.zone);
      const dur = ms('--zm-cam');
      // 영역→영역 이동은 중간에 전체로 한 번 빠져나왔다 들어감(지도 앱의 이동 방식)
      let frames = [{ transform: tf(cam) }, { transform: tf(next) }];
      if (cam.s > 1 && next.s > 1) {
        const mid = 1.1; // 중간 배율: 두 영역의 가운데를 보며 거의 전체로
        const cx = ((50 - cam.tx) / cam.s + (50 - next.tx) / next.s) / 2;
        const cy = ((50 - cam.ty) / cam.s + (50 - next.ty) / next.s) / 2;
        frames.splice(1, 0, { transform: tf({ s: mid, tx: 50 - mid * cx, ty: 50 - mid * cy }), offset: .5 });
      }
      world.getAnimations().forEach(function (a) { a.cancel(); });
      world.style.transform = tf(next);
      if (dur) world.animate(frames, { duration: dur, easing: 'cubic-bezier(.6,0,.25,1)' });
      cam = next;

      root.classList.toggle('is-zoomed', !!step.zone);
      zones.forEach(function (z) {
        const on = z.dataset.key === step.zone;
        z.classList.toggle('is-on', on);
        keyZones[z.dataset.key].classList.toggle('is-on', on);
      });
      // 키플랜의 카메라 틀 = 지금 화면에 보이는 범위
      frame.style.left = (-next.tx / next.s) + '%'; frame.style.top = (-next.ty / next.s) + '%';
      frame.style.width = (100 / next.s) + '%'; frame.style.height = (100 / next.s) + '%';

      meterX.textContent = '×' + next.s.toFixed(1);
      meterLb.textContent = step.label;
      status.textContent = '단계 ' + i + ' / ' + (ZM_STEPS.length - 1) + ' · ' + step.label;
    }

    function play() {
      clearTimers();
      if (reduce) { go(ZM_STEPS.length - 1); return; }
      go(0);
      const hold = ms('--zm-hold');
      for (let i = 1; i < ZM_STEPS.length; i++) timers.push(setTimeout(function () { go(i); }, hold * i - hold * .6));
    }
    function stepOnce() {
      clearTimers();
      go(idx >= ZM_STEPS.length - 1 ? 1 : idx + 1); // 끝에서 누르면 첫 확대부터
    }

    layoutKey();
    go(idx);
    root.querySelector('.zm-replay').addEventListener('click', play);
    root.querySelector('.zm-next').addEventListener('click', stepOnce);
    slide.addEventListener('click', stepOnce); // 발표처럼 장표를 눌러도 다음 단계
    window.addEventListener('resize', layoutKey);
    // 완성 상태를 먼저 보여 준 뒤 한 번 재생
    if (!reduce) timers.push(setTimeout(play, 1200));
  });
})();
</script>

바꿀 지점

  1. 색·시간 — .zm의 CSS 변수. 영역 색은 --zm-a~--zm-d, 카메라 이동은 --zm-cam, 자동 재생 간격은 --zm-hold.
  2. 영역 내용 — <section class="zm-zone" data-key="A"> 안의 이름·세부 과제. 세부 과제를 4개로 늘리면 .zm-chips의 repeat(3, 1fr)을 4로.
  3. 영역 위치 — .zm-zone[data-key="A"]의 left/top/width/height(%). 카메라와 키플랜은 위치를 읽어 자동 계산하므로 따로 고칠 곳이 없습니다. 연결선만 <svg>의 path 좌표를 맞춥니다.
  4. 순서 — 스크립트 맨 위 ZM_STEPS 배열. 순서를 바꾸거나 영역을 빼면 그대로 반영됩니다. FIT은 확대 시 화면을 채우는 비율입니다.

03PowerPoint로 옮기기

PowerPoint에는 "카메라"가 없으므로 도면 전체를 그룹으로 묶어 슬라이드마다 크기·위치를 바꾸고, 화면 전환 – 모핑으로 잇는 방식이 가장 가깝습니다. 슬라이드 6장(전체 → A → B → C → D → 전체)을 만듭니다.

모핑으로 만드는 법(권장, Microsoft 365·2019 이상)

  1. 1번 슬라이드에 전체 도면을 그립니다. 영역 상자 4개, 교육청 상자, 연결선, 세부 과제 제목 12개.
  2. 홈 – 정렬 – 선택 창에서 개체 이름을 바꿉니다. 이름 앞에 !!를 붙이면 모핑이 같은 이름끼리 반드시 짝을 짓습니다: !!도면(전체 그룹), !!영역A~!!영역D, !!키플랜, !!키플랜_틀, !!배율.
  3. 슬라이드를 복제(Ctrl+D)해 2번을 만들고, !!도면 그룹을 Shift를 누른 채 모서리로 약 210% 키운 뒤 A 영역이 화면 가운데 오도록 옮깁니다. 슬라이드 밖으로 나간 부분은 잘려 보이므로 그대로 둡니다.
  4. 2번에서 A 외 영역·교육청 상자는 도형 서식 – 채우기·선 투명도 70%, 글자 투명도 70%로 흐리게. 세부 설명 텍스트 상자 3개를 A 안에 추가합니다(1번에는 없음).
  5. 오른쪽 아래에 작은 !!키플랜(전체 도면 축소 사본을 그림으로 붙여넣기)과 청록 테두리 사각형 !!키플랜_틀을 놓습니다. 1번 슬라이드엔 같은 이름으로 두되 투명도 100%.
  6. 3~5번도 같은 방식으로 B·C·D를 가운데로, 6번은 1번을 복제합니다. 2~6번 모두 전환 – 모핑, 효과 옵션 개체, 기간 01.20.

확대/축소 기능으로 만드는 법(대안)

  1. A~D 영역마다 상세 슬라이드를 1장씩 만들고 각각 구역 추가로 구역을 나눕니다(구역 이름: A 놀이 환경 등).
  2. 전체 도면 슬라이드에서 삽입 – 확대/축소 – 구역 확대/축소로 4개 구역을 고른 뒤, 생긴 썸네일을 각 영역 상자 위에 크기를 맞춰 겹칩니다.
  3. 썸네일 선택 후 확대/축소 탭 – 확대/축소로 돌아가기를 켜고, 확대/축소 전환 기간 01.00. 이렇게 하면 상세를 보고 나서 자동으로 전체 도면으로 빠져나옵니다.
  4. 발표 중엔 순서와 관계없이 원하는 영역을 클릭해 들어갈 수 있습니다(질의응답에 유리). 한 장만 부르려면 슬라이드 확대/축소를 씁니다.
안 되는 동작과 근사법. ① 확대 중 다른 영역 흐림(blur): 도형엔 흐리게 효과가 없으므로 투명도 70%로 대신합니다. 꼭 흐림이 필요하면 해당 영역을 그림(PNG)으로 붙여넣고 그림 서식 – 꾸밈 효과 – 흐리게를 쓰되, 모핑 짝은 그림끼리만 맞춥니다. ② 영역→영역 이동 중간에 전체로 빠졌다 들어가는 궤적: 모핑은 직선 이동만 하므로 A와 B 사이에 전체 도면 슬라이드를 끼우고 두 전환을 각 00.60으로 둡니다. 확대/축소 기능의 "돌아가기"를 쓰면 자연스럽게 이 궤적이 됩니다. ③ 배율 숫자(×1.0→×2.1): !!배율 텍스트를 슬라이드마다 직접 고치면 모핑이 글자 단위로 바꿔 줍니다(효과 옵션 문자).
