02동작 설명

왼쪽 단자는 연수 대상, 오른쪽 단자대는 연수 과정입니다. 클릭마다 과정 하나가 꽂히는데, 맞춤 과정(직무 심화·현장 컨설팅·온라인 자율)을 먼저 그려 대상마다 갈래가 다르다는 걸 보여 주고, 마지막에 필수 공통 선 다섯 가닥을 꽂습니다.

허브 단계에서 공통 선만 굵어지고 맞춤 선은 물러나므로 "모두가 거치는 과정 하나 + 직무별 갈래 셋"이라는 편성 원칙이 숫자 없이 읽힙니다. 단자대 높이는 연결 수에 비례해, 선이 몰리는 곳이 곧 허브입니다.

선은 모두 왼쪽에서 오른쪽으로만 흐르고, 양 끝 단자는 상대편 높이 순으로 배정해 교차를 최소화했습니다. 대상이나 과정에 마우스를 올리면 그 선만 남습니다.

이렇게도 씁니다

  • 특수교육 대상학생 ↔ 지원 서비스치료지원·통학비·보조인력 배치 중 무엇이 공통이고 무엇이 개별인지
  • 방과후·돌봄 프로그램 ↔ 학년군전 학년 공통 프로그램과 학년별 프로그램 구분
  • 담당자 ↔ 세부 사업업무분장에서 여러 사업을 겸하는 사람(허브)을 드러내기

      l.p2 = el('circle', {cx:RX, cy:l.y2, r:8, 'class':'sw-plug', 'data-c':l.c, 'data-t':l.t, style:st}, svg);
    });

    // ── 상태: k단계까지 완료된 모습을 즉시 그림 ──
    function stepOf(cid){ for (var i = 0; i < N; i++) if (SW_STEPS[i].course === cid) return i + 1; return Infinity; }
    function cancelAll(){ timers.forEach(clearTimeout); timers = []; anims.forEach(function(a){ a.cancel(); }); anims = []; }
    function caption(k){ stepno.textContent = k + '/' + N; captxt.textContent = k === 0 ? SW_START_TEXT : SW_STEPS[k - 1].text; }
    function apply(k){
      cancelAll();
      kit.classList.add('is-snap');
      links.forEach(function(l){ var on = stepOf(l.c) <= k; [l.path, l.p1, l.p2].forEach(function(n){ n.classList.toggle('is-on', on); }); });
      SW_COURSES.forEach(function(c){ c.led.classList.toggle('is-lit', stepOf(c.id) <= k); });
      var hubOn = SW_STEPS.some(function(s, i){ return s.hub && i < k; });
      kit.classList.toggle('is-hub', hubOn);
      caption(k); state = k;
      void kit.offsetWidth; kit.classList.remove('is-snap');
    }
    // 한 단계 재생에 걸리는 시간
    function durOf(k){
      var s = SW_STEPS[k - 1];
      if (s.hub) return ms(kit, '--sw-hub');
      var n = links.filter(function(l){ return l.c === s.course; }).length;
      return (n - 1) * ms(kit, '--sw-stagger') + ms(kit, '--sw-draw') + 220;
    }
    // ── k-1 → k 로 움직이며 넘어가기 ──
    function play(k){
      if (reduce) { apply(k); return; }
      apply(k - 1);
      var s = SW_STEPS[k - 1], draw = ms(kit, '--sw-draw'), stag = ms(kit, '--sw-stagger');
      caption(k); state = k;
      if (s.hub) { kit.classList.add('is-hub'); return; }
      var mine = links.filter(function(l){ return l.c === s.course; }).sort(function(a, b){ return a.y1 - b.y1; });
      mine.forEach(function(l, i){
        var d = i * stag;
        [l.path, l.p1, l.p2].forEach(function(n){ n.classList.add('is-on'); });
        // 선: 왼쪽 플러그 → 선이 그어짐 → 오른쪽 플러그 꽂힘
        anims.push(l.p1.animate([{transform:'scale(0)'}, {transform:'scale(1)'}], {duration:200, delay:d, fill:'backwards', easing:'cubic-bezier(.3,1.6,.6,1)'}));
        anims.push(l.path.animate([{strokeDashoffset:l.len}, {strokeDashoffset:0}], {duration:draw, delay:d, fill:'backwards', easing:'cubic-bezier(.45,0,.2,1)'}));
        anims.push(l.p2.animate([{transform:'scale(0)'}, {transform:'scale(1)'}], {duration:220, delay:d + draw, fill:'backwards', easing:'cubic-bezier(.3,1.6,.6,1)'}));
      });
      // 과정 표시등은 마지막 선이 꽂힐 때 켜짐
      timers.push(setTimeout(function(){ course[s.course].led.classList.add('is-lit'); }, (mine.length - 1) * stag + draw));
    }
    function replay(){
      apply(0);
      if (reduce) { apply(N); return; }
      var t = 500, pause = ms(kit, '--sw-pause'), k;
      for (k = 1; k <= N; k++) (function(k, at){ timers.push(setTimeout(function(){ var keep = timers; timers = []; play(k); timers = keep.concat(timers); }, at)); })(k, t), t += durOf(k) + pause;
    }
    function step(){ if (state >= N) apply(0); else play(state + 1); }

    kit.querySelector('[data-sw="replay"]').addEventListener('click', replay);
    kit.querySelector('[data-sw="step"]').addEventListener('click', step);
    kit.querySelector('.sw-slide').addEventListener('keydown', function(e){
      if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'Enter') { e.preventDefault(); step(); }
      if (e.key === 'ArrowLeft') { e.preventDefault(); apply(Math.max(0, state - 1)); }
    });
    // 대상·과정에 올리면 관련 선만 남김
    svg.querySelectorAll('.sw-node').forEach(function(g){
      var key = g.getAttribute('data-t') ? 'data-t' : 'data-c', val = g.getAttribute(key);
      function on(){ kit.classList.add('is-focus'); svg.querySelectorAll('.sw-wire,.sw-plug').forEach(function(n){ n.classList.toggle('is-hot', n.getAttribute(key) === val); }); }
      function off(){ kit.classList.remove('is-focus'); }
      g.addEventListener('pointerenter', on); g.addEventListener('pointerleave', off);
      g.addEventListener('focus', on); g.addEventListener('blur', off);
    });

    // 열면 완성 상태를 먼저 보여 주고, 잠시 뒤 처음부터 한 번 재생 후 최종 상태로 멈춤
    apply(N);
    if (!reduce) setTimeout(replay, 1400);
  }
  document.querySelectorAll('.sw-kit:not([data-sw-ready])').forEach(init);
})();
</script>

바꿀 지점

  • SW_TARGETS 왼쪽 대상. group이 같은 것끼리 괄호로 묶입니다.
  • SW_COURSES 오른쪽 과정. hub:true인 과정이 마지막에 굵어집니다. 단자대 높이는 연결 수로 자동 계산.
  • SW_LINKS [대상, 과정] 쌍. 추가만 하면 단자 배정과 곡선은 자동입니다.
  • SW_STEPS 클릭 순서. course 단계는 그 과정 선을 그리고, hub 단계는 강조만 합니다.
  • --sw-* 변수: 색, 선 굵기(--sw-w, --sw-w-hub), 타이밍(--sw-draw, --sw-stagger, --sw-pause).
  • 장표 제목·계획안 위치는 .sw-head 안의 글자를 바꿉니다.
  • 한 페이지에 여러 개를 넣어도 됩니다. 각 .sw-kit이 따로 초기화됩니다(데이터 배열은 공유).

04PPT로 옮기기

선은 삽입 › 도형 › 선 › 곡선 연결선으로 단자(작은 원)의 연결점에 붙여 그립니다. 모든 선이 왼쪽→오른쪽으로만 진행하도록 설계했기 때문에 나타내기 › 닦아내기(왼쪽에서)가 선을 따라 그려 나가는 것과 거의 같게 보입니다. 클릭 1–4는 한 장 안의 애니메이션, 클릭 5(허브 강조)는 다음 장으로 넘기며 모핑을 씁니다.

모핑 장표 만드는 법

1. 장 복제
클릭 1–4가 끝난 장을 복제하고, 복제본의 애니메이션은 모두 지웁니다.
2. 복제본에서 바꿀 것
공통 선 5개 두께 4.5pt → 10pt, 맞춤 선 7개 투명도 70%, 필수 공통 단자대 너비 1.7배·주황 채우기, "5개 대상 공유" 글자 추가.
3. 개체 이름 맞추기
선택 창(Alt+F10)에서 두 장의 같은 개체에 같은 이름을 붙이고 앞에 !!를 붙입니다. 예: !!선-공통-유치원, !!선-심화-초등, !!단자-공통. 그래야 굵기·투명도 변화가 엉뚱한 개체끼리 섞이지 않습니다.
4. 첫 화면 완성본
발표 시작 전에 완성 모습을 보여 주려면 허브 장을 한 장 더 복사해 맨 앞 표지 자리에 둡니다.

그대로 안 되는 동작과 근사법

마우스 올리면 관련 선만 남기기
PPT에는 마우스 오버 효과가 없습니다. 대상 이름에 트리거(애니메이션 창 › 타이밍 › 트리거 › 다음 개체 클릭 시)를 걸어 그 대상의 선 묶음에 강조 › 펄스를 주거나, 대상별 강조 장을 따로 만들어 하이퍼링크 + 모핑으로 오갑니다.
플러그가 살짝 튀며 꽂힘
확대/축소 뒤에 강조 › 크게/작게(110%, 자동 되감기, 0.2초)를 이전 효과 다음에 붙입니다.
선 굵어짐(모핑 없는 버전)
굵은 공통 선 5개를 미리 겹쳐 두고 나타내기 › 밝기 변화, 맞춤 선에는 강조 › 투명도 70%를 이전 효과와 함께 겁니다.
