Ⅴ. 추진 일정 · ○○시 중앙도서관 리모델링 추진 계획(안)

6월 실시설계 확정이 넘어져야 재개관까지 이어집니다

도미노 1개 = 1개월넘어지면 완료밀리면 뒤가 모두 밀리는 절차
가상 예시

동작 설명 무엇을 움직임으로 보여 주나

일정표의 막대 대신 한 달에 도미노 1개를 세웠습니다. 앞 절차가 넘어져야 다음 절차가 넘어지므로, 화면을 보는 사람은 설명 없이도 선후 관계를 읽습니다. 1월 기본계획 뒤에서 시설·운영 두 줄로 갈라졌다가 이듬해 4월의 큰 도미노에서 다시 합쳐집니다.

넘어진 도미노는 곧 완료된 일입니다. 다 넘어진 최종 화면은 그대로 진척표가 되고, 주황색 실시설계 도미노와 그 뒤 10개월 괄호가 "여기가 밀리면 뒤가 모두 밀린다"는 결론을 남깁니다. 한 단계씩을 누르면 절차마다 앞 도미노에 기대어 멈춥니다.

가져갈 장치 ① 개수로 세는 기간

도미노 1개 = 1개월. 기간이 막대 길이가 아니라 세울 수 있는 개수가 되어 "몇 달 남았나"가 손가락으로 세어집니다.

응용: 공사 공정표, 예산 집행 월별 계획, 채용 절차 주차별 일정

가져갈 장치 ② 넘어짐 = 완료

같은 장표를 진행 상황 보고에 다시 씁니다. 오늘까지 끝난 달만 넘어뜨려 두면 서 있는 도미노가 남은 일입니다.

응용: 분기 점검 보고, 사업 중간 보고, 정책 과제 이행 현황

가져갈 장치 ③ 갈라짐과 합류

한 도미노가 두 줄을 동시에 밀고, 두 줄이 큰 도미노 하나를 함께 넘어뜨립니다. 병행 과제와 공동 목표가 한 번에 보입니다.

응용: 시스템 개통(개발·교육 병행), 행사 준비(시설·홍보), 조직 개편 일정

(function(){

  var root = document.getElementById('dmn');

  var cv = root.querySelector('.dmn-canvas');

  function cvar(n){ return css.getPropertyValue(n).trim(); }
  var COL = {a:cvar('--dmn-a'), b:cvar('--dmn-b'), hot:cvar('--dmn-hot'), goal:cvar('--dmn-goal'), ivory:'#FBFAF6', pip:'#1E2836'};

  /* ── 1. 도미노 배치 (월 = x, 줄 = y, 높이 = z) ── */
  var YA = 0, YB = 2, N = DMN.months;
  var ms = {};                       // 'a3' → 마일스톤 정보

  var doms = [];
  function add(x, y, kind, info){
    var big = kind === 'goal', mile = !!info && !big;
    var d = {x:x, y:y, info:info || null, kind:kind,
      h: big ? 2.3 : mile ? 1.5 : 1.25, w: big ? 2.2 : mile ? .7 : .6, t: big ? .36 : mile ? .24 : .2,

    d.pips = big ? [6, 6] : [(Math.round(x * 3) + (kind === 'b' ? 2 : 0)) % 7, (Math.round(x * 5) + 1) % 7];
    doms.push(d); return d;
  }
  function link(a, b){ a.next.push(b); b.prev.push(a); }
  var A = [], B = [];
  for (var m = 1; m < N; m++) A[m] = add(m, YA, 'a', ms['a' + m]);

  for (m = DMN.bFrom; m < N; m++) B[m] = add(m, YB, 'b', ms['b' + m]);

  for (m = 1; m < N - 1; m++) link(A[m], A[m + 1]);
  link(A[1], b1); link(b1, b2); link(b2, B[DMN.bFrom]);
  for (m = DMN.bFrom; m < N - 1; m++) link(B[m], B[m + 1]);

  doms.forEach(function(d){

    d.phi = n ? Math.atan2(n.y - d.y, n.x - d.x) : 0;

    d.ca = Math.cos(d.phi); d.sa = Math.sin(d.phi);
  });

  /* ── 2. 접촉 각도 미리 계산: 뒤 도미노가 φ만큼 기울었을 때 앞 도미노가 멈추는 각 ── */
  function contactAngle(i, j, phiDeg){
    var dA = (j.x - i.x) * i.ca + (j.y - i.y) * i.sa;
    var pjA = dA + j.t / 2, ph = phiDeg * Math.PI / 180;
    var bA = pjA - j.t * Math.cos(ph), bZ = j.t * Math.sin(ph);          // 뒤 도미노의 뒷면 아래 모서리
    var uA = Math.sin(ph), uZ = Math.cos(ph), nA = -Math.cos(ph), nZ = Math.sin(ph);
    var tA = bA + j.h * uA, tZ = bZ + j.h * uZ;                           // 뒷면 위 모서리
    for (var th = 0; th <= 90; th += .25){
      var r = th * Math.PI / 180, s = Math.sin(r), c = Math.cos(r);
      var cA = i.t / 2 + i.h * s, cZ = i.h * c;                          // 앞 도미노의 윗 모서리
      var dist = (cA - bA) * nA + (cZ - bZ) * nZ, par = (cA - bA) * uA + (cZ - bZ) * uZ;
      if (dist <= 0 && par >= -.05 && par <= j.h + .02) return th;
      var fd = (tA - i.t / 2) * c - tZ * s, fp = (tA - i.t / 2) * s + tZ * c; // 뒤 도미노 윗 모서리 vs 앞면
      if (fd <= 0 && fp >= 0 && fp <= i.h) return th;
    }
    return 90;
  }
  doms.forEach(function(d){
    d.tab = d.next.map(function(j){ var a = []; for (var p = 0; p <= 90; p++) a.push(contactAngle(d, j, p)); return a; });
  });
  function lookup(tab, p){ var k = Math.max(0, Math.min(90, p)), f = Math.floor(k), c = Math.min(90, f + 1); return tab[f] + (tab[c] - tab[f]) * (k - f); }

  /* ── 3. 넘어지기 시작하는 시각: 앞 도미노가 닿는 순간 ── */
  A[1].s = 0;
  var order = doms.slice().sort(function(p, q){ return p.x - q.x; });
  for (var pass = 0; pass < 3; pass++) order.forEach(function(d){
    d.next.forEach(function(j, k){
      var tc = d.dur * Math.sqrt(d.tab[k][0] / 90);
      if (d.s + tc < j.s) j.s = d.s + tc;
    });
  });
  var END = G.s + G.dur + .35;
  var BOUND = DMN.steps.map(function(st){
    var t = 0;
    st.items.forEach(function(it){
      var d = doms.filter(function(q){ return q.info === it; })[0];
      var tc = d.next.length ? d.dur * Math.sqrt(d.tab[0][0] / 90) : d.dur + .35;
      t = Math.max(t, d.s + tc);
    });
    return t;
  });
  BOUND[BOUND.length - 1] = END;

  function freeAngle(d, t){
    var u = (t - d.s) / d.dur;
    if (u <= 0) return 0;
    if (u < 1) return 90 * u * u;                                         // 중력처럼 점점 빠르게
    if (d.kind === 'goal'){ var v = (t - d.s - d.dur) / .3; return v < 1 ? 90 - 5 * Math.sin(Math.PI * v) : 90; } // 바닥에 닿아 살짝 튐
    return 90;
  }
  var byStart = doms.slice().sort(function(p, q){ return q.s - p.s; });
  function solve(t){
    byStart.forEach(function(d){
      var th = freeAngle(d, t);
      d.next.forEach(function(j, k){ th = Math.min(th, lookup(d.tab[k], j.th)); });
      d.th = th;
    });
  }

  /* ── 4. 카메라(왼쪽 앞 위에서 내려다봄) ── */
  var cam = (function(){
    var yaw = -2 * Math.PI / 180, pit = 52 * Math.PI / 180, D = 48;
    var T = [9.6, .8, .4];
    var v = [Math.sin(yaw) * Math.cos(pit), -Math.cos(yaw) * Math.cos(pit), Math.sin(pit)];
    var C = [T[0] + D * v[0], T[1] + D * v[1], T[2] + D * v[2]];
    var f = [-v[0], -v[1], -v[2]];
    var r = [f[1], -f[0], 0], rl = Math.hypot(r[0], r[1]); r = [r[0] / rl, r[1] / rl, 0];
    var u = [r[1] * f[2] - r[2] * f[1], r[2] * f[0] - r[0] * f[2], r[0] * f[1] - r[1] * f[0]];
    return {C:C, f:f, r:r, u:u, k:1, ox:0, oy:0};
  })();
  function proj(P){
    var dx = P[0] - cam.C[0], dy = P[1] - cam.C[1], dz = P[2] - cam.C[2];
    var zc = dx * cam.f[0] + dy * cam.f[1] + dz * cam.f[2];
    var xc = dx * cam.r[0] + dy * cam.r[1] + dz * cam.r[2];
    var yc = dx * cam.u[0] + dy * cam.u[1] + dz * cam.u[2];
    return [cam.ox + cam.k * xc / zc, cam.oy - cam.k * yc / zc, zc];
  }
  var FX0 = -.4, FX1 = N + 2.8, FY0 = -1.35, FY1 = 2.85, FZ = .26;      // FY0 = 앞 가장자리      // 탁자 범위
  (function fit(){                                                       // 장면을 장표 안 정해진 칸(cqw 단위)에 맞춘다
