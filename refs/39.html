TAKEAWAY다른 발표에 가져갈 장치

1-4-3-3

포메이션 숫자로 조직 요약

조직도 대신 "몇 명이 어느 줄에 서는가"를 숫자 한 줄로 말합니다. 앞줄로 갈수록 예방, 뒷줄로 갈수록 지휘라는 위치 의미가 생깁니다.

응용 신규 사업단 구성, TF 역할 분담, 축제 운영 인력 배치 표지

인쇄층 / 마커층

확정은 인쇄, 계획은 손글씨

바뀌지 않는 틀(피치 라인)은 인쇄처럼 가늘고 반듯하게, 이번 계획의 움직임·이름표는 보드마커 손글씨로 겹칩니다. 두 층이 갈라져 "무엇이 새로 정한 것인가"가 한눈에 보입니다.

응용 기존 제도 위에 개선안 덧그리기, 평면도 위 동선 계획, 연간 일정표 위 변경 표시

포메이션 착지

조직이 제자리에 서는 순간을 보여 주기

자석이 손에 들려 떠 있다가(그림자가 멀고 흐림) 제자리에 "탁" 내려앉습니다. 감독석 → 골키퍼 → 수비 → 미드 → 공격 순서가 곧 지휘 → 최후 방어 → 현장 → 지원 → 예방의 설명 순서입니다.

응용 조직 개편 발표, TF 구성, 행사 인력 배치. PPT는 자석마다 나타내기 › 확대/축소 0.35초 + 그림자 개체 밝기 변화, 줄마다 이전 효과 다음

마주 보는 화살표

대치선 하나로 과제 선언

우리 쪽 파란 점선 화살표와 상대 쪽 빨간 화살표가 하프라인에서 만납니다. "어디서 막을 것인가"가 문장 없이 위치로 전달됩니다.

응용 경쟁사 대응 전략, 민원 급증 대비, 감염병 유입 차단 계획

FLOW흐름 설계 메모: 계획안 ↔ 경기 준비

계획안 구성보드 위 표현안쪽 장표 아이디어
추진 배경지난 시즌 실점 장면작년 침수 지점을 피치 위 빨간 X로 다시 찍고, 들어온 경로를 화살표로 복기
목표이번 시즌 목표보드 머리 칸에 "인명 피해 0 · 침수 신고 처리 2시간 이내" 두 줄만
협업 조직포메이션 1-4-3-3자석 11개 = 부서 11개. 줄마다 부서명을 자석 위에 작게, 표지와 같은 배치 유지
단계별 대응세트피스예비 → 주의보 → 경보 세 장면에서 자석 위치만 바뀌는 연속 장표(모핑)
일정·훈련시즌 일정표6~9월을 경기 일정처럼, 5월 합동 훈련은 "친선 경기"로 표시
평가·환류경기 후 리뷰같은 보드에 지운 자국을 남기고 고칠 화살표만 새 색으로

    // 피치: 잔디 깎은 결처럼 아주 옅은 세로 띠
    const P = pitch();
    const bands = 14;
    for(let i=0;i<bands;i++){ ctx.fillStyle = i%2 ? '#eef8f1' : '#f6fbf7'; ctx.fillRect(P.x+P.w*i/bands, P.y, P.w/bands+1, P.h); }

    // 지운 마커 자국(아주 옅게) — 쓰던 보드라는 흔적
    ctx.strokeStyle='rgba(60,80,110,.05)'; ctx.lineWidth=W*0.012; ctx.lineCap='round';
    ctx.beginPath(); ctx.moveTo(W*.6,H*.83); ctx.quadraticCurveTo(W*.7,H*.76,W*.8,H*.85); ctx.stroke();

    lines(P);
    techArea(P);
    card();

    // 법랑 광택: 왼쪽 위에서 비스듬한 반사 띠
    const s = ctx.createLinearGradient(x,y,x+w*.55,y+h);
    s.addColorStop(0,'rgba(255,255,255,0)'); s.addColorStop(.42,'rgba(255,255,255,.0)'); s.addColorStop(.5,'rgba(255,255,255,.55)'); s.addColorStop(.6,'rgba(255,255,255,0)');
    ctx.fillStyle=s; ctx.fillRect(x,y,w,h);
    ctx.restore();

    // 테두리 안쪽 턱 그림자
    ctx.save(); rr(x,y,w,h,W*0.004); ctx.clip();
    ctx.shadowColor='rgba(20,30,40,.35)'; ctx.shadowBlur=W*0.006; ctx.shadowOffsetX=W*0.0015; ctx.shadowOffsetY=W*0.002;
    ctx.lineWidth=W*0.006; ctx.strokeStyle='rgba(120,130,140,.6)'; rr(x-W*0.003,y-W*0.003,w+W*0.006,h+W*0.006,W*0.006); ctx.stroke();
    ctx.restore();

    corners();
    tray();
  }
  // 보드 아래 알루미늄 마커 트레이: 빨강·파랑 마커와 펠트 지우개
  function tray(){
    const x0=W*0.64, x1=W*0.94, y0=H*0.935, y1=H*0.99;
    ctx.save();
    ctx.shadowColor='rgba(16,24,32,.3)'; ctx.shadowBlur=W*0.006; ctx.shadowOffsetY=W*0.002;
    let g=ctx.createLinearGradient(0,y0,0,y1); g.addColorStop(0,'#F4F6F8'); g.addColorStop(.35,'#C6CDD3'); g.addColorStop(1,'#9AA3AB');
    ctx.fillStyle=g; rr(x0,y0,x1-x0,y1-y0,W*0.003); ctx.fill(); ctx.restore();
    ctx.fillStyle='#7D868E'; ctx.fillRect(x0+W*0.004,y0+H*0.008,x1-x0-W*0.008,H*0.012);         // 홈
    // 마커(누운 원통): 몸통 흰색 + 색 캡
    const pen=(px,col)=>{ const py=y0+H*0.004, L=W*0.068, d=H*0.022;
      let gg=ctx.createLinearGradient(0,py,0,py+d); gg.addColorStop(0,'#FFFFFF'); gg.addColorStop(.5,'#E9ECEE'); gg.addColorStop(1,'#B8BFC5');
      ctx.fillStyle=gg; rr(px,py,L,d,d/2); ctx.fill();
      gg=ctx.createLinearGradient(0,py,0,py+d); gg.addColorStop(0,shade(col,.25)); gg.addColorStop(1,shade(col,-.3));
      ctx.fillStyle=gg; rr(px+L*.72,py-d*.06,L*.3,d*1.12,d*.5); ctx.fill();
      ctx.fillStyle=col; ctx.fillRect(px+L*.2,py+d*.32,L*.34,d*.36); };
    pen(x0+W*0.012,'#C9332C'); pen(x0+W*0.09,'#24439A');
    // 지우개: 손잡이(회색) + 펠트(짙은 회색)
    const ex=x0+W*0.19, ey=y0-H*0.012, ew=W*0.075, eh=H*0.036;
    ctx.fillStyle='#3E454C'; rr(ex,ey+eh*.62,ew,eh*.42,W*0.002); ctx.fill();
    g=ctx.createLinearGradient(0,ey,0,ey+eh*.7); g.addColorStop(0,'#D9DEE2'); g.addColorStop(1,'#A9B1B8');
    ctx.fillStyle=g; rr(ex+ew*.04,ey,ew*.92,eh*.7,W*0.004); ctx.fill();
  }

  // 감독석(테크니컬 에어리어): 하프라인 아래 터치라인 밖 점선 칸
  function techArea(P){
    const cx = W*0.6, y = P.y+P.h+H*0.03, w = W*0.13, h = H*0.1;
    ctx.save(); ctx.strokeStyle='#2e9e5b'; ctx.lineWidth=Math.max(1,W*0.0018); ctx.setLineDash([W*0.006,W*0.005]);
    ctx.strokeRect(cx-w*0.62, y, w, h); ctx.restore();
  }
  // 제목 카드: 보드에 자석 두 개로 붙인 흰 카드(두께·그림자). 피치 선이 제목 글자와 겹치지 않게 한다
  function card(){
    const x=W*0.53, y=H*0.19, w=W*0.425, h=H*0.6, r=W*0.006;
    ctx.save(); ctx.shadowColor='rgba(16,32,44,.22)'; ctx.shadowBlur=W*0.012; ctx.shadowOffsetX=W*0.002; ctx.shadowOffsetY=W*0.004;
    rr(x,y,w,h,r); ctx.fillStyle='#ffffff'; ctx.fill(); ctx.restore();
    ctx.save(); rr(x,y,w,h,r); ctx.strokeStyle='rgba(16,32,44,.10)'; ctx.lineWidth=1; ctx.stroke(); ctx.restore();
    [[x+W*0.02,y+H*0.03],[x+w-W*0.02,y+H*0.03]].forEach(([mx,my]) => pin(mx,my));
  }
  function pin(x,y){ // 카드를 누르는 작은 은색 자석
    const r=W*0.009;
    ctx.save(); ctx.fillStyle='rgba(20,30,45,.25)'; ctx.filter=`blur(${r*.35}px)`; ctx.beginPath(); ctx.arc(x+r*.25,y+r*.35,r,0,Math.PI*2); ctx.fill(); ctx.restore();
    ctx.fillStyle='#8E979F'; ctx.beginPath(); ctx.arc(x,y+r*.18,r,0,Math.PI*2); ctx.fill();          // 두께
    const g=ctx.createLinearGradient(x-r,y-r,x+r,y+r); g.addColorStop(0,'#E6EAED'); g.addColorStop(1,'#B3BCC4');
    ctx.fillStyle=g; ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2); ctx.fill();
    ctx.strokeStyle='rgba(255,255,255,.8)'; ctx.lineWidth=r*.12; ctx.beginPath(); ctx.arc(x,y,r*.8,Math.PI*.9,Math.PI*1.6); ctx.stroke();
    ctx.strokeStyle='rgba(70,80,90,.35)'; ctx.lineWidth=r*.08; ctx.beginPath(); ctx.arc(x,y,r*.8,0,Math.PI*2); ctx.stroke();
  }

  function corners(){
    // 보강 모서리 캡(검은 플라스틱 L자)
    const L = W*0.05, t = W*0.016;
    const cs = [[0,0,1,1],[W,0,-1,1],[0,H,1,-1],[W,H,-1,-1]];
    cs.forEach(([cx,cy,sx,sy]) => {
      ctx.save(); ctx.translate(cx,cy); ctx.scale(sx,sy);
      ctx.shadowColor='rgba(0,0,0,.25)'; ctx.shadowBlur=W*0.004; ctx.shadowOffsetY=W*0.0012;
      ctx.beginPath(); ctx.moveTo(0,0); ctx.lineTo(L,0); ctx.quadraticCurveTo(L,t,L-t*.4,t); ctx.lineTo(t*1.1,t); ctx.quadraticCurveTo(t,t,t,t*1.1); ctx.lineTo(t,L-t*.4); ctx.quadraticCurveTo(t,L,0,L); ctx.closePath();
      const g = ctx.createLinearGradient(0,0,t,t); g.addColorStop(0,'#59616a'); g.addColorStop(1,'#2a3037');
      ctx.fillStyle=g; ctx.fill(); ctx.shadowColor='transparent';
      ctx.strokeStyle='rgba(255,255,255,.18)'; ctx.lineWidth=1; ctx.stroke();
      ctx.restore();
    });
  }

  function pitch(){ return {x:W*0.045, y:H*0.075, w:W*0.91, h:H*0.725}; }   // 아래 띠 = 벤치·라벨 자리

  function lines(P){
    const sx = P.w/105, sy = P.h/68;
    ctx.strokeStyle='#2e9e5b'; ctx.lineWidth=Math.max(1,W*0.0024); ctx.lineCap='butt';
    ctx.strokeRect(P.x,P.y,P.w,P.h);
    const cx = P.x+P.w/2, cy = P.y+P.h/2;
    ctx.beginPath(); ctx.moveTo(cx,P.y); ctx.lineTo(cx,P.y+P.h); ctx.stroke();
    ctx.beginPath(); ctx.ellipse(cx,cy,9.15*sx,9.15*sy,0,0,Math.PI*2); ctx.stroke();
    dot(cx,cy);
    [1,-1].forEach(side => {
      const gx = side>0 ? P.x : P.x+P.w;
      const pa = 16.5*sx*side, ga = 5.5*sx*side;
      box(gx, cy, pa, 40.3*sy); box(gx, cy, ga, 18.32*sy);
      const spot = gx + 11*sx*side; dot(spot, cy);
      // 페널티 아크(박스 밖 부분만)
      ctx.save(); ctx.beginPath();
      if(side>0) ctx.rect(gx+pa, P.y, P.w, P.h); else ctx.rect(P.x, P.y, (gx+pa)-P.x, P.h);
      ctx.clip(); ctx.beginPath(); ctx.ellipse(spot,cy,9.15*sx,9.15*sy,0,0,Math.PI*2); ctx.stroke(); ctx.restore();
      // 골대
      ctx.save(); ctx.lineWidth = W*0.003; ctx.strokeStyle='#2e9e5b';
      ctx.strokeRect(side>0 ? gx-2.2*sx : gx, cy-3.66*sy, 2.2*sx, 7.32*sy); ctx.restore();
    });
    // 코너 아크
    [[P.x,P.y,0],[P.x+P.w,P.y,Math.PI/2],[P.x+P.w,P.y+P.h,Math.PI],[P.x,P.y+P.h,Math.PI*1.5]].forEach(([x,y,a])=>{ ctx.beginPath(); ctx.ellipse(x,y,1.2*sx,1.2*sy,0,a,a+Math.PI/2); ctx.stroke(); });
    function box(gx,cy,d,hh){ ctx.strokeRect(Math.min(gx,gx+d), cy-hh/2, Math.abs(d), hh); }
    function dot(x,y){ ctx.beginPath(); ctx.arc(x,y,W*0.0025,0,Math.PI*2); ctx.fillStyle='#2e9e5b'; ctx.fill(); }
  }

  function shade(hex, k){ // k>0 밝게, k<0 어둡게
    const n = parseInt(hex.slice(1),16); let r=n>>16, g=(n>>8)&255, b=n&255;
    const t = k>0?255:0, a=Math.abs(k);
    return `rgb(${Math.round(r+(t-r)*a)},${Math.round(g+(t-g)*a)},${Math.round(b+(t-b)*a)})`;
  }

  // lift: 0 = 보드에 붙음, 1 = 손에 들려 떠 있음(그림자가 멀어지고 흐려진다), sq: 착지 순간 눌림
  function puck(fx,fy,base,lift=0,sq=0,square=false,num=''){
    const gx=fx*W, gy=fy*H, r=W*0.0215*(1+lift*0.18), th=r*0.2;
    const x=gx-lift*W*0.012, y=gy-lift*H*0.05;
    // 바닥 그림자(광원: 왼쪽 위) — 바닥 위치에 남는다
    ctx.save(); ctx.fillStyle=`rgba(20,30,45,${.30-lift*.18})`; ctx.filter=`blur(${r*(0.28+lift*0.9)}px)`;
    ctx.beginPath(); ctx.ellipse(gx+r*0.22,gy+r*0.34,r*(1.02+lift*.25),r*(0.96+lift*.2),0,0,Math.PI*2); ctx.fill(); ctx.restore();
    ctx.save(); ctx.translate(x,y); ctx.scale(1+sq*0.08,1-sq*0.08); ctx.translate(-x,-y);
    if(square){ squarePuck(x,y,r,th,base); ctx.restore(); return; }
    // 납작한 무광 원판 자석: 옆 두께 띠 → 모따기 테 → 평평한 윗면 → 찍힌 등번호
    const t2 = r*0.3;
    ctx.fillStyle=shade(base,-.42); ctx.beginPath(); ctx.arc(x,y+t2,r,0,Math.PI*2); ctx.fill();
    ctx.fillStyle=shade(base,-.22); ctx.beginPath(); ctx.arc(x,y+t2*.45,r,0,Math.PI*2); ctx.fill();
    const tg = ctx.createLinearGradient(x-r,y-r,x+r,y+r);
    tg.addColorStop(0,shade(base,.1)); tg.addColorStop(1,shade(base,-.08));
    ctx.fillStyle=tg; ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2); ctx.fill();
    ctx.lineWidth=r*0.07; ctx.strokeStyle=shade(base,.32); ctx.beginPath(); ctx.arc(x,y,r*.9,Math.PI*.95,Math.PI*1.7); ctx.stroke();
    ctx.strokeStyle='rgba(0,0,0,.12)'; ctx.beginPath(); ctx.arc(x,y,r*.9,Math.PI*-.05,Math.PI*.7); ctx.stroke();
    if(num){ ctx.fillStyle=base==='#E2A93B'?'#3A2A0A':'#FFFFFF'; ctx.font=`700 ${r*0.92}px "IBM Plex Sans KR","Malgun Gothic",sans-serif`;
      ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText(num,x,y+r*.05); }
    ctx.restore();
  }
  // 감독 자석: 모서리 둥근 네모(다른 모양 = 선수가 아니라 지휘)
  function squarePuck(x,y,r,th,base){
    const s=r*1.7, k=r*.35;
    ctx.fillStyle='#05090D'; rr(x-s/2,y-s/2+r*.3,s,s,k); ctx.fill();
    const g=ctx.createLinearGradient(x-s/2,y-s/2,x+s/2,y+s/2); g.addColorStop(0,shade(base,.18)); g.addColorStop(1,shade(base,-.1));
    ctx.fillStyle=g; rr(x-s/2,y-s/2,s,s,k); ctx.fill();
    ctx.fillStyle='#ffffff'; ctx.font=`700 ${r*0.95}px "IBM Plex Sans KR","Malgun Gothic",sans-serif`; ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText('C',x,y+r*.04);
    ctx.strokeStyle='rgba(255,255,255,.22)'; ctx.lineWidth=r*.07; rr(x-s/2+r*.12,y-s/2+r*.12,s-r*.24,s-r*.24,k*.7); ctx.stroke();
  }

  // 보드마커 획: 2차 곡선을 점으로 나눠 진행률 t까지 긋기
  function marker(s, t){
    const [a,b,c] = s.p.map(([px,py]) => [px*W,py*H]);
    const N = 60, pts=[];
    for(let i=0;i<=N;i++){ const u=i/N, v=1-u; pts.push([v*v*a[0]+2*v*u*b[0]+u*u*c[0], v*v*a[1]+2*v*u*b[1]+u*u*c[1]]); }
    const n = Math.max(1, Math.floor(N*t));
    const lw = W*0.0042;
    ctx.save(); ctx.lineCap='round'; ctx.lineJoin='round'; ctx.strokeStyle=s.c;
    if(s.dash) ctx.setLineDash([lw*2.6, lw*2.2]);
    // 잉크 두 겹: 바깥 반투명 + 진한 심
    [[lw*1.25,.35],[lw,.92]].forEach(([w,al]) => {
      ctx.globalAlpha=al; ctx.lineWidth=w; ctx.beginPath(); ctx.moveTo(...pts[0]);
      for(let i=1;i<=n;i++) ctx.lineTo(...pts[i]); ctx.stroke();
    });
    ctx.setLineDash([]);
    if(t>=1){ // 화살촉
      const [ex,ey]=pts[N], [px,py]=pts[N-6]; const ang=Math.atan2(ey-py,ex-px), L=W*0.014;
      ctx.globalAlpha=.95; ctx.lineWidth=lw*1.1; ctx.beginPath();
      ctx.moveTo(ex-L*Math.cos(ang-0.5), ey-L*Math.sin(ang-0.5)); ctx.lineTo(ex,ey); ctx.lineTo(ex-L*Math.cos(ang+0.5), ey-L*Math.sin(ang+0.5)); ctx.stroke();
    }
    ctx.restore();
  }

  // 상대 표시: 보드마커로 그린 비구름 + 빗방울 3개 (t: 0~1 그려진 정도)
  function cloud(t){
    if(t<=0) return;
    const cx=W*0.915, cy=H*0.115, u=W*0.014;
    ctx.save(); ctx.strokeStyle='#d9302a'; ctx.lineCap='round'; ctx.lineJoin='round';
    const lw=W*0.0042;
    [[lw*1.25,.35],[lw,.92]].forEach(([w,al]) => {
      ctx.globalAlpha=al; ctx.lineWidth=w; ctx.beginPath();
      const a=Math.min(1,t*1.6);
      ctx.arc(cx-u*1.3, cy, u*0.9, Math.PI*0.5, Math.PI*(0.5+1.05*a));
      if(a>0.4) ctx.arc(cx, cy-u*0.55, u*1.25, Math.PI*1.05, Math.PI*(1.05+1.0*Math.min(1,(a-.4)/.6)));
      ctx.stroke();
      if(a>=1){ ctx.beginPath(); ctx.arc(cx+u*1.35, cy+u*0.05, u*0.85, Math.PI*1.4, Math.PI*2.5); ctx.moveTo(cx+u*1.35,cy+u*0.9); ctx.lineTo(cx-u*1.3,cy+u*0.9); ctx.stroke(); }
    });
    const drops=Math.floor(Math.max(0,(t-.6)/.4)*3.999);
    for(let k=0;k<drops;k++){ const x=cx-u*0.9+k*u*0.9, y=cy+u*1.6; ctx.globalAlpha=.92; ctx.lineWidth=lw;
      ctx.beginPath(); ctx.moveTo(x,y); ctx.lineTo(x-u*0.25,y+u*0.75); ctx.stroke(); }
    ctx.restore();
  }

  const ease = p => 1-Math.pow(1-p,3);
  const labels = [...cv.querySelectorAll('[data-g]')];
  let shown = reduce ? T.end : 0;   // 지금 그려 둔 시각(ms). 크기가 바뀌면 이 시각으로 다시 그린다

  // ms: 재생 시작부터 지난 시간. T.end 이상이면 완성 화면
  function draw(ms){
    shown = ms;
    ctx.setTransform(dpr,0,0,dpr,0,0);
    ctx.clearRect(0,0,W,H);
    ctx.drawImage(noise,0,0,W,H);
    const place = (pos, g, k, color, square, num) => {
      const p = (ms - T.group[g] - k*T.gap) / T.drop;
      if(p <= 0) return;
      const q = Math.min(1,p), lift = 1-ease(q), sq = q>=1 ? 0 : Math.max(0,(q-.8)/.2)*Math.sin(q*Math.PI);
      ctx.save(); ctx.globalAlpha = Math.min(1, q*3); puck(pos[0],pos[1],color,lift,sq,square,num); ctx.restore();
    };
    place(coach, 0, 0, '#10202c', true);
    keeper.forEach(pos => place(pos, 1, 0, '#E2A93B', false, '1'));
    const seen = [0,0,0,0,0], NUM = ['2','3','4','5','6','8','10','7','9','11'];
    team.forEach((pos,i) => { const g = groupOf(i); place(pos, g, seen[g]++, '#2B4C8C', false, NUM[i]); });
    cloud(Math.max(0,Math.min(1,(ms - T.cloud)/500)));
    strokes.forEach(s => { const st = (ms - s.at)/T.stroke; if(st>0) marker(s, Math.min(1,st)); });
    labels.forEach(el => { el.style.opacity = ms >= T.group[+el.dataset.g] + T.drop*0.6 ? 1 : 0; });
  }

  function layout(){
    const b = cv.getBoundingClientRect(); W=b.width; H=b.height; dpr=Math.min(2, window.devicePixelRatio||1);
    cnv.width=Math.round(W*dpr); cnv.height=Math.round(H*dpr);
    // 움직이지 않는 보드(테두리·피치·제목 카드)는 한 번만 그려 둔다
    noise = document.createElement('canvas'); noise.width=cnv.width; noise.height=cnv.height;
    ctx = noise.getContext('2d'); ctx.setTransform(dpr,0,0,dpr,0,0); frame(); board(); ctx = mainCtx;
    draw(shown);
  }

  let raf = 0;
  function play(){
    cancelAnimationFrame(raf);
    if(reduce){ draw(T.end); return; }
    const t0 = performance.now();
    const step = now => { const ms = now - t0; draw(ms); if(ms < T.end) raf = requestAnimationFrame(step); };
    draw(0); raf = requestAnimationFrame(step);
  }

  document.getElementById('replay').addEventListener('click', play);
  layout();
  new ResizeObserver(() => layout()).observe(cv);
  play();
})();
