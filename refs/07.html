    블록이 과제 순서대로 끼워지며 하나의 유아학교가 됩니다.

    가져갈 장치

    Takeaways

    과제 목록 = 인벤토리 핫바

    세부 과제를 화면 아래 칸에 블록으로 넣습니다. 칸 번호는 추진 순서, 칸 아래 숫자는 비중. 목차 장표에도 그대로 씁니다.

    완성률 게이지 = 구성 막대

    게이지를 과제 색으로 나눠 채우면 진행률과 구성비가 한 줄에 들어갑니다. 예산 비중, 추진률 어디에나 맞습니다.

    자리→

    스냅 가이드

    들어갈 자리를 점선으로 먼저 보여 주고, 끼워지는 순간만 강조합니다. 과제 하나를 설명하는 장표의 기본 틀입니다.

    완성→1회 조립→정지

    완성 상태로 시작하는 오프닝

    첫 화면이 이미 완성이라 캡처·인쇄해도 표지가 됩니다. 조립 장면은 발표를 열 때 한 번만 재생합니다.

    흐름 설계 메모

    Plan ↔ Blocks

    두 기관은 이미 있는 큰 블록이고, 세부 과제는 둘을 하나로 묶는 작은 블록입니다. 블록 크기와 게이지 칸 너비가 같은 비중을 가리키게 맞춥니다. 비중 수치는 가상입니다.

    블록계획안 항목건물에서 하는 일비중

    안쪽 장표 아이디어

    1. 1표지완성 화면 → 조립 1회 → 정지. 게이지 100%.
    2. 2추진 배경기초판 없이 떨어져 있는 두 블록, 게이지 0%. 문장은 "같은 나이, 다른 기준" 하나만.
    3. 3목표·방침설계도 모드: 완성 건물을 점선 고스트로만 그리고 목표 한 줄.
    4. 4–8세부 과제 5장장마다 블록 하나가 스냅되고 게이지가 그 색만큼 찹니다. 남은 블록은 고스트, 오른쪽에 요지 3줄.
    5. 9기대 효과표지와 같은 구도로 돌아와 완성 건물을 보여 주고 핫바 전체에 ✓.
    6. 10마무리·문의블록을 내려놓은 빈 기초판 위에 담당 부서와 연락처.

    표지에는 일정 띠를 두지 않습니다. 일정이 필요하면 4–8번 장표의 핫바 칸 위에 시기 태그를 작게 답니다.

    중상. 블록 그림은 아래 PNG로 해결되고, 어려운 건 순서 애니메이션 타이밍

    블록 PNG 세트 (투명 배경)

    이미지를 우클릭(휴대폰은 길게 누르기)해 저장하면 PPT에 바로 넣을 수 있습니다.

    PPT로 옮길 때

    1. 배치: 블록 PNG 7장을 완성 위치에 겹쳐 놓습니다. 뒤에 있는 블록부터 앞으로(기초판 → 유치원 → 연결 → 어린이집 → 출입구 → 2층 → 지붕) 쌓아야 가려짐이 맞습니다.
    2. 이동 경로: 과제 블록마다 '나타내기' + 사용자 지정 경로(화면 위에서 제자리로). 효과 옵션에서 부드럽게 끝내기 0, 바운스 끝 0.1초. '이전 효과 다음에' 0.3초씩.
    3. 두 날개: 유치원은 왼쪽, 어린이집은 오른쪽에서 수평 경로로 동시에 들어오게 '이전 효과와 함께'.
    4. 게이지: 둥근 사각형 5개를 비중대로 너비를 줍니다(막대 10cm면 1.5·3.0·2.5·1.0·2.0cm). 블록 착지와 '이전 효과와 함께' 닦아내기(왼쪽에서).
    5. 모핑: 안쪽 장표는 1장에 블록을 핫바 칸에 작게, 다음 장에 건물 자리에 크게 두고 같은 개체 이름(예: '!!지붕')으로 맞춘 뒤 모핑 전환 1초.
    6. 글자: 제목·부제·기관·날짜와 칸 이름만. 설명은 발표자 노트로 보냅니다.

  ctx.setTransform(DPR,0,0,DPR,0,0);ctx.clearRect(0,0,W,H);drawGrid();
  var pl=placedList();
  // 바닥 그림자
  ctx.save();var s=[proj(O,U,-1,-1,0),proj(O,U,11,-1,0),proj(O,U,11,7,0),proj(O,U,-1,7,0)].map(function(p){return[p[0]+U*.25,p[1]+U*.45]});
  poly(ctx,s);ctx.fillStyle="rgba(29,43,79,.16)";ctx.filter="blur("+Math.round(U*.5)+"px)";ctx.fill();ctx.restore();
  if(state.ghosts)B.forEach(function(b){if(!state.placed[b.id]&&!state.fly[b.id])ghost(ctx,b,O,U,false)});
  B.forEach(function(b){if(state.fly[b.id])ghost(ctx,b,O,U,true)});
  B.forEach(function(b){if(state.placed[b.id])drawBlock(ctx,b,O,U,null,pl,{split:b.id==="E"?1:0})});
  if(state.placed.E)drawBlock(ctx,byId.E,O,U,null,pl,{split:2});
  B.forEach(function(b){var f=state.fly[b.id];if(f)drawBlock(ctx,b,O,U,f,pl.concat([b]))});
  // 이름표
  if(state.placed.K){var a=proj(O,U,0,6,3.4);pill("유치원",a[0]-FS*.8,a[1],"#FFC22E","right")}
  if(state.placed.H){var c=proj(O,U,10,0,3.4);pill("어린이집",c[0]+FS*.8,c[1],"#2F7BF5","left")}
  // 스냅 효과
  var now=performance.now();state.fx=state.fx.filter(function(e){return now-e.t<620});
  state.fx.forEach(function(e){
    var k=Math.max(0,(now-e.t)/620),p=proj(O,U,e.x,e.y,e.z),r=U*(1.2+k*3.4);
    ctx.save();ctx.globalAlpha=1-k;ctx.strokeStyle="#fff";ctx.lineWidth=Math.max(2,U*.2)*(1-k*.6);
    ctx.beginPath();ctx.ellipse(p[0],p[1],r*1.22,r*.71,0,0,7);ctx.stroke();
    for(var i=0;i<8;i++){var an=i/8*Math.PI*2,d=U*(1.4+k*3.2),sx=p[0]+Math.cos(an)*d*1.2,sy=p[1]+Math.sin(an)*d*.7-k*U;
      ctx.fillStyle=i%2?"#FFC22E":"#fff";ctx.save();ctx.translate(sx,sy);ctx.rotate(an+k*3);ctx.fillRect(-U*.13,-U*.13,U*.26,U*.26);ctx.restore()}
    ctx.font="700 "+Math.round(FS*1.15)+"px "+getComputedStyle(document.body).getPropertyValue("--f-hud");
    ctx.textAlign="center";ctx.fillStyle="#1D2B4F";ctx.textBaseline="bottom";ctx.globalAlpha=Math.min(1,(1-k)*2);
    ctx.fillText("딱!",p[0],p[1]-U*1.4-k*U*1.2);ctx.textAlign="left";ctx.restore();
  });
}

function renderSolo(b,w,h,pad){
  var c=document.createElement("canvas"),d=Math.min(window.devicePixelRatio||1,2);c.width=w*d;c.height=h*d;var x=c.getContext("2d");x.scale(d,d);
  var minx=Infinity,maxx=-Infinity,miny=Infinity,maxy=-Infinity;
  [b.x[0],b.x[1]].forEach(function(xx){[b.y[0],b.y[1]].forEach(function(yy){[b.z[0],b.z[1]+(b.studs?.25:0)].forEach(function(zz){
    var p=proj([0,0],1,xx,yy,zz);minx=Math.min(minx,p[0]);maxx=Math.max(maxx,p[0]);miny=Math.min(miny,p[1]);maxy=Math.max(maxy,p[1])})})});
  var u=Math.min((w-2*pad)/(maxx-minx),(h-2*pad)/(maxy-miny)),o=[w/2-(minx+maxx)/2*u,h/2-(miny+maxy)/2*u];
  drawBlock(x,b,o,u,null,[],{allStuds:true});return c;
}

}

  [].forEach.call(hot.children,function(li){li.classList.toggle("wait",!done[li.dataset.id]);li.classList.remove("now")});
}

// ---------- 조립 애니메이션 (1회) ----------
function backOut(t){var s=1.9;t-=1;return t*t*((s+1)*t+s)+1}
var running=false;
function play(){
  if(running||reduce)return;running=true;
  var done={};state.placed={};state.fly={};state.ghosts=true;setGauge(done);
  var t0=performance.now()+450,STEP=620,DUR=520,landed={};
  function frame(now){
    ORDER.forEach(function(grp,gi){
      var st=t0+gi*STEP;
      grp.forEach(function(id){
        var b=byId[id];if(landed[id])return;
        if(now<st){delete state.fly[id];return}
        var p=Math.min(1,(now-st)/DUR),e=backOut(p);
        if(p>=1){landed[id]=true;delete state.fly[id];state.placed[id]=true;
          var cx=(b.x[0]+b.x[1])/2,cy=(b.y[0]+b.y[1])/2;state.fx.push({t:now,x:cx,y:cy,z:b.z[1]});
          if(b.task){done[id]=true;setGauge(done)}return}
        var f=b.from,off=[0,0,0];
        if(f.dz)off[2]=f.dz*(1-e);
        if(f.dx){off[0]=f.dx*(1-e);off[2]=Math.sin(Math.PI*p)*1.2}
        state.fly[id]=off;
        if(b.task){var li=hot.querySelector('[data-id="'+id+'"]');li.classList.add("now")}
      });
    });
    draw();
    var end=t0+(ORDER.length-1)*STEP+DUR+650;
    if(now<end)requestAnimationFrame(frame);
    else{state.ghosts=false;state.fx=[];B.forEach(function(b){state.placed[b.id]=true});state.fly={};setGauge(allDone);draw();running=false}
  }
  requestAnimationFrame(frame);
}
var btn=document.getElementById("replay");
if(reduce)btn.hidden=true;
btn.addEventListener("click",play);

function refresh(){if(layout()&&!running)draw();paintIcons()}
var ro=new ResizeObserver(function(){refresh()});ro.observe(stage);
