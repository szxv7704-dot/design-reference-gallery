01동작 설명

핀 텀블러 자물쇠의 단면을 빌려, 성과 지표 다섯 개를 핀 다섯 개로 세웠습니다. 2025년 열쇠(현재값)로는 핀이 정렬선 아래에 걸려 빨간 구간이 남습니다. 열쇠의 톱니가 하나씩 목표 모양으로 변형될 때마다 해당 핀이 정렬선까지 올라가고 숫자가 목표치로 바뀝니다. 다섯 개가 모두 맞으면 고리가 튀어 올라 상위 목표(건강수명)가 열립니다.

핵심은 정규화된 목표선입니다. 지표마다 단위와 목표치는 달라도 "목표 도달"을 같은 높이의 선 하나로 맞춰, 무엇이 얼마나 모자란지(빨간 길이)와 모두 맞아야 결과가 난다는 조건(AND)을 한 화면에 보여 줍니다. 최종 화면에는 2025년 열쇠 모양이 점선으로 남아 전후 비교가 됩니다.

응용 예

  • 인증·평가 준비: 필수 기준 항목을 핀으로, 기준 충족선을 정렬선으로 두고 "모두 충족해야 인증"을 표현
  • 사업 착수 조건: 예산 확보·조례 개정·부지 확보·인력 충원이 모두 갖춰져야 착공이 열리는 선행 조건 도식
  • 기업 KPI: 매출·이익률·고객 유지율·재구매율을 핀으로 두고 목표 연도 열쇠로 바꿔 끼우는 연간 목표 장표

02HTML로 가져다 쓰기

지표 이름·값: 스크립트 맨 위 DATA 배열의 name(두 줄), now, goal을 바꿉니다. 핀 개수를 바꾸면 PIN_X도 같은 개수로 맞춥니다.

결과값: RESULT의 now·goal·unit.

글자·색: 제목·부제는 .lk-title, .lk-sub 마크업에서, 색은 .lk-wrap의 --lk- 변수에서 바꿉니다. 차이가 너무 작게 보이면 K(1%p당 높이)를 키웁니다.

속도: DUR(단계 길이), GAP(단계 사이).

  /* 상태: t[i] 0(현재)→1(목표), open 0→1 */
  var st = { t:DATA.map(function(){return 0;}), open:0 };
  var parts = [];

    el("rect",{x:x-31,y:SHEAR,width:62,height:72,fill:"url(#lk-gCav)"},ch);
    var g = el("g",{},pins);
    var p = {
      spring: el("path",{fill:"none",stroke:"#7D8891","stroke-width":4,"stroke-linejoin":"round"},g),
      driver: el("rect",{x:x-25,width:50,height:DRIVER,rx:9,fill:"url(#lk-gDriver)",stroke:"#5E6973","stroke-width":2},g),
      keypin: el("rect",{x:x-25,width:50,height:KEYPIN,rx:9,fill:"url(#lk-gKeyPin)",stroke:"#8A5E18","stroke-width":2},g),
      red:    el("rect",{x:x-25,y:SHEAR,width:50,height:0,fill:"#E2553B",opacity:.88},g),
      ok:     el("rect",{x:x-34,y:SHEAR-3,width:68,height:6,rx:3,fill:"#0B8A7C",opacity:0},g)
    };
    var c = el("g",{},cards);
    p.card = el("rect",{x:x-56,y:765,width:112,height:126,rx:12,fill:"#FFFFFF","stroke-width":3},c);

  function gapOf(i, t){ return (DATA[i].goal - DATA[i].now) * K * (1 - t); }
  function keyPath(tFn){

      var tt = SHEAR + KEYPIN + gapOf(i, tFn(i));

  function springPath(x, y0, y1){
    var n = 7, w = 19, d = "M"+x+" "+y0, h = (y1 - y0) / n;
    for (var k=0; k<n; k++) d += " L"+(x+(k%2?-w:w))+" "+(y0+h*(k+.5)).toFixed(1);
    return d + " L"+x+" "+y1.toFixed(1);
  }

  function render(){

    $("lk-bigcap").textContent = o >= .999 ? "2030 목표 · 자물쇠 열림" : "2025 현재";
  }

  /* ===== 단계: 지표 다섯 개를 하나씩 맞춘 뒤, 자물쇠가 열린다 ===== */
  var STEPS = DATA.map(function(d, i){
    return { label: "지표 " + (i+1) + "/5 · " + d.name.join(" ") + " " + d.now + "% → " + d.goal + "%", run: function(e){ st.t[i] = e; } };
  }).concat([{ label: "다섯 핀이 정렬선에 맞춤 → 고리가 열림 · 건강수명 " + RESULT.now + " → " + RESULT.goal + RESULT.unit, run: function(e){ st.open = e; } }]);

  function easeBack(x){ var c = 1.6; return 1 + (c+1)*Math.pow(x-1,3) + c*Math.pow(x-1,2); }
  var timer = null, raf = null, idx = 0;
  function stopAll(){ clearTimeout(timer); cancelAnimationFrame(raf); }
  function tween(step, done){
    var t0 = performance.now();
    (function frame(now){
      var x = Math.min(1, (now - t0) / DUR);
      step.run(x < 1 ? easeBack(x) : 1); render();
      if (x < 1) raf = requestAnimationFrame(frame); else if (done) done();
    })(t0);
  }
  function setLabel(s){ $("lk-step").textContent = s; }
  function reset(){ st.t = st.t.map(function(){return 0;}); st.open = 0; idx = 0; render(); }
  function finalState(){ st.t = st.t.map(function(){return 1;}); st.open = 1; idx = STEPS.length; render(); }

  function play(){
    stopAll(); reset(); setLabel("2025 현재 열쇠: 다섯 핀이 모두 정렬선 아래에 걸림");
    (function next(){
      if (idx >= STEPS.length) { setLabel("완료 · " + STEPS[STEPS.length-1].label); return; }
      var s = STEPS[idx++]; setLabel(s.label);
      tween(s, function(){ timer = setTimeout(next, GAP); });
    })();
  }
  function stepOnce(){
    stopAll();
    if (idx >= STEPS.length) { reset(); setLabel("처음: 2025 현재 열쇠 (다시 누르면 한 단계씩)"); return; }
    STEPS.slice(0, idx).forEach(function(s){ s.run(1); });
    var s = STEPS[idx++]; setLabel(s.label); tween(s);
  }

  $("lk-replay").addEventListener("click", play);
  $("lk-next").addEventListener("click", stepOnce);

  /* 처음 열면: 완성 도식 → 처음부터 한 번 재생 → 최종 정지. 움직임 줄이기 설정이면 완성 도식만 */
  finalState(); setLabel("완성 도식");
  var reduce = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  if (!reduce) timer = setTimeout(play, 700);
})();
</script>

03PPT로 옮기기

자물쇠 그림은 도형(사각형·원호)과 그라데이션 채우기로 그리거나, 이 페이지의 최종 화면을 캡처해 몸통·창만 배경으로 쓰고 핀·톱니·고리·숫자만 개체로 올립니다. 움직임은 모핑 전환 6장으로 만드는 것이 가장 가깝습니다.

  1. 1번 슬라이드(현재): 열쇠 날을 자유형 도형으로 그리고 이름을 !!key로 바꿉니다([홈] → [정렬] → [선택 창]). 핀 위아래 두 개를 그룹으로 묶어 !!pin1~!!pin5, 빨간 구간 !!red1~!!red5, 고리 !!shackle, 지표 카드의 숫자 상자 !!val1~!!val5, 결과 숫자 !!result.
  2. 2~6번 슬라이드: 앞 슬라이드를 복제한 뒤 한 장에 지표 하나씩 바꿉니다. 열쇠 날을 우클릭 → [점 편집]으로 해당 톱니 꼭짓점만 위로 올리고, 같은 핀 그룹을 정렬선까지 위로 옮기고, 빨간 구간은 높이 0(또는 삭제)으로, 숫자는 목표치로 고칩니다. 카드 테두리 색은 주황 → 청록.
  3. 7번 슬라이드(열림): 고리 !!shackle을 위로 약 1cm 옮기고 결과 숫자를 72.0세로, 정렬선에 청록 [네온] 효과를 줍니다.
  4. 2~7번 슬라이드 모두 [전환] → 모핑, [효과 옵션] → 개체(숫자가 바뀌는 장면은 문자), 기간 0.70초, [화면 전환] → 마우스를 클릭할 때(자동 재생은 [다음 시간 후] 00:00.30).
