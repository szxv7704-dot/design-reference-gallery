학생은 2030년까지 3분의 1이 줄지만,
학급과 교원은 10% 안쪽으로 준다

가상 예시 · 출처: ○○교육청 교육통계(각 연도 4. 1. 기준, 초·중·고), 2026~2030년은 출생아 수 기반 추계 4
좁은 화면에서는 장표를 옆으로 밀어 보세요.
계열 값 바꿔 보기

01구조

지표 세 개(학생·학급·교원)를 한 그래프에 겹치지 않고, 같은 연도 축을 공유하는 세 줄로 나란히 놓았습니다. 각 줄의 막대는 그 지표의 0부터 선형으로 그려서, 학생 막대는 눈에 띄게 낮아지고 학급·교원 막대는 거의 같은 높이를 유지합니다. 단위가 서로 달라 높이만으로는 비교가 어려우므로, 오른쪽 열에서 같은 기간 증감률(%)을 하나의 공통 눈금으로 다시 비교합니다.

가져갈 장치: 줄마다 따로 그린 추이 + 공통 눈금 증감률 열 + 전망 구역 면단위가 다른 지표는 각자의 0부터 그리고, 비교는 오른쪽 % 열 하나에 몰아서 합니다. 2026년 이후는 같은 막대를 빗금으로 바꾸고 흰 면 위에 올려서, 실적과 추계가 정지 화면에서도 갈립니다. 강조색은 학생 수 증감률 한 곳에만 씁니다.
  • 교육 예산: 총예산·인건비·학교운영비 연도별 추이 + 2030년 중기재정 전망
  • 늘봄·돌봄: 참여 학생 수·프로그램 수·강사 수 추이와 증감률
  • 학교 시설: 노후 건물 동수·석면 잔여 학교 수·개선 예산 추이

02HTML로 가져다 쓰기

  • 데이터는 스크립트 맨 위 ROWS 배열만 고치면 됩니다. 값 15개(실적 10 + 전망 5), 강조할 줄 하나에 key:true.
  • 색·글꼴은 .tr48의 --ds-* 변수에서 바꿉니다. 계열 값은 감싸는 요소에 data-fam="B"(또는 D, F)를 붙이면 됩니다.
  • 연도가 바뀌면 YEARS와 현재 위치 NOW를 함께 고칩니다.
  • AI 채팅에는 링크 대신 이 코드를 붙여 넣으세요.

(function(){
  /* ── 데이터: 여기만 바꾸면 된다 ──

  var NOW = 9; /* 2025 = 현재 = 마지막 실적 */
  var ROWS = [
    {name:"학생 수", unit:"천 명 · 초·중·고 재적", key:true, dec:1,
     vals:[212.4,208.1,203.6,199.0,195.2,191.3,186.9,181.7,176.0,170.2,164.1,157.6,150.8,144.5,139.3]},

  ];
  var PCT_MAX = 40; /* 증감률 막대 눈금 끝(%) */

  /* ── 배치(가상 1600×900 좌표) ── */

  var u = function(n){ return "calc(var(--u)*" + n + ")"; };
  function el(cls, css, html, parent){
    var d = document.createElement("div"); d.className = "tr48-abs " + cls;
    for (var k in css) d.style[k] = (typeof css[k] === "number") ? u(css[k]) : css[k];
    if (html != null) d.innerHTML = html; (parent || body).appendChild(d); return d;
  }
  function fmt(v, dec){ return v.toLocaleString("ko-KR",{minimumFractionDigits:dec,maximumFractionDigits:dec}); }

  /* 전망 구역 면과 열 머리 */
  var zx = X0 + 10*STEP - 6;
  el("tr48-zone", {left:zx, top:226, width:5*STEP + 10, height:572});
  el("tr48-zlab", {left:X0, top:234}, "<strong>실적</strong> 2016–2025");

  el("tr48-colhead", {left:RX - RW - 10, top:234, width:RW + 10, textAlign:"right"}, "<strong>증감률</strong> 2016→2030");
  /* 증감률 눈금선 */
  [0, 20, 40].forEach(function(t){

    el("tr48-grid", {left:x, top:268, height:520});
    el("tr48-tick", {left:x - 30, top:794}, t === 0 ? "0" : "−" + t + "%");
  });

  ROWS.forEach(function(r, ri){
    var g = el("tr48-in", {left:0, top:0, width:1600, height:900, animationDelay:(ri*0.35) + "s"});
    var top = ROW_TOP[ri], base = top + BASE_DY, max = Math.max.apply(null, r.vals);

    el("", {left:80, top:top + 52, width:250},

    el("tr48-base", {left:X0 - 4, top:base, width:15*STEP + 4}, null, g);
    el("tr48-zero", {left:X0 - 30, top:base - 9}, "0", g);
    r.vals.forEach(function(v, i){
      var h = PLOT_H * v / max;              /* 선형: 0부터 */
      var b = el("tr48-bar" + (i > NOW ? " is-proj" : ""),
        {left:X0 + i*STEP + (STEP - BW)/2, top:base - h, width:BW, height:h}, null, g);
      b.style.setProperty("--tr48-col", col);

    });
    /* 증감률 */
    var a = r.vals[0], z = r.vals[r.vals.length - 1], p = (z - a) / a * 100;
    var w = RW * Math.abs(p) / PCT_MAX;
    el("tr48-pct" + (r.key ? " is-key" : ""), {left:RX - RW, top:top + 40, width:RW},

    el("tr48-rbar" + (r.key ? " is-key" : ""), {left:RX - w, top:top + 84, width:w, height:24}, null, g);
    el("tr48-from", {left:RX - RW, top:top + 116, width:RW},
      fmt(a, r.dec) + " → " + fmt(z, r.dec), g);
  });

  /* 연도 축 + 파생 지표 */
  var gy = el("tr48-in", {left:0, top:0, width:1600, height:900, animationDelay:"1.05s"});
  YEARS.forEach(function(y, i){
    el("tr48-year" + (i === NOW ? " is-now" : ""), {left:X0 + i*STEP, top:794},
      (i === 0 || i === NOW || i === YEARS.length - 1) ? String(y) : "’" + String(y).slice(2), gy);
  });
  var s = ROWS[0].vals, c = ROWS[1].vals;
  var per = function(i){ return (s[i] * 1000 / c[i]).toFixed(1); };
  el("tr48-derive", {left:80, top:796},
    "학급당 학생 수 <b>" + per(0) + " → " + per(NOW) + " → " + per(YEARS.length - 1) + "명</b>", gy);

03PPT로 옮기기

  1. 뼈대: 슬라이드 13.33×7.5in 기준, 좌우 여백 0.67in. 머리말 띠는 위 0.33in에 텍스트 상자 2개(왼쪽 섹션, 오른쪽 계획안 이름 11pt 회색) + 아래 가는 선 0.75pt. 제목은 위 0.93in, 28pt 굵게, 핵심어 "3분의 1"만 강조색.
