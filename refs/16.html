가져갈 장치

이 표지에서 떼어 다른 발표에 붙일 수 있는 장치 두 가지입니다.

장치 A

한 개체가 자라면서 일지에 체크가 찍히는 표지

씨앗 하나가 60일 동안 자라고, 그날에 닿을 때마다 일지 줄에 체크가 찍힙니다. 성과표 대신 "언제, 무엇이 달라지는지"를 시간 순서로 보여 주므로 수치가 아직 없는 계획안에서도 일정 약속이 됩니다.

  • 연수 결과보고회차가 지날 때마다 한 줄씩 체크
  • 특수교육 지원개별화 목표를 월별로 체크, 사진은 그달 모습
  • 유아 놀이 사업관찰 기록 문화와 그대로 맞물림
장치 B

땅 위보다 땅 아래가 먼저, 더 넓게 자라는 단면

재생해 보면 처음 2주 동안은 땅 위에 아무것도 없고 뿌리만 뻗습니다. 꽃이 필 무렵 뿌리는 줄기보다 몇 배 넓게 퍼져 있습니다. 예산이나 인력처럼 겉으로 드러나지 않는 투입이 왜 먼저 필요한지 설명할 때 힘이 있습니다.

땅 위: 학부모·학생이 체감하는 변화 (Day 18부터 보임)
땅 아래: 진단 도구, 협력강사, 연수, 예산 (Day 2부터 자람)
  • 예산 설명아래 = 투입 내역, 위 = 산출
  • 조직·업무 개편아래 = 지원 체계, 위 = 현장 서비스

흐름 설계 메모

계획안의 장과 일지 한 줄, 생장 시점을 1:1로 맞춥니다. 표지에서 한 번 끝까지 자라는 모습을 보여 준 뒤, 각 장 도입 장표에서는 그 장의 날까지만 자란 그림을 다시 띄웁니다. 발표가 진행될수록 식물도 함께 자랍니다.

      if (t >= 46) {
        const b = seg(t, 46, 52), o = seg(t, 52, 59), fx = sx(yt), fy = yt - 6 * b;
        above += `<path d="M${r1(fx - 5 * b)},${r1(yt + 2)} L${r1(fx)},${r1(yt - 6 * b)} L${r1(fx + 5 * b)},${r1(yt + 2)}Z" fill="#5E9E48"/>`;
        if (o < 1) above += `<ellipse cx="${r1(fx)}" cy="${r1(fy - 3 * b)}" rx="${r1(5.5 * b)}" ry="${r1(9 * b)}" fill="url(#gBud)" opacity="${r1((1 - o) * 10) / 10}"/>`;
        if (o > 0) {
          for (let k = 0; k < 5; k++) {
            const a = (-90 + 12 + k * 72) * D2R, px = fx + Math.cos(a) * 10.5 * o, py = fy - 4 + Math.sin(a) * 10.5 * o;
            above += `<ellipse cx="${r1(px)}" cy="${r1(py)}" rx="${r1(12.5 * o)}" ry="${r1(8.8 * o)}" transform="rotate(${r1(a / D2R)} ${r1(px)} ${r1(py)})" fill="url(#gPetal)" stroke="#D86F8E" stroke-width=".8"/>`;
          }
          above += `<circle cx="${r1(fx)}" cy="${r1(fy - 4)}" r="${r1(5.2 * o)}" fill="url(#gCore)"/>`;
        }
        const ld = document.getElementById('ldBloom'), dot = document.getElementById('ldBloomDot');
        ld.setAttribute('d', `M${r1(fx + 24)},${r1(fy - 4)} L508,${r1(fy - 4)}`); dot.setAttribute('cx', r1(fx + 24)); dot.setAttribute('cy', r1(fy - 4));
      }
    }
    plant.innerHTML = below + `<g filter="url(#fDrop)">${above}</g>`;
  }

  // timeline + journal sync
  const range = document.getElementById('day'), out = document.getElementById('dayOut').querySelector('b');
  const rows = [...document.querySelectorAll('.log li')], photoDay = document.getElementById('photoDay');
  const gated = [...document.querySelectorAll('[data-from]')];
  const bPlay = document.getElementById('bPlay'), bRe = document.getElementById('bRe');
  let day = 60, playing = false, last = 0, raf = 0;
  const SPEED = 60 / 9;   // whole 60 days in 9 seconds

  function show(t) {
    day = Math.max(0, Math.min(60, t));
    render(day);
    const d = Math.floor(day + 1e-6);
    out.textContent = d; photoDay.textContent = 'Day ' + d;
    range.value = day.toFixed(1);
    range.style.setProperty('--p', (day / 60 * 100).toFixed(2) + '%');
    let now = -1;
    rows.forEach((r, i) => { const done = day >= +r.dataset.d - 1e-6; r.classList.toggle('done', done); if (done) now = i; });
    rows.forEach((r, i) => r.classList.toggle('now', i === now));
    gated.forEach(el => el.classList.toggle('off', day < +el.dataset.from));
  }
  function setPlaying(p) {
    playing = p; bPlay.classList.toggle('on', p); bPlay.setAttribute('aria-label', p ? '일시정지' : '재생');
    cancelAnimationFrame(raf);
    if (p) { last = performance.now(); raf = requestAnimationFrame(tick); }
  }
  function tick(now) {
    const dt = Math.min(.1, (now - last) / 1000); last = now;
    show(day + dt * SPEED);
    if (day >= 60) { setPlaying(false); return; }
    raf = requestAnimationFrame(tick);
  }
  bPlay.addEventListener('click', () => { if (playing) setPlaying(false); else { if (day >= 60) show(0); setPlaying(true); } });
  bRe.addEventListener('click', () => { show(0); setPlaying(true); });
  range.addEventListener('input', () => { setPlaying(false); show(+range.value); });

  show(60);
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let touched = false;
  [bPlay, bRe, range].forEach(el => el.addEventListener('pointerdown', () => { touched = true; }));
  range.addEventListener('keydown', () => { touched = true; });
  if (!reduce) setTimeout(() => { if (!touched) { show(0); setPlaying(true); } }, 1400);
  window.__growTo = show;
})();
