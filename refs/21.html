01동작 설명

학생 한 명 한 명을 점으로 놓고, 점들이 세 겹의 체를 차례로 통과하는 모습으로 선별·연계 체계를 보여 줍니다. 위에서 모든 점이 첫 체에 모이고(전수 검사), 대부분은 옆 받침으로 미끄러져 학급 예방교육으로 갑니다. 관심 신호가 있는 점만 주황으로 바뀌어 아래 체로 떨어집니다. 체는 아래로 갈수록 폭이 좁아지고 그물이 촘촘해지며, 받침의 지원은 깊어집니다. 점 개수가 실제 인원 비율(점 1개 = 50명)과 같아서, 그림만 봐도 '대상은 좁아지고 지원은 깊어진다'가 읽힙니다.

움직임이 끝난 완성 화면에도 체 세 겹이 나무 테와 그물로 남고, 각 체 위에는 그 단계에서 걸러진 점 몇 개, 체 사이에는 아래로 떨어지는 주황 점이 남아 있습니다. 인쇄하거나 정지 화면으로 넘겨도 '거른다'는 흐름이 읽힙니다.

기초학력 진단·지원1차 진단 → 학습지원팀 → 클리닉센터. 받침 이름과 인원만 바꾸면 됩니다.
학교폭력 사안 처리신고 접수 → 학교장 자체 해결 → 심의위원회. 체마다 처리 기준을 아래 설명 줄에 씁니다.
공모 사업 심사신청 → 서류 → 현장 → 최종 선정. 체를 4단으로 늘려 탈락·보완·선정 받침을 나눕니다.

  var cur = SV_STEPS.length, token = 0, raf = 0;
  function applyState(n){ cancelAnimationFrame(raf); cur = n; dots.forEach(function(d){ place(d, d.pos[ORDER[n]], n); }); setGroups(n); show(); }
  function show(){ if (!status) return; status.textContent = cur >= SV_STEPS.length ? '완성 상태 · ' + SV_STEPS.length + '단계' : cur === 0 ? '시작 전 · 0/' + SV_STEPS.length : cur + '/' + SV_STEPS.length + ' · ' + SV_STEPS[cur - 1].label; }
  function ease(t){ return t < .5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2; }
  function playStep(i){
    return new Promise(function(done){
      cur = i + 1; show(); setGroups(cur);
      var from = ORDER[i], to = ORDER[i + 1], dur = ms('--sv-move-ms'), st = ms('--sv-stagger-ms'), t0 = performance.now();
      var movers = dots.filter(function(d){ var a = d.pos[from], b = d.pos[to]; return a[0] !== b[0] || a[1] !== b[1]; });
      var total = dur + movers.length * st;
      dots.forEach(function(d){ d.el.setAttribute('fill', colorAt(d, cur)); d.el.style.opacity = 1; });
      function frame(now){
        movers.forEach(function(d, k){ var t = Math.min(1, Math.max(0, (now - t0 - k * st) / dur)); var e = ease(t), a = d.pos[from], b = d.pos[to];
          var lift = (b[1] > a[1] + 40) ? 0 : Math.sin(Math.PI * e) * -18;
          d.el.setAttribute('cx', (a[0] + (b[0] - a[0]) * e).toFixed(1)); d.el.setAttribute('cy', (a[1] + (b[1] - a[1]) * e + lift).toFixed(1)); });
        if (now - t0 < total) raf = requestAnimationFrame(frame); else done();
      }
      raf = requestAnimationFrame(frame);
    });
  }
  function wait(t){ return new Promise(function(r){ setTimeout(r, t); }); }
  // 다시 보기: 처음부터 끝까지 한 번 재생하고 완성 상태에서 멈춤
  function replay(){
    var my = ++token;
    if (reduced){ applyState(SV_STEPS.length); return; }
    applyState(0);
    var chain = wait(400);
    SV_STEPS.forEach(function(_, i){ chain = chain.then(function(){ if (my === token) return playStep(i).then(function(){ return wait(ms('--sv-gap-ms')); }); }); });
  }
  // 한 단계씩: 클릭할 때마다 다음 동작 하나 (끝까지 가면 처음부터)
  function stepOnce(){
    ++token;
    var next = cur >= SV_STEPS.length ? 0 : cur;
    applyState(next);
    if (reduced){ applyState(next + 1); return; }
    playStep(next);
  }
  root.querySelector('[data-sv-replay]').addEventListener('click', replay);
  root.querySelector('[data-sv-step]').addEventListener('click', stepOnce);
  applyState(SV_STEPS.length);            // 첫 화면은 완성된 도식
  if (!reduced) setTimeout(replay, 1200); // 잠시 뒤 한 번 재생
})();
</script>
<!-- ===== 3단계 선별 체 도식 끝 ===== -->
  1. 색은 .sv-root의 --sv-t1~--sv-t3(받침별 색)와 --sv-flag(관심 신호)만 바꾸면 됩니다. 체의 테 색은 --sv-rim(윗면)·--sv-rim-side(옆면), 그물 색은 --sv-mesh입니다.
  2. 체 이름·담당은 SV_SIEVES에서 바꿉니다. 같은 줄의 rx(체 폭의 절반)와 mesh(그물 칸 크기)를 아래 체일수록 작게 두면 '좁아지고 촘촘해지는' 모양이 유지됩니다.
  3. 받침 이름·인원·점 개수는 SV_TRAYS에서 바꿉니다. 점 개수(n)는 인원 비율에 맞춰 정하세요(예: 1개 = 50명).
  4. 완성 화면에 남는 점 자리는 REST 배열(체 중심에서의 거리)로 정합니다.
  5. 속도는 --sv-move-ms(이동 시간)와 --sv-stagger-ms(점 사이 시차)로 조정합니다. 점이 300개를 넘으면 시차를 2ms 이하로 줄이세요.
  6. 단계 문구와 순서는 SV_STEPS 배열 한 곳에서 정합니다. 한 페이지에 한 번만 넣는 것을 전제로 합니다.
