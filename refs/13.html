②동작 설명

카드가 시간표 판에 안착하면 채움, 판에 맞고 튕겨 나오면 덜어 냄입니다. 같은 던지기 동작에서 결과만 갈리기 때문에, 청중은 "후보 → 기준 → 편성"이라는 판정 구조를 글 없이 읽습니다. 안착한 카드가 옆으로 펼쳐지는 폭은 요일 수(매일·주 2회·주 3회)를 뜻하고, 왼쪽 숫자는 채운 것과 덜어 낸 것의 누계입니다.

질감은 농구의 슛과 리바운드에서 빌렸습니다. 들어간 공은 그물에서 한 번 출렁이고(스쿼시·오버슈트), 림에 맞은 공은 뒤로 튀어 굴러 떨어집니다. 거절 순간 판 테두리가 잠깐 붉어지는 것이 "기준에 걸렸다"는 신호입니다.

응용 예

  • 늘봄학교 프로그램 편성요일 × 교시 판에 프로그램을 던져 넣고, 수요조사 미달 프로그램은 튕겨 냅니다.
  • 특수교육 지원 인력 배치학교 × 학기 칸에 인력 카드를 안착시키고, 배치 기준에 맞지 않는 요청은 '재검토'로 보냅니다.
  • 예산 사업 구조조정유지 사업은 판에 안착, 일몰·통합 사업은 튕겨 나와 '정리 대상' 트레이에 쌓입니다.

    setCounts(STEPS.length);
  }

  // 4) 움직임: 포물선 좌표를 프레임으로 찍어 등속 수평·가속 수직 궤적을 만듭니다
  function T(x, y, r, sx, sy) {
    return 'translate(' + x.toFixed(1) + 'px,' + y.toFixed(1) + 'px) rotate(' + r.toFixed(1) + 'deg) scale(' + sx + ',' + sy + ')';
  }
  function arcFrames(frames, from, to, lift, o0, o1, r0, r1, clip) {
    var N = 12;
    for (var i = 1; i <= N; i++) {
      var t = i / N;
      var x = from.x + (to.x - from.x) * t;
      var y = from.y + (to.y - from.y) * t - lift * 4 * t * (1 - t);
      var f = { offset: o0 + (o1 - o0) * t, transform: T(x, y, r0 + (r1 - r0) * t, 1, 1) };
      if (clip) f.clipPath = clip;
      frames.push(f);
    }
  }

  // 채움: 후보 더미 → 칸 중앙으로 포물선, 착지하며 찌그러졌다가(스쿼시) 띠가 요일 폭만큼 펼쳐짐
  function throwTile(el, delay) {
    var S = mid(slide), a = mid(launch), b = mid(el);
    var hold = ms('--ttb-hold'), fly = ms('--ttb-fly'), land = ms('--ttb-land'), total = hold + fly + land;
    var o1 = hold / total, o2 = (hold + fly) / total;
    var cw = Math.min(b.w, S.w * 0.13), p = ((b.w - cw) / 2 / b.w * 100).toFixed(2);
    var rad = (S.w * 0.0045).toFixed(1) + 'px';
    var tight = 'inset(0% ' + p + '% 0% ' + p + '% round ' + rad + ')', open = 'inset(0% 0% 0% 0% round ' + rad + ')';
    var from = { x: a.x - b.x, y: a.y - b.y }, zero = { x: 0, y: 0 };
    var frames = [
      { offset: 0, transform: T(from.x, from.y, -12, 1, 1), clipPath: tight },
      { offset: o1, transform: T(from.x, from.y, -12, 1, 1), clipPath: tight }
    ];
    arcFrames(frames, from, zero, S.h * 0.2, o1, o2, -12, 0, tight);
    var L = 1 - o2;
    frames.push({ offset: o2 + L * 0.18, transform: T(0, 0, 0, 1.08, 0.78), clipPath: tight });
    frames.push({ offset: o2 + L * 0.55, transform: T(0, 0, 0, 0.98, 1.05), clipPath: open });
    frames.push({ offset: o2 + L * 0.8, transform: T(0, 0, 0, 1.01, 0.99), clipPath: open });
    frames.push({ offset: 1, transform: T(0, 0, 0, 1, 1), clipPath: open });
    return el.animate(frames, { duration: total, delay: delay, fill: 'backwards', easing: 'linear' });
  }

  // 덜어 냄: 후보 더미 → 판에 부딪혀 납작해짐 → 판이 붉게 거절 → 굴러 떨어지며 트레이로
  function throwCut(el) {
    var S = mid(slide), a = mid(launch), b = mid(el), bd = board.getBoundingClientRect();
    var hold = ms('--ttb-hold'), fly = ms('--ttb-fly'), land = ms('--ttb-land');
    var total = hold + fly * 1.9 + land;
    var o1 = hold / total, o2 = (hold + fly) / total, o3 = o2 + 0.06, o4 = (hold + fly * 1.9) / total;
    var from = { x: a.x - b.x, y: a.y - b.y };
    var hit = { x: bd.left + bd.width * 0.22 - b.x, y: bd.top + bd.height * 0.5 - b.y };
    var frames = [
      { offset: 0, transform: T(from.x, from.y, -12, 1, 1) },
      { offset: o1, transform: T(from.x, from.y, -12, 1, 1) }
    ];
    arcFrames(frames, from, hit, S.h * 0.2, o1, o2, -12, 0);
    frames.push({ offset: o3, transform: T(hit.x, hit.y, 0, 0.72, 1.12) });
    arcFrames(frames, hit, { x: 0, y: 0 }, S.h * 0.08, o3, o4, 0, -360);
    var L = 1 - o4;
    frames.push({ offset: o4 + L * 0.3, transform: T(0, 0, -360, 1.06, 0.86) });
    frames.push({ offset: o4 + L * 0.7, transform: T(0, 0, -360, 0.99, 1.02) });
    frames.push({ offset: 1, transform: T(0, 0, -360, 1, 1) });
    var cut = getComputedStyle(root).getPropertyValue('--ttb-cut').trim();
    var ring = Math.round(S.w * 0.006) + 'px';
    board.animate([
      { boxShadow: '0 0 0 0 transparent' },
      { boxShadow: '0 0 0 ' + ring + ' ' + cut, offset: 0.25 },
      { boxShadow: '0 0 0 0 transparent' }
    ], { duration: 520, delay: hold + fly * 0.92, easing: 'ease-out' });
    return el.animate(frames, { duration: total, fill: 'backwards', easing: 'linear' });
  }

  // 한 단계 재생: 카드가 여러 장이면 0.11초 간격으로 연달아 던짐
  function playStep(i) {
    var s = STEPS[i], anims = [];
    say(i);
    s.els.forEach(function (e, k) {
      e.classList.remove('ttb-off');
      if (reduce) return;
      anims.push(s.kind === 'fill' ? throwTile(e, k * 110) : throwCut(e));
    });
    return Promise.all(anims.map(function (a) { return a.finished; })).then(function () {
      if (s.kind === 'cut') s.els.forEach(function (e) { e.classList.add('ttb-struck'); });
      setCounts(i + 1);
    });
  }

  function playAll() {
    var my = ++runId;
    stepIdx = -1;
    btnStep.textContent = '한 단계씩';
    if (reduce) { setFull(); say(-1); return; }
    setEmpty();
    var chain = wait(350);
    STEPS.forEach(function (s, i) {
      chain = chain.then(function () {
        if (my !== runId) throw 0;
        return playStep(i);
      }).then(function () { return wait(ms('--ttb-gap')); });
    });
    chain.then(function () { if (my === runId) say(-1); }).catch(function () {});
  }

  // 발표 클릭처럼 한 단계씩: 첫 클릭에 비우고 1단계, 이후 클릭마다 다음 단계
  function nextStep() {
    runId++;
    if (stepIdx < 0 || stepIdx >= STEPS.length) { setEmpty(); stepIdx = 0; }
    else root.getAnimations({ subtree: true }).forEach(function (a) { a.finish(); });
    playStep(stepIdx).catch(function () {});
    stepIdx++;
    btnStep.textContent = stepIdx >= STEPS.length ? '한 단계씩 · 처음부터' : '한 단계씩 (' + stepIdx + '/' + STEPS.length + ')';
  }

  root.querySelector('[data-ttb-act="replay"]').addEventListener('click', playAll);
  btnStep.addEventListener('click', nextStep);
  slide.addEventListener('keydown', function (e) {
    if (['ArrowRight', 'PageDown', ' ', 'Enter'].indexOf(e.key) > -1) { e.preventDefault(); nextStep(); }
  });

  // 5) 시작: 완성 상태를 먼저 보여 주고, 화면에 들어오면 처음부터 한 번 재생
  setFull();
  if (reduce) return;
  var started = false;
  function start() { if (started) return; started = true; setTimeout(function () { if (stepIdx < 0 && runId === 0) playAll(); }, 1200); }
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function (en) { if (en[0].isIntersecting) { io.disconnect(); start(); } }, { threshold: 0.5 });
    io.observe(slide);
  } else start();
})();
</script>
</div>

바꿀 지점

  1. 색·글꼴·타이밍: 맨 위 .ttb { … }의 CSS 변수. 카테고리 색은 --ttb-care·--ttb-play·--ttb-special·--ttb-home, 속도는 --ttb-fly(비행)·--ttb-land(안착)·--ttb-gap(단계 사이).
  2. 출처와 제목: .ttb-path에 계획안 위치(한 줄, 길면 말줄임), .ttb-src b에 원문 한 줄, .ttb-title에 결론 한 문장.
  3. 판의 축: 스크립트의 TIMES(줄)와 DAYS(칸). 개수를 바꾸면 CSS의 repeat(5,1fr) 두 곳도 같은 수로 맞춥니다.
  4. 단계 목록: STEPS 배열 하나가 전부입니다. kind가 'fill'이면 row(줄)와 days(요일 묶음)에 안착, 'cut'이면 튕겨 나와 '덜어 낸 것'에 쌓입니다. 새 카테고리는 .ttb-tile[data-cat="…"] 색 규칙과 범례 한 줄을 추가하세요.
  5. 발표 조작: 장표를 한 번 클릭해 초점을 준 뒤 → · PageDown · 스페이스 · Enter로 한 단계씩 넘어갑니다. 무선 프레젠터도 대부분 이 키를 보냅니다.

PPT에서 안 되는 동작과 근사법

  • 후보 더미에서 날아오는 포물선: 바운드는 위에서 떨어지는 동작이라 궤적이 다릅니다. 궤적까지 살리려면 공 카드를 후보 더미 위치에 두고 이동 경로 › 아치형(또는 사용자 지정 경로)으로 칸 가운데까지 그린 뒤, 효과 옵션 › 효과 탭에서 '바운드 끝내기'를 0.2~0.3초로 둡니다.
  • 착지 스쿼시: 같은 카드에 강조 › 크게/작게(가로 110%, 자동 반대로, 0.15초)를 '이전 효과 다음에'로 추가합니다.
  • 튕겨 나올 때 한 바퀴 회전: 강조 › 회전(360° 시계 반대 방향)을 이동 경로 후반부와 '이전 효과와 함께', 지연 0.70초로 겹칩니다.
  • 꽉 찬 카드 → 점선 카드: 점선 테두리 사본을 겹쳐 두고 나타내기 › 밝기 변화로 바꿉니다.

모핑으로 만들 때

  • 장표를 단계 수만큼 복제합니다(채움 6 + 덜어 냄 2×2 = 10장 + 빈 판 1장).
  • 첫 장은 모든 카드를 후보 더미 자리에 칸 하나 크기로 겹쳐 둡니다. 다음 장마다 카드 하나를 최종 칸으로 옮기고 띠는 줄 폭까지 늘립니다. 모핑이 크기 변화를 '펼쳐짐'으로 보여 줍니다.
  • 덜어 낼 카드는 두 장에 걸쳐 판 왼쪽 → 트레이로 옮깁니다.
  • 모든 장에서 같은 카드는 선택 창(홈 › 정렬 › 선택 창)에서 !!간식, !!바깥, !!영어처럼 !!로 시작하는 같은 이름을 줍니다.
  • 전환 › 모핑, 효과 옵션 › 개체, 기간 0.80초. 포물선과 튕김은 모핑으로 안 되므로 직선으로 미끄러지는 정도로 근사됩니다.
