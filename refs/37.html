레퍼런스 #37 · 분광 프리즘

실험형 표지. 계획 보고 발표에 과학 실험(뉴턴의 프리즘 분광)을 빌렸습니다. 한 줄기 흰 빛이 유리 프리즘을 지나 다섯 색으로 펼쳐지듯, 한 번의 돌봄 신청이 창구를 거쳐 다섯 갈래 서비스로 나뉩니다.

가상 예시

TAKE AWAY가져갈 장치

분광 목차

흰 빛 한 줄기(핵심 정책)가 프리즘(실행 체계)을 지나 색 띠(세부 과제)로 펼쳐집니다. 표지 자체가 목차가 되고, 여기서 태어난 색이 이후 장의 색이 됩니다.

응용‘하나의 브랜드 → 여러 서비스’, ‘한 예산 → 분야별 배분’ 표지에 그대로 씁니다.

색이 태어나는 표지

장 구분 색을 임의로 정하지 않고 스펙트럼 순서(빨강→보라)에서 꺼냅니다. 청중은 표지에서 본 띠 색만 보고도 지금 몇 번째 과제를 듣는지 압니다.

응용안쪽 장표 왼쪽 가장자리에 해당 색 띠 하나만 남겨 장 표시로 씁니다.

한 갈래만 비추기

라벨을 누르면 그 색 띠만 남고 나머지는 흐려집니다. 질의응답 때 “요양 쪽은요?”라는 질문에 표지로 돌아가 그 갈래만 비춥니다.

응용PPT에서는 띠별 투명도만 다른 복제 슬라이드 + 모핑으로 같은 효과를 냅니다.

FLOW흐름 설계 메모

계획안 구성은유안쪽 장표 아이디어
추진 배경흩어진 빛(창구가 여러 곳)방향 없이 흩어진 가는 빛줄기 여러 개 → “신청 창구 7곳, 서류 반복 제출”
목적·방침슬릿을 통과한 한 줄기 흰 빛슬릿 판 하나만 크게. “신청은 한 번, 한 곳에서”
운영 체계(원스톱 창구)유리 프리즘프리즘 단면 안에 접수 → 사례 판정 → 연계의 3면을 표기

추진 일정빛이 스크린에 닿기까지의 거리빛줄기 위에 분기 눈금을 새긴 가로 타임라인

발표는 빛이 지나는 순서대로 왼쪽에서 오른쪽으로만 흐릅니다. 세부 과제 장에서는 다섯 띠 중 하나만 이어 받고, 마무리에서 다섯 띠가 스크린 위에 다시 함께 맺힙니다.

CHECK재현 체크

빌린 대표 장치

  • 슬릿 판: 흰 빛을 한 줄기로 모으는 검은 판
  • 삼각 유리 프리즘: 투명 면, 밝은 모서리, 초록빛 유리 단면
  • 분산 순서: 빨강이 가장 덜, 보라가 가장 많이 꺾임(빨강 위·보라 아래)
  • 스크린 위 스펙트럼 띠

조사 출처

  • Wikipedia, Minimum deviation: 60° 프리즘, n=1.5일 때 최소 편향 약 37°
  • LibreTexts, Refraction by a Prism
  • 광선은 스넬 법칙으로 계산했고, 띠가 잘 보이도록 굴절률 차이만 실제보다 크게 잡았습니다(n 1.47~1.68).

PPT 이식

  • 프리즘: 이등변 삼각형 2개 + 사다리꼴 1개, 채우기 투명도 70~80%, 흰 윤곽 3pt, 아랫변만 연한 초록 윤곽
  • 빛줄기: 자유형 삼각형 다섯 개, 그라데이션 채우기(프리즘 쪽 투명 90% → 스크린 쪽 20%)
  • 흰 빛: 흰 선 6pt + 노란 ‘네온’ 효과 8pt
  • 등장: 흰 빛 ‘닦아내기(왼쪽에서)’ 0.6초 → 띠 묶음 ‘확대/축소’ 0.8초 ‘이전 효과 다음에’

(() => {
  // ── 데이터: 띠 라벨(파장 nm 기준 위치) ──
  // f = 스크린에 맺힌 띠 높이 비율(0 위 빨강 ~ 1 아래 보라). 띠를 고르게 나눠 라벨이 겹치지 않게
  const BANDS = [
    { name: '의료',     f: .08 },
    { name: '요양',     f: .29 },
    { name: '주거',     f: .50 },
    { name: '식사',     f: .71 },
    { name: '마음돌봄', f: .92 },
  ];
  const W = 1600, H = 900;                       // 논리 좌표
  const cover = document.getElementById('cover');
  const cv = document.getElementById('cv');
  const ctx = cv.getContext('2d');
  const labelBox = document.getElementById('labels');
  const reduce = matchMedia('(prefers-reduced-motion: reduce)').matches;

  // 프리즘 앞면 삼각형과 깊이(뒤로 물러나는 방향)
  const A = [860, 372], B = [700, 650], C = [1020, 650];
  const D = [62, -40];
  const SLIT = { x: 250, y: 0 };                 // y는 계산
  const SCREEN_X = 1180;                          // 스크린 앞면 왼쪽 선

  // 파장 → 굴절률(시각용 과장), 파장 → RGB
  const nOf = nm => 1.47 + 0.21 * (700 - nm) / 300;
  function rgbOf(nm) {
    let r = 0, g = 0, b = 0;
    if (nm < 440) { r = (440 - nm) / 60; b = 1; }
    else if (nm < 490) { g = (nm - 440) / 50; b = 1; }
    else if (nm < 510) { g = 1; b = (510 - nm) / 20; }
    else if (nm < 580) { r = (nm - 510) / 70; g = 1; }
    else if (nm < 645) { r = 1; g = (645 - nm) / 65; }
    else { r = 1; }
    const f = nm > 680 ? .75 + .25 * (700 - nm) / 20 : nm < 420 ? .6 + .4 * (nm - 380) / 40 : 1;
    if (r < .35 && b < .35) g *= .82;              // 형광 초록을 조금 눌러 줌
    const k = v => Math.round(255 * Math.pow(Math.min(1, v * f), .8));
    return [k(r), k(g), k(b)];
  }

  // 벡터 도우미
  const sub = (p, q) => [p[0] - q[0], p[1] - q[1]];
  const norm = v => { const l = Math.hypot(v[0], v[1]); return [v[0] / l, v[1] / l]; };
  const dot = (a, b) => a[0] * b[0] + a[1] * b[1];
  function refract(d, n, eta) {                  // n: 입사 쪽을 향한 법선
    const c = -dot(n, d), k = 1 - eta * eta * (1 - c * c);
    if (k < 0) return null;
    const t = eta * c - Math.sqrt(k);
    return norm([eta * d[0] + t * n[0], eta * d[1] + t * n[1]]);
  }
  function hitSeg(p, d, a, b) {                  // 광선 p + s·d 와 선분 ab
    const e = sub(b, a), den = d[0] * e[1] - d[1] * e[0];
    if (Math.abs(den) < 1e-9) return null;
    const w = sub(a, p);
    const s = (w[0] * e[1] - w[1] * e[0]) / den, u = (w[0] * d[1] - w[1] * d[0]) / den;
    return (s > 0 && u >= 0 && u <= 1) ? [p[0] + s * d[0], p[1] + s * d[1]] : null;
  }

  // 입사 지점과 방향(최소 편향 근처)
  const P = [B[0] + .52 * (A[0] - B[0]), B[1] + .52 * (A[1] - B[1])];
  const inAng = -17 * Math.PI / 180;
  const dIn = [Math.cos(inAng), Math.sin(inAng)];
  SLIT.y = P[1] + (SLIT.x - P[0]) * Math.tan(inAng) * -1 * -1;
  SLIT.y = P[1] - (P[0] - SLIT.x) * Math.tan(inAng);
  const nLeft = norm([-(A[1] - B[1]), A[0] - B[0]]);           // 바깥 법선 후보
  const nL = dot(nLeft, dIn) < 0 ? nLeft : [-nLeft[0], -nLeft[1]];
  const nRight0 = norm([-(C[1] - A[1]), C[0] - A[0]]);

  function trace(nm) {
    const n = nOf(nm);
    const d1 = refract(dIn, nL, 1 / n);
    const Q = hitSeg(P, d1, A, C);
    let nR = dot(nRight0, d1) > 0 ? [-nRight0[0], -nRight0[1]] : nRight0;   // 안쪽(입사 쪽)을 향하도록
    const d2 = refract(d1, nR, n);
    const s = (SCREEN_X - Q[0]) / d2[0];
    return { Q, d1, d2, S: [SCREEN_X, Q[1] + s * d2[1]] };
  }
  const NMS = []; for (let nm = 700; nm >= 400; nm -= 4) NMS.push(nm);
  const RAYS = NMS.map(nm => ({ nm, ...trace(nm), rgb: rgbOf(nm) }));
  const topY = RAYS[0].S[1], botY = RAYS[RAYS.length - 1].S[1];

  // 스크린 카드(약간 비스듬히 선 흰 판)
  const scr = { x0: SCREEN_X, x1: SCREEN_X + 96, top0: topY - 64, bot0: botY + 46, skew: -26 };

  let focus = -1;   // 강조 띠 index

  function resize() {
    const r = cover.getBoundingClientRect(), dpr = Math.min(2, devicePixelRatio || 1);
    cv.width = Math.round(r.width * dpr); cv.height = Math.round(r.height * dpr);
    ctx.setTransform(cv.width / W, 0, 0, cv.height / H, 0, 0);
    draw(1);
  }

  function rgba(c, a) { return `rgba(${c[0]},${c[1]},${c[2]},${a})`; }
  function bandAlpha(nm) {
    if (focus < 0) return 1;
    const r = RAYS.find(x => x.nm === nm), y = BANDS[focus].y;
    return Math.abs(r.S[1] - y) <= (botY - topY) * .1 ? 1 : .12;
  }

  function draw(t) {
    // t: 0~1 (0~.35 흰 빛 진입, .35~1 분광 펼침)
    const tBeam = Math.min(1, t / .35), tFan = Math.max(0, (t - .35) / .65);
    ctx.clearRect(0, 0, W, H);

    // 바탕: 밝은 스튜디오 스윕
    let g = ctx.createLinearGradient(0, 0, 0, H);
    g.addColorStop(0, '#F8FAFC'); g.addColorStop(.55, '#EEF2F6'); g.addColorStop(1, '#DFE5EC');
    ctx.fillStyle = g; ctx.fillRect(0, 0, W, H);

    g.addColorStop(0, 'rgba(255,255,255,0)'); g.addColorStop(.5, 'rgba(255,255,255,.55)'); g.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = g; ctx.fillRect(0, 600, W, 160);

    drawRail();
    drawShadows(tFan);
    drawTable();
    drawSlit();
    carrier(SLIT.x + 6);
    drawScreenMount();
    drawScreenBack();

    // 흰 빛(슬릿 → 입사점)
    const ex = SLIT.x + (P[0] - SLIT.x) * tBeam, ey = SLIT.y + (P[1] - SLIT.y) * tBeam;
    ctx.save();
    ctx.lineCap = 'round';
    ctx.strokeStyle = 'rgba(255,214,120,.35)'; ctx.lineWidth = 26; ctx.filter = 'blur(8px)';
    ctx.beginPath(); ctx.moveTo(SLIT.x, SLIT.y); ctx.lineTo(ex, ey); ctx.stroke();
    ctx.filter = 'none';
    ctx.strokeStyle = 'rgba(205,170,90,.45)'; ctx.lineWidth = 11;
    ctx.beginPath(); ctx.moveTo(SLIT.x, SLIT.y); ctx.lineTo(ex, ey); ctx.stroke();
    ctx.strokeStyle = '#FFFFFF'; ctx.lineWidth = 7;
    ctx.beginPath(); ctx.moveTo(SLIT.x, SLIT.y); ctx.lineTo(ex, ey); ctx.stroke();
    ctx.restore();

    // 분광 띠(출사점 → 스크린)
    if (tFan > 0) {
      ctx.save();
      for (let i = 0; i < RAYS.length - 1; i++) {
        const r0 = RAYS[i], r1 = RAYS[i + 1];
        const a = bandAlpha(r0.nm);
        const len = tFan;
        const e0 = [r0.Q[0] + (r0.S[0] - r0.Q[0]) * len, r0.Q[1] + (r0.S[1] - r0.Q[1]) * len];
        const e1 = [r1.Q[0] + (r1.S[0] - r1.Q[0]) * len, r1.Q[1] + (r1.S[1] - r1.Q[1]) * len];
        const lg = ctx.createLinearGradient(r0.Q[0], 0, SCREEN_X, 0);
        lg.addColorStop(0, rgba(r0.rgb, .1 * a)); lg.addColorStop(1, rgba(r0.rgb, .5 * a));
        ctx.fillStyle = lg;
        ctx.beginPath(); ctx.moveTo(r0.Q[0], r0.Q[1]); ctx.lineTo(e0[0], e0[1]); ctx.lineTo(e1[0], e1[1] + .6); ctx.lineTo(r1.Q[0], r1.Q[1] + .6); ctx.closePath(); ctx.fill();
      }
      ctx.restore();
    }

    drawPrism(tBeam, tFan);
    drawScreenFront(tFan);
  }

    ctx.strokeStyle = 'rgba(20,26,32,.45)'; ctx.lineWidth = 1;
    for (let k = -rw + 3; k < rw; k += 3) { ctx.beginPath(); ctx.moveTo(cx + k, cy - h / 2 + 1); ctx.lineTo(cx + k, cy + h / 2 - 1); ctx.stroke(); }
    ctx.fillStyle = '#B9C2CC'; ctx.beginPath(); ctx.ellipse(cx, cy - h / 2, rw, 3, 0, 0, Math.PI * 2); ctx.fill();
  }

  const TBL = { cx: 893, cy: 632, rx: 236, ry: 42, t: 12 };

    const t = TBL;
    ctx.save();
    post(t.cx, t.cy + t.ry + t.t - 4, RAIL.y - 24, 15);
    carrier(t.cx);
    // 테이블 두께
    let g = ctx.createLinearGradient(t.cx - t.rx, 0, t.cx + t.rx, 0);

    g = ctx.createLinearGradient(0, t.cy - t.ry, 0, t.cy + t.ry);

  }

  function drawPrism(tBeam, tFan) {
    const A2 = [A[0] + D[0], A[1] + D[1]], B2 = [B[0] + D[0], B[1] + D[1]], C2 = [C[0] + D[0], C[1] + D[1]];
    const poly = (pts) => { ctx.beginPath(); ctx.moveTo(...pts[0]); for (let i = 1; i < pts.length; i++) ctx.lineTo(...pts[i]); ctx.closePath(); };
    ctx.save();
    // 뒷면(유리 통해 보임)
    poly([A2, B2, C2]); ctx.fillStyle = 'rgba(200,218,232,.10)'; ctx.fill();
    // 오른쪽 경사면 (A-C-C2-A2)
    let g = ctx.createLinearGradient(A[0], A[1], C2[0], C2[1]);
    g.addColorStop(0, 'rgba(250,253,255,.7)'); g.addColorStop(.45, 'rgba(160,196,214,.5)'); g.addColorStop(1, 'rgba(112,156,180,.6)');
    poly([A, C, C2, A2]); ctx.fillStyle = g; ctx.fill();
    // 앞면
    g = ctx.createLinearGradient(B[0], A[1], C[0], C[1]);
    g.addColorStop(0, 'rgba(235,245,252,.42)'); g.addColorStop(.6, 'rgba(196,218,232,.30)'); g.addColorStop(1, 'rgba(160,192,212,.40)');
    poly([A, B, C]); ctx.fillStyle = g; ctx.fill();

    // 유리 안의 빛(입사점 → 출사점), 살짝 무지갯빛
    if (tBeam >= 1) {
      const k = tFan > 0 ? 1 : 0;
      ctx.save(); poly([A, B, C]); ctx.clip();
      for (const r of RAYS) {
        if (r.nm % 12) continue;
        ctx.strokeStyle = rgba(r.rgb, .22 * Math.max(k, .4) * bandAlpha(r.nm));
        ctx.lineWidth = 5;
        ctx.beginPath(); ctx.moveTo(P[0], P[1]); ctx.lineTo(r.Q[0], r.Q[1]); ctx.stroke();
      }
      ctx.strokeStyle = 'rgba(255,255,255,.85)'; ctx.lineWidth = 4;
      const mid = RAYS[Math.floor(RAYS.length / 2)];
      ctx.beginPath(); ctx.moveTo(P[0], P[1]); ctx.lineTo(mid.Q[0], mid.Q[1]); ctx.stroke();
      ctx.restore();
      // 입사·출사점 번쩍임
      ctx.save(); ctx.globalCompositeOperation = 'lighter';
      let rg = ctx.createRadialGradient(P[0], P[1], 0, P[0], P[1], 20);
      rg.addColorStop(0, 'rgba(255,250,230,.55)'); rg.addColorStop(1, 'rgba(255,250,230,0)');
      ctx.fillStyle = rg; ctx.fillRect(P[0] - 40, P[1] - 40, 80, 80);
      ctx.restore();
    }

    // 뒷 모서리(옅게)
    ctx.lineJoin = 'round';
    ctx.strokeStyle = 'rgba(110,140,165,.2)'; ctx.lineWidth = 1.2;
    ctx.beginPath(); ctx.moveTo(...A2); ctx.lineTo(...B2); ctx.lineTo(...C2); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(...B); ctx.lineTo(...B2); ctx.stroke();
    // 바닥 유리 단면(초록빛 두께)
    ctx.strokeStyle = 'rgba(90,160,140,.55)'; ctx.lineWidth = 5;
    ctx.beginPath(); ctx.moveTo(B[0] + 3, B[1] - 2); ctx.lineTo(C[0] - 3, C[1] - 2); ctx.lineTo(C2[0] - 2, C2[1] - 2); ctx.stroke();
    // 윤곽(어두운 선)
    ctx.strokeStyle = 'rgba(70,95,120,.55)'; ctx.lineWidth = 1.6;
    poly([A, B, C]); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(...A); ctx.lineTo(...A2); ctx.lineTo(...C2); ctx.lineTo(...C); ctx.stroke();
    // 하이라이트(밝은 모서리)
    ctx.strokeStyle = 'rgba(255,255,255,.95)'; ctx.lineWidth = 3;
    ctx.beginPath(); ctx.moveTo(B[0] + 5, B[1] - 6); ctx.lineTo(A[0], A[1] + 6); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(A[0] + 2, A[1] - 1); ctx.lineTo(A2[0] - 2, A2[1] + 1); ctx.stroke();
    // 왼쪽 면 안쪽 반사띠
    g = ctx.createLinearGradient(B[0], B[1], B[0] + 60, B[1] - 20);
    g.addColorStop(0, 'rgba(255,255,255,.55)'); g.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = g;
    poly([[A[0], A[1] + 18], [B[0] + 16, B[1] - 8], [B[0] + 50, B[1] - 8], [A[0] + 6, A[1] + 50]]); ctx.fill();
    // 오른쪽 면 세로 광택
    g = ctx.createLinearGradient(A[0] + 20, 0, A[0] + 90, 0);
    g.addColorStop(0, 'rgba(255,255,255,0)'); g.addColorStop(.5, 'rgba(255,255,255,.45)'); g.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = g;
    poly([[A[0] + 30, A[1] + 30], [A[0] + 52, A[1] + 16], [A[0] + 150, A[1] + 186], [A[0] + 128, A[1] + 200]]); ctx.fill();
    ctx.restore();
  }

  function scrQuad() {

  function drawScreenFront(tFan) {
    const q = scrQuad();
    ctx.save();
    let g = ctx.createLinearGradient(q[0][0], 0, q[1][0], 0);
    g.addColorStop(0, '#FFFFFF'); g.addColorStop(1, '#F1F4F8');
    ctx.fillStyle = g;
    ctx.beginPath(); q.forEach((p, i) => i ? ctx.lineTo(...p) : ctx.moveTo(...p)); ctx.closePath(); ctx.fill();
    ctx.strokeStyle = 'rgba(120,140,160,.35)'; ctx.lineWidth = 1.2; ctx.stroke();
    // 스크린 위에 맺힌 스펙트럼(판의 기울기를 따라)
    if (tFan >= .98) {
      ctx.save();
      ctx.beginPath(); q.forEach((p, i) => i ? ctx.lineTo(...p) : ctx.moveTo(...p)); ctx.closePath(); ctx.clip();
      const sl = scr.skew / (scr.x1 - scr.x0);
      for (let i = 0; i < RAYS.length - 1; i++) {
        const r0 = RAYS[i], r1 = RAYS[i + 1], a = bandAlpha(r0.nm);
        const w = 74;
        const lg = ctx.createLinearGradient(SCREEN_X, 0, SCREEN_X + w, 0);
        lg.addColorStop(0, rgba(r0.rgb, .95 * a)); lg.addColorStop(1, rgba(r0.rgb, .35 * a));
        ctx.fillStyle = lg;
        ctx.beginPath();
        ctx.moveTo(SCREEN_X, r0.S[1]); ctx.lineTo(SCREEN_X + w, r0.S[1] + sl * w);
        ctx.lineTo(SCREEN_X + w, r1.S[1] + sl * w + .8); ctx.lineTo(SCREEN_X, r1.S[1] + .8); ctx.closePath(); ctx.fill();
      }
      // 맺힌 빛의 번짐
      ctx.globalCompositeOperation = 'screen';
      ctx.filter = 'blur(10px)';
      for (let i = 0; i < RAYS.length; i += 6) {
        const r = RAYS[i];
        ctx.fillStyle = rgba(r.rgb, .12 * bandAlpha(r.nm));

  // ── 라벨: 스크린 오른쪽, 해당 파장 높이 ──
  function rayAtY(y) {
    let best = RAYS[0];
    for (const r of RAYS) if (Math.abs(r.S[1] - y) < Math.abs(best.S[1] - y)) best = r;
    return best;
  }

    el.type = 'button'; el.className = 'c-label';
    b.y = topY + (botY - topY) * b.f;
    const y = b.y, sl = scr.skew / (scr.x1 - scr.x0);
    const yr = y + sl * 96;                        // 스크린 오른쪽 끝에서의 높이
    el.style.left = ((scr.x1 + 6) / W * 100) + '%';
    el.style.top = (yr / H * 100) + '%';
    const c = rayAtY(b.y).rgb;
    el.innerHTML = `<i></i>${b.name}`;
    el.setAttribute('aria-pressed', 'false');
    el.addEventListener('click', () => setFocus(focus === i ? -1 : i));
    labelBox.appendChild(el);
    return el;
  });
  function setFocus(i) {
    focus = i;
    cover.classList.toggle('focus', i >= 0);
    labelEls.forEach((el, k) => { el.classList.toggle('on', k === i); el.setAttribute('aria-pressed', String(k === i)); });
    draw(1);
  }

  // ── 다시 비추기 ──
  let raf = 0;
  function play() {
    cancelAnimationFrame(raf);
    if (reduce) { draw(1); return; }
    const t0 = performance.now(), dur = 1900;
    const step = now => {
      const t = Math.min(1, (now - t0) / dur);
      draw(1 - Math.pow(1 - t, 2.2));
      if (t < 1) raf = requestAnimationFrame(step);
    };
    raf = requestAnimationFrame(step);
  }
  document.getElementById('replay').addEventListener('click', play);

  new ResizeObserver(resize).observe(cover);
  resize();
