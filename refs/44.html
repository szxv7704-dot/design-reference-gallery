Ⅳ. 추진체계 · ○○시 창업·중소기업 원스톱 지원 계획(안)

통합창구 하나에 다섯 기관이 결합해, 기업은 한 번만 신청합니다

    이중 결합상시 협업 · 창구에 직원 상주
    단일 결합요청 시 연계 · 3일 안에 회신
    원자 크기처리 비중
    가상 예시

    동작 설명

    추진체계를 공 막대 분자 모형으로 바꿨습니다. 원자는 기관, 막대는 협업 관계이고 막대 수가 협업 강도입니다. 이중 결합은 창구에 직원이 상주하는 상시 협업, 단일 결합은 요청이 오면 연계하는 관계입니다. 원자 크기는 기관이 맡는 처리 비중입니다.

    움직임 순서가 조직이 짜이는 순서입니다. 통합창구가 먼저 가운데 내려앉고, 상시 기관 둘이 던져져 이중 결합으로 붙고, 요청 시 기관 셋이 단일 결합으로 붙습니다. 결합이 걸리는 순간 원자가 창구 쪽으로 한 번 당겨졌다 돌아와 '붙었다'는 느낌을 줍니다. 이어 기관끼리 직접 잇는 결합이 생기고, 마지막에 기업 원자 하나가 창구에만 붙은 뒤 신청 신호가 창구 → 신용보증재단 → 창업진흥원으로 흐릅니다.

    이렇게 응용할 수 있습니다

    • 부서 간 협업 체계: 주관 부서를 핵으로 두고 상시 TF는 이중, 자문은 단일 결합으로.
    • 민관 거버넌스: 지자체·기업·주민단체를 원자로, 협약 수준을 결합 차수(삼중 결합 = 공동 예산)로.
    • 합병·통합 전후 비교: 같은 원자를 두고 결합만 다시 걸어 조직 개편 전후를 보여 주기.

      x = a.x + dx / L * 300 * (1 - e); y = a.y + dy / L * 300 * (1 - e);
    }
    // 결합이 걸린 직후 허브 쪽으로 한 번 당겨졌다 돌아오는 반동
    BONDS.forEach(function(b){
      if ((b.a === a.id && b.b === 'hub') || (b.b === a.id && b.a === 'hub')){
        var k = (t - (b.t0 + b.dur)) / .45;
        if (k > 0 && k < 1){ var hx = hub.x - a.x, hy = hub.y - a.y, L2 = Math.hypot(hx, hy);
          var m = Math.sin(k * Math.PI) * (1 - k) * 14; x += hx / L2 * m; y += hy / L2 * m; }
      }
    });
    return {x:x, y:y, a:al, sq:sq, p:p};
  }

  function render(t){
    var W = cv.width, H = cv.height, sc = W / VW;
    ctx.setTransform(1,0,0,1,0,0); ctx.clearRect(0, 0, W, H); ctx.setTransform(sc,0,0,sc,0,0);
    var S = {}; ATOMS.forEach(function(a){ S[a.id] = atomState(a, t); });

    // 1) 바닥 그림자
    ATOMS.forEach(function(a){ var s = S[a.id]; if (!s) return;
      var sz = 1 + a.z * .18, r = a.r * sz, lift = (FLOOR - s.y) / 600;
      var g = ctx.createRadialGradient(s.x + 30, FLOOR, 0, s.x + 30, FLOOR, r * 1.5);
      g.addColorStop(0, 'rgba(30,55,80,' + (.22 * s.a / (1 + lift)) + ')'); g.addColorStop(1, 'rgba(30,55,80,0)');
      ctx.fillStyle = g; ctx.beginPath(); ctx.ellipse(s.x + 30, FLOOR, r * 1.5, r * .32, 0, 0, Math.PI*2); ctx.fill(); });

    // 2) 결합 막대(뒤쪽 것부터)
    BONDS.slice().sort(function(m, n){ return (m.A.z + m.B.z) - (n.A.z + n.B.z); }).forEach(function(b){
      var sa = S[b.a], sb = S[b.b]; if (!sa || !sb) return;
      var e = easeOut(clamp((t - b.t0) / b.dur)); if (e <= 0) return;
      var w = 13 * (1 + (b.A.z + b.B.z) * .09);
      var x2 = sa.x + (sb.x - sa.x) * e, y2 = sa.y + (sb.y - sa.y) * e;
      if (b.order === 2){ var dx = sb.x - sa.x, dy = sb.y - sa.y, L = Math.hypot(dx, dy), nx = -dy / L * w * .85, ny = dx / L * w * .85;
        stick(sa.x + nx, sa.y + ny, x2 + nx, y2 + ny, w * .82); stick(sa.x - nx, sa.y - ny, x2 - nx, y2 - ny, w * .82);
      } else stick(sa.x, sa.y, x2, y2, w);
    });

    // 3) 신호(마지막 단계): 기업 → 창구 → 필요한 기관
    var pt0 = starts[4] + STEPS[4].dur * .25, pdur = STEPS[4].dur * .6, pp = (t - pt0) / pdur, glowAt = {};
    if (pp > 0 && pp < 1.15){
      var segs = PULSE.length - 1, u = clamp(pp) * segs, i = Math.min(segs - 1, Math.floor(u)), f = u - i;
      var A = S[PULSE[i]], B = S[PULSE[i + 1]];
      if (A && B){ var px = A.x + (B.x - A.x) * f, py = A.y + (B.y - A.y) * f, fade = pp > 1 ? 1 - (pp - 1) / .15 : 1;
        var g = ctx.createRadialGradient(px, py, 0, px, py, 34);
        g.addColorStop(0, 'rgba(255,255,255,' + fade + ')'); g.addColorStop(.35, 'rgba(232,86,47,' + (.85 * fade) + ')'); g.addColorStop(1, 'rgba(232,86,47,0)');
        ctx.fillStyle = g; ctx.beginPath(); ctx.arc(px, py, 34, 0, Math.PI*2); ctx.fill(); }
      PULSE.forEach(function(id, k){ var d = Math.abs(u - k); if (d < .5) glowAt[id] = 1 - d * 2; });
    }

    // 4) 원자(뒤쪽 것부터)
    ATOMS.slice().sort(function(m, n){ return m.z - n.z; }).forEach(function(a){
      var s = S[a.id]; if (!s) return;
      var r = a.r * (1 + a.z * .18);
      if (glowAt[a.id]){ ctx.save(); ctx.globalAlpha = glowAt[a.id] * .9; ctx.strokeStyle = '#E8562F'; ctx.lineWidth = 6;
        ctx.beginPath(); ctx.arc(s.x, s.y, r + 12, 0, Math.PI*2); ctx.stroke(); ctx.restore(); }
      ctx.save(); ctx.translate(s.x, s.y + r * (1 - s.sq)); ctx.scale(2 - s.sq, s.sq);
      sphere(0, 0, r, a.id === 'biz' ? '#E9EEF2' : a.color, s.a); ctx.restore();
    });

    // 5) 이름표·단계 표시
    ATOMS.forEach(function(a){ var s = S[a.id]; a.lab.style.opacity = s ? clamp((s.p - .35) / .5) : 0; });
    var cur = -1; for (var k = 0; k < STEPS.length; k++) if (t >= starts[k] - 1e-6) cur = k;
    var fin = t >= TOTAL - 1e-6;
    [].forEach.call(stepList.children, function(li, k){
      li.className = fin ? 'done' : k === cur ? 'on' : k < cur ? 'done' : ''; });
    nowEl.textContent = fin ? '완성: 통합창구 1곳 · 기관 5곳 · 결합 8개' : (cur >= 0 ? (cur + 1) + '단계 ' + STEPS[cur].cap : '');
  }

  // ── 재생 제어 ──
  var t = TOTAL, target = TOTAL, raf = null, last = null;
  function tick(now){
    if (last === null) last = now;
    t = Math.min(target, t + (now - last) / 1000); last = now; render(t);
    if (t < target) raf = requestAnimationFrame(tick); else raf = null;
  }
  function play(from, to){ if (raf) cancelAnimationFrame(raf); t = from; target = to; last = null; render(t); raf = requestAnimationFrame(tick); }
  var reduce = window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
  document.getElementById('molReplay').onclick = function(){ reduce ? render(TOTAL) : play(0, TOTAL); };
  document.getElementById('molStep').onclick = function(){
    var from = t >= TOTAL - 1e-6 ? 0 : (raf ? target : t), k = 0;
    if (from >= TOTAL - 1e-6) from = 0;
    while (k < STEPS.length - 1 && starts[k] + STEPS[k].dur <= from + 1e-6) k++;
    var end = starts[k] + STEPS[k].dur;
    if (reduce){ t = end; render(end); } else play(from, end);
  };

  function fit(){ var r = slide.getBoundingClientRect(), d = window.devicePixelRatio || 1;
    cv.width = Math.round(r.width * d); cv.height = Math.round(r.height * d); render(t); }
  if (window.ResizeObserver) new ResizeObserver(fit).observe(slide); else window.addEventListener('resize', fit);
  fit();
  // 열면 완성 도식 → 잠시 뒤 처음부터 한 번 재생 → 최종 정지
  if (!reduce) setTimeout(function(){ play(0, TOTAL); }, 900);
})();
</script>

    모핑으로 만들기(권장)

    • 완성 슬라이드를 만든 뒤 5번 복제합니다. 첫 장부터 단계 순서대로 아직 나오지 않은 개체를 지웁니다.
    • 이름은 선택 창(Alt+F10)에서 !!창구, !!진흥원, !!결합-창구-진흥원처럼 모든 슬라이드에서 똑같이 붙입니다. !!로 시작하는 같은 이름끼리 모핑이 짝을 짓습니다.
    • 앞 슬라이드에는 원자를 지우지 말고 슬라이드 바깥(완성 위치에서 창구 반대 방향)으로 옮겨 두면 날아 들어오고, 결합 막대는 길이를 0에 가깝게 줄여 창구 원 위에 두면 뻗어 나옵니다.
    • 각 슬라이드에 전환 → 모핑, 효과 옵션 개체, 기간 1.00초. 신호는 마지막 장을 3장으로 나눠 !!신호 원을 창구 → 신용보증재단 → 창업진흥원 위치로 옮기면 됩니다.

    안 되는 동작과 근사법

    • 통합창구 착지 눌림: 바운드 효과로 대신하거나, 강조 → 크기 조정(세로 92%, 자동 반대로, 0.15초)을 이어 붙입니다.
    • 원자 뒤로 숨는 막대의 앞뒤 순서: PPT는 개체 순서를 고정하므로, 막대를 모두 원자 뒤로 보내 두면(맨 뒤로 보내기) 같은 결과가 됩니다.
    • 신호가 지날 때 원자 테두리 빛: 강조 → 펄스를 해당 원자에 0.3초, 이동 경로 효과와 함께 시작하고 지연을 0.4초·0.8초로 엇갈립니다.
