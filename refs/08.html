완성 상태

01동작 설명

병풍이 한 폭씩 펼쳐지는 순서가 지원 단계의 순서이고, 앞 폭 안에 다음 폭이 접혀 있다는 구조가 "앞 단계에서 해결되지 않은 학생만 다음 단계로 넘어간다"는 포함 관계를 보여 줍니다. 각 폭 아래 날개가 내려오면서 그 단계의 세부 지원이 드러나고, 마지막 강도 띠가 겹이 넓어질수록 지원은 깊어지고 대상은 줄어든다는 결론을 정리합니다.

연간 추진 일정분기 4폭 병풍. 한 분기씩 펼치고 날개에 월별 행사를 넣는다.
기존안 대 개선안한 폭을 뒤집어 앞면은 기존안, 뒷면은 개선안. 펼침 동작이 곧 "바뀐다"는 말이 된다.
대상별 지원 유형유아·초등·특수 3폭. 날개에 대상별 지원 메뉴를 접어 두고 질의응답 때 필요한 폭만 연다.

    var sh = mv.shade && shadeOf(node);
    if (sh) { sh.getAnimations().forEach(function(a){ a.cancel(); }); sh.style.opacity = 0; }
    Object.assign(node.style, clean(done ? mv.kf[mv.kf.length - 1] : mv.kf[0]));
    node.style.visibility = (!done && mv.hide) ? 'hidden' : 'visible';
  }
  function applyState(n){
    FS_STEPS.forEach(function(s, i){ s.items.forEach(function(it){ setItem(it, i < n); }); });
    cur = n; show();
  }
  function show(){
    if (!status) return;
    status.textContent = cur >= FS_STEPS.length ? '완성 상태 · ' + FS_STEPS.length + '단계'
      : cur === 0 ? '시작 전 · 0/' + FS_STEPS.length : cur + '/' + FS_STEPS.length + ' · ' + FS_STEPS[cur - 1].label;
  }
  function playItem(it){
    var node = el(it.t), mv = FS_MOVES[it.m];
    if (!node || !mv) return Promise.resolve();
    node.style.visibility = 'visible';
    var opt = { duration: ms(mv.dur), delay: it.delay || 0, easing: mv.ease, fill: 'both' };
    var a = node.animate(mv.kf, opt);
    var sh = mv.shade && shadeOf(node);
    if (sh) sh.animate(SHADE_KF, opt);
    return a.finished.then(function(){ Object.assign(node.style, clean(mv.kf[mv.kf.length - 1])); a.cancel(); }, function(){});
  }
  function playStep(i){
    cur = i + 1; show();
    return Promise.all(FS_STEPS[i].items.map(playItem));
  }
  function wait(t){ return new Promise(function(r){ setTimeout(r, t); }); }

  // 다시 보기: 처음부터 끝까지 한 번 재생하고 완성 상태에서 멈춤
  function replay(){
    var my = ++token;
    if (reduced) { applyState(FS_STEPS.length); return; }
    applyState(0);
    var chain = wait(500);
    FS_STEPS.forEach(function(_, i){
      chain = chain.then(function(){ if (my === token) return playStep(i).then(function(){ return wait(ms('--fs-gap-ms')); }); });
    });
  }
  // 한 단계씩: 클릭할 때마다 다음 동작 하나 (끝까지 가면 처음부터)
  function stepOnce(){
    ++token;
    var next = cur >= FS_STEPS.length ? 0 : cur;
    applyState(next);
    if (reduced) { applyState(next + 1); return; }
    playStep(next);
  }
  root.querySelector('[data-fs-replay]').addEventListener('click', replay);
  root.querySelector('[data-fs-step]').addEventListener('click', stepOnce);

  applyState(FS_STEPS.length);           // 첫 화면은 완성된 도식
  if (!reduced) setTimeout(replay, 1200); // 잠시 뒤 한 번 재생
})();
</script>
<!-- ===== 세 겹 병풍 도식 끝 ===== -->
  1. 색은 .fs-root의 --fs-t1~--fs-t3(겹별 색)와 --fs-accent(인원 숫자)만 바꾸면 됩니다.
  2. 폭 크기와 날개 높이는 --fs-panel-w, --fs-panel-h, --fs-flap-h로 조정합니다. 3폭 합이 장표 너비를 넘지 않게 하세요.
  3. 속도는 --fs-unfold-ms, --fs-flap-ms, --fs-gap-ms로 바꿉니다.
  4. 클릭 순서는 스크립트의 FS_STEPS 배열 한 곳에서 정합니다. 날개를 먼저 다 펼치고 싶으면 줄 순서만 바꾸면 됩니다.
  5. 폭을 하나 더 늘리려면 .fs-p3 안에 같은 구조의 .fs-panel을 넣고 data-fs 이름을 붙여 배열에 한 줄 추가합니다. 한 페이지에 한 번만 넣는 것을 전제로 합니다.
