다른 발표에 가져갈 장치 세 가지

장치 1

수렴 파형

서로 다른 주기와 세기로 흔들리던 여러 가닥이 오른쪽으로 갈수록 주기·위상이 맞아 한 줄이 됩니다. 가닥 수가 곧 참여 주체 수이고, 합쳐지는 지점이 곧 공통 기준입니다.

응용기관 간 협업 체계, 공통 매뉴얼 도입, 서로 다른 사업을 하나의 지표로 묶는 장표

장치 2

조율되는 글자

부제의 글자들이 제각각 높낮이로 떨리다가 기준선에 차례로 내려앉습니다. 글자 움직임 자체가 "기준을 맞춘다"는 메시지를 한 번 더 말합니다.

응용규정 정비·개정안 발표, "표준안 마련" 결론 슬라이드, 장 간지의 한 줄 요약

장치 3

기준 오브제 한 개

추상 개념(공통 기준)을 금속 소리굽쇠 하나로 고정합니다. 광원·반사·각인이 있는 실물급 렌더 한 개만 두고 나머지는 선과 글자로 비워 두어 시선이 흩어지지 않습니다.

응용저울(균형), 수평계(형평), 나침반(방향) 등 "기준 도구" 하나로 표지를 세우는 방식

Flow memo

계획안 목차를 리허설 순서로 옮기기

오케스트라는 연주 전에 오보에가 낸 A음에 모든 악기가 차례로 음을 맞춥니다. 연수 계획안의 목차를 이 리허설 순서에 겹쳐 안쪽 장표를 설계합니다.

계획안 구성은유안쪽 장표 아이디어
추진 배경조율 전, 제각각의 음현장 설문에서 역할 이해가 어긋난 지점 4개를 가닥별 맥놀이 폭으로 표시. 어긋남이 클수록 파형이 크게 출렁임
목적·방침기준음 A4 = 440 Hz방침 세 줄을 소리굽쇠 각인처럼 금속판에 새긴 한 장. 숫자 대신 문장이 주인공
연수 대상악기 편성표교원·실무사·늘봄 인력·협력강사를 무대 배치도의 자리(현·관·타악)처럼 반원형으로 배치하고 인원을 자리 크기로
세부 과정조율 → 파트 연습 → 합주공통 과정(같은 음 맞추기) → 직무별 과정(파트 연습) → 합동 사례 연구(합주)를 악보 세 마디로 구분
추진 일정시즌 공연 프로그램3·5·8·11월 회차를 공연 시즌 프로그램처럼 날짜·장소·편성 세 칸으로
기대 효과튜너 바늘 0 cent-50~+50 cent 튜너 눈금에서 바늘이 가운데로 들어오는 반원 게이지. 역할 혼선 감소·협업 만족도 같은 지표를 바늘 위치로

빌린 장치와 출처

빌린 대표 장치

  • 기준음 A4 = 440 Hz — 1955년 ISO가 표준 음높이로 채택한 값. 표지 눈썹 글과 소리굽쇠 각인에만 사용
  • 오보에가 먼저 A를 내고 나머지가 맞춘다 — 리허설 순서를 흐름 설계의 뼈대로
  • 맥놀이 — 주파수가 조금 다른 두 소리가 겹치면 소리가 주기적으로 커졌다 작아지고, 같아지면 사라짐. 파형 가닥의 진폭 출렁임으로 표현
  • 소리굽쇠 — 두 가지(tine)가 마주 떨며 한 음만 내는 기준 도구

  for (var i=0;i<txt.length;i++){ var s=document.createElement('span'); s.textContent=txt[i]; s.setAttribute('aria-hidden','true'); sub.appendChild(s); chars.push({el:s, seed:Math.sin(i*12.9898)*43758.5453 % 1}); }

  function size(){
    var r = cover.getBoundingClientRect(); DPR = Math.min(window.devicePixelRatio||1, 2);
    W = r.width; H = r.height; cv.width = Math.round(W*DPR); cv.height = Math.round(H*DPR);
    ctx.setTransform(DPR,0,0,DPR,0,0);
    // 라벨을 가닥 시작점 왼쪽에 둔다
    var maxR=0;
    STRANDS.forEach(function(s,i){ var l=document.getElementById('lab'+i); l.style.left='5.2%'; l.style.top=(s.y*100)+'%'; maxR=Math.max(maxR, l.offsetLeft+l.offsetWidth); });
    // 라벨이 길거나 화면이 좁으면 가닥 시작점을 라벨 오른쪽으로 민다
    X0 = Math.max(.17, (maxR + Math.max(8, W*.012))/W);
  }
  function ease(x){ return x<0?0:x>1?1:1-Math.pow(1-x,3); }
  function smooth(x){ x=Math.max(0,Math.min(1,x)); return x*x*(3-2*x); }

  // ── 소리굽쇠 기하(회전 전 좌표, 단위 = W) ──
  function forkGeom(){
    var u=W/1280, cx=W*.815, top=H*.075;
    return {u:u, cx:cx, top:top, gap:30*u, tw:27*u, yU:H*.545, sw:21*u, yStem:H*.88, ball:16*u, ang:-.11};
  }
  // 합류 지점: 회전된 손잡이 위 한 점(그림 좌표로 변환)
  function stemPoint(g){
    var py = H*.76, px = g.cx - g.sw/2;
    var ox=g.cx, oy=H*.5, a=g.ang, dx=px-ox, dy=py-oy;
    return {x: ox + dx*Math.cos(a) - dy*Math.sin(a), y: oy + dx*Math.sin(a) + dy*Math.cos(a)};
  }

  function drawWaves(time, D){
    var g=forkGeom(), P=stemPoint(g), x0=W*X0, xm=W*XM;
    var k0 = (2*Math.PI)/(W*.052), A0 = H*.018, w = time*2.2;
    ctx.lineCap='round'; ctx.lineJoin='round';
    STRANDS.forEach(function(s){
      ctx.beginPath();
      for (var x=x0; x<=xm; x+=1.5){
        var t=(x-x0)/(xm-x0), e=smooth(t), off=(1-e);
        var base = H*s.y + (P.y - H*s.y)*e;
        // 주기·위상 어긋남은 합류할수록 0으로, 맥놀이(진폭 출렁임)도 함께 사라짐
        var phase = k0*x*(1 + s.dk*.5*off*off*D) + s.ph*off*D - w;
        var amp = A0*(1 - .55*off*D*(.5+.5*Math.cos(s.beat*t*6.283 + time*1.3)));
        var y = base + amp*Math.sin(phase);
        if (x===x0) ctx.moveTo(x,y); else ctx.lineTo(x,y);
      }
      ctx.strokeStyle=s.c; ctx.globalAlpha=.92; ctx.lineWidth=Math.max(1.4, W*.0022); ctx.stroke();
    });
    ctx.globalAlpha=1;
    // 합류 뒤 한 줄: 남색 기준선
    ctx.beginPath();
    for (var x=xm-W*.01; x<=P.x; x+=1.5){
      var y = P.y + A0*Math.sin(k0*x - w);
      if (x<=xm-W*.01+.1) ctx.moveTo(x,y); else ctx.lineTo(x,y);
    }
    ctx.strokeStyle='#14213D'; ctx.lineWidth=Math.max(2, W*.0034); ctx.stroke();
    // 시작점 작은 고정 핀
    STRANDS.forEach(function(s){ ctx.beginPath(); ctx.arc(x0, H*s.y, Math.max(1.6,W*.0026), 0, 7); ctx.fillStyle=s.c; ctx.fill(); });
  }

  // 금속 면: 좌상단 광원 기준의 가로 그라데이션(평평한 막대 정면)
  function steelBar(x, w){
    var gr = ctx.createLinearGradient(x,0,x+w,0);
    gr.addColorStop(0,'#7C8592'); gr.addColorStop(.08,'#F4F6F9'); gr.addColorStop(.2,'#D3D8DF');
    gr.addColorStop(.42,'#BCC3CC'); gr.addColorStop(.62,'#A3ABB7'); gr.addColorStop(.86,'#C9CFD7'); gr.addColorStop(.95,'#8D96A3'); gr.addColorStop(1,'#5E6773');
    return gr;
  }
  function tinePath(x, w, top, yU, dx){
    // dx: 떨림으로 끝이 벌어지는 양(위로 갈수록 커짐)
    ctx.beginPath();
    ctx.moveTo(x, yU); ctx.lineTo(x, top+w*.35);
    ctx.quadraticCurveTo(x+dx*.0, top, x+w*.5+dx, top);
    ctx.quadraticCurveTo(x+w, top, x+w, top+w*.35);
    ctx.lineTo(x+w, yU); ctx.closePath();
  }
  function silhouette(g){
    var L=g.cx-g.gap/2-g.tw, R=g.cx+g.gap/2, ro=g.gap/2+g.tw, ri=g.gap/2;
    ctx.beginPath();
    ctx.rect(L, g.top, g.tw, g.yU-g.top); ctx.rect(R, g.top, g.tw, g.yU-g.top);
    ctx.moveTo(g.cx-ro, g.yU); ctx.arc(g.cx, g.yU, ro, Math.PI, 0, true); ctx.lineTo(g.cx+ri, g.yU); ctx.arc(g.cx, g.yU, ri, 0, Math.PI, false); ctx.closePath();
    ctx.rect(g.cx-g.sw/2, g.yU+ro-6*g.u, g.sw, g.yStem-(g.yU+ro)+6*g.u);
    ctx.moveTo(g.cx+g.ball*1.15, g.yStem+g.ball*.6); ctx.arc(g.cx, g.yStem+g.ball*.6, g.ball*1.15, 0, 7);
  }
  function brushed(x, y, w, h, seed){
    // 헤어라인 금속 결: 세로로 아주 옅은 선
    ctx.save(); ctx.globalAlpha=.07; ctx.lineWidth=.6;
    for (var i=0;i<w;i+=1.3){ var r=Math.abs(Math.sin((i+seed)*91.7)); ctx.strokeStyle = r>.5?'#FFFFFF':'#47505C'; ctx.beginPath(); ctx.moveTo(x+i, y); ctx.lineTo(x+i, y+h); ctx.stroke(); }
    ctx.restore();
  }

  function drawFork(time, vibAmp){
    var g=forkGeom(), u=g.u, ro=g.gap/2+g.tw, ri=g.gap/2;
    var vib = vibAmp*Math.sin(time*38);
    ctx.save();
    ctx.translate(g.cx, H*.5); ctx.rotate(g.ang); ctx.translate(-g.cx, -H*.5);

    // 바닥 그림자(오른쪽 아래로 떨어지는 부드러운 그림자)
    ctx.save(); ctx.translate(16*u, 20*u); ctx.shadowColor='rgba(20,33,61,.22)'; ctx.shadowBlur=26*u; ctx.fillStyle='rgba(20,33,61,.10)';
    silhouette(g); ctx.fill(); ctx.restore();

    // 손잡이(원기둥): 가로 그라데이션으로 둥근 양감
    var sx=g.cx-g.sw/2, sTop=g.yU+ro-6*u;
    var cg=ctx.createLinearGradient(sx,0,sx+g.sw,0);
    cg.addColorStop(0,'#5F6875'); cg.addColorStop(.18,'#AEB6C1'); cg.addColorStop(.32,'#F7F9FB'); cg.addColorStop(.5,'#C3CAD3'); cg.addColorStop(.78,'#8A93A0'); cg.addColorStop(1,'#4E5763');
    ctx.fillStyle=cg; ctx.fillRect(sx, sTop, g.sw, g.yStem-sTop);
    // 손잡이 위 각인 "A 440"
    ctx.save(); ctx.translate(g.cx+1.2*u, g.yStem-62*u); ctx.rotate(-Math.PI/2);
    ctx.font='600 '+(10.5*u)+'px "IBM Plex Mono", Consolas, monospace'; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillStyle='rgba(255,255,255,.75)'; ctx.fillText('A 440', .7*u, .7*u);
    ctx.fillStyle='rgba(40,48,60,.78)'; ctx.fillText('A 440', 0, 0); ctx.restore();

    // 끝 구슬(구): 방사형 그라데이션 + 반사광
    var by=g.yStem+g.ball*.6, br=g.ball*1.15;
    var rg=ctx.createRadialGradient(g.cx-br*.4, by-br*.45, br*.08, g.cx, by, br);
    rg.addColorStop(0,'#FFFFFF'); rg.addColorStop(.25,'#D9DEE5'); rg.addColorStop(.7,'#8E97A4'); rg.addColorStop(1,'#545D69');
    ctx.beginPath(); ctx.arc(g.cx, by, br, 0, 7); ctx.fillStyle=rg; ctx.fill();
    ctx.beginPath(); ctx.arc(g.cx, by, br*.82, .2, 1.4); ctx.strokeStyle='rgba(255,255,255,.35)'; ctx.lineWidth=1.2*u; ctx.stroke();

    // U자 굽이: 바깥 반원 - 안쪽 반원
    ctx.beginPath(); ctx.arc(g.cx, g.yU, ro, Math.PI, 0, true); ctx.arc(g.cx, g.yU, ri, 0, Math.PI, false); ctx.closePath();
    var ug=ctx.createLinearGradient(g.cx-ro,0,g.cx+ro,0);
    ug.addColorStop(0,'#7C8592'); ug.addColorStop(.06,'#EEF1F5'); ug.addColorStop(.3,'#C4CAD3'); ug.addColorStop(.5,'#9AA3AF'); ug.addColorStop(.72,'#BAC1CA'); ug.addColorStop(.94,'#8B94A1'); ug.addColorStop(1,'#5E6773');
    ctx.fillStyle=ug; ctx.fill();
    var ush=ctx.createLinearGradient(0,g.yU,0,g.yU+ro); ush.addColorStop(0,'rgba(20,33,61,0)'); ush.addColorStop(1,'rgba(20,33,61,.22)');
    ctx.fillStyle=ush; ctx.fill();

    // 두 가지(tine): 떨림 잔상 → 본체 → 결 → 하이라이트
    var L=g.cx-g.gap/2-g.tw, R=g.cx+g.gap/2;
    [[L,-1],[R,1]].forEach(function(p){
      var x=p[0], dir=p[1];
      if (vibAmp>0){ ctx.save(); ctx.globalAlpha=.22; ctx.fillStyle=steelBar(x,g.tw); tinePath(x,g.tw,g.top,g.yU+1,dir*vibAmp); ctx.fill(); tinePath(x,g.tw,g.top,g.yU+1,-dir*vibAmp*.6); ctx.fill(); ctx.restore(); }
      ctx.save();
      // 떨림: 위로 갈수록 바깥으로 기울도록 기울임 변환
      ctx.translate(x+g.tw/2, g.yU); ctx.transform(1,0,-dir*vib/(g.yU-g.top),1,0,0); ctx.translate(-(x+g.tw/2), -g.yU);
      tinePath(x,g.tw,g.top,g.yU+1,0); ctx.fillStyle=steelBar(x,g.tw); ctx.fill();
      ctx.save(); ctx.clip(); brushed(x, g.top, g.tw, g.yU-g.top, x);
      // 위쪽은 하늘빛 반사, 아래쪽은 살짝 어둡게
      var vg=ctx.createLinearGradient(0,g.top,0,g.yU); vg.addColorStop(0,'rgba(255,255,255,.28)'); vg.addColorStop(.35,'rgba(214,226,246,.10)'); vg.addColorStop(1,'rgba(20,33,61,.12)');
      ctx.fillStyle=vg; ctx.fillRect(x, g.top, g.tw, g.yU-g.top);
      // 비스듬한 반사 띠
      ctx.globalAlpha=.35; ctx.fillStyle='#FFFFFF'; ctx.beginPath(); ctx.moveTo(x, g.top+H*.18); ctx.lineTo(x+g.tw, g.top+H*.12); ctx.lineTo(x+g.tw, g.top+H*.15); ctx.lineTo(x, g.top+H*.21); ctx.fill();
      ctx.restore();
      tinePath(x,g.tw,g.top,g.yU+1,0); ctx.strokeStyle='rgba(60,68,80,.55)'; ctx.lineWidth=.8*u; ctx.stroke();
      ctx.restore();
    });
    ctx.restore();
  }

  // 소리 동심호: 가지 끝 오른쪽으로 퍼지는 옅은 호(제목 쪽으로는 뻗지 않음)
  function drawRings(time, on){
    var g=forkGeom(), cx=g.cx+W*.012, cy=g.top+H*.14;
    ctx.save(); ctx.lineWidth=Math.max(1, W*.0012);
    for (var i=0;i<4;i++){
      var p = on ? ((time*.35 + i/4) % 1) : (i+1)/5;
      var r = W*(.035 + p*.11);
      ctx.strokeStyle='rgba(43,89,195,'+(0.42*(1-p)).toFixed(3)+')';
      ctx.beginPath(); ctx.arc(cx, cy, r, -0.85, 0.55); ctx.stroke();
      ctx.beginPath(); ctx.arc(cx-W*.024, cy, r, Math.PI-0.55, Math.PI+0.85); ctx.stroke();
    }
    ctx.restore();
  }

  function frame(now){
    var time=(now-t0)/1000;
    var k = reduce ? 1 : ease(time/2.4);           // 0 → 1: 조율 진행도
    var D = 1 + (1-k)*1.6;                           // 처음엔 더 크게 어긋남
    ctx.clearRect(0,0,W,H);
    ctx.fillStyle='#F6F8FB'; ctx.fillRect(0,0,W,H);
    // 아주 옅은 오선(5줄) — 하단 띠의 배경 결
    ctx.strokeStyle='rgba(20,33,61,.06)'; ctx.lineWidth=1;
    for (var i=0;i<5;i++){ var yy=H*(.62+i*.06); ctx.beginPath(); ctx.moveTo(W*X0-6,yy); ctx.lineTo(W*.70,yy); ctx.stroke(); }
    var tt = reduce ? 0 : time;
    drawWaves(tt, D);
    drawRings(tt, !reduce);
    drawFork(tt, reduce ? 0 : (k<1 ? .9*W/1280 : .55*W/1280));
    // 조율되는 글자
    chars.forEach(function(c,i){
      var local = reduce ? 1 : ease((time - .25 - i*.06)/1.2);
      var off = (1-local) * (c.seed*2-1) * 0.55;    // em 단위 상하 어긋남
      var rot = (1-local) * (c.seed*2-1) * 6;
      c.el.style.transform = 'translateY('+off.toFixed(3)+'em) rotate('+rot.toFixed(2)+'deg)';
      c.el.style.opacity = (0.55 + .45*local).toFixed(3);
    });
    if (!reduce) raf=requestAnimationFrame(frame);
  }
  function start(){ cancelAnimationFrame(raf); size(); t0=performance.now(); frame(t0); }
  document.getElementById('replay').addEventListener('click', start);
  window.addEventListener('resize', function(){ size(); if(reduce) frame(performance.now()); });
  start();

  }
  (document.fonts && document.fonts.ready ? document.fonts.ready : Promise.resolve()).then(drawMinis);
