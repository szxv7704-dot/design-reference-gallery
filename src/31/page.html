사진 게이지

사진 속 칸을 하나씩 켜기

철골 사이로 보이는 하늘 조각을 사진에서 직접 찾아, 그 실제 삼각형 칸을 아래에서 위로 하나씩 금빛으로 켭니다. 켜지는 순간 잠깐 밝게 번쩍여 "한 칸 = 한 사람"이 눈에 보입니다. 모금·참여·공정률처럼 "다 채워지면 완성"인 목표에 씁니다.

숫자 동기화

차오르는 높이와 숫자를 함께

왼쪽 숫자가 0에서 목표까지 올라가는 동안 사진이 같은 속도로 채워집니다. 눈이 숫자와 그림을 오가도 같은 이야기를 읽습니다.

은유 고르기

구조가 곧 메시지인 건축물

지오데식 돔은 약한 삼각형 수천 개가 서로 힘을 나눠 가장 가볍고 튼튼한 구를 만듭니다. "한 사람의 제안은 작아도 모이면 예산이 된다"를 설명 없이 보여 줍니다.

흐름 설계

  1. 제목: 가장 크고 진한 남색, 동사 '짓는'만 주황으로 강조합니다.
  2. 사진: 1967 몬트리올 엑스포 미국관. 파란 하늘 앞 은빛 돔이 첫눈에 들어옵니다.
  3. 칸 켜기: 돔의 삼각형 칸이 바닥부터 한 칸씩 금빛으로 켜집니다. 은색 철골은 그대로 남아 금빛 유리를 끼운 돔처럼 마무리되고, 하늘의 파랑과 보색 대비로 끝납니다.
  4. 숫자: 참여 시민 0 → 10,000명. 표지에 남는 마지막 숫자가 목표 그 자체입니다.

동작

  • 0.5초 뒤 시작해 5.6초 동안 칸 약 1,000개가 아래에서 위로 켜집니다. 같은 높이의 칸은 순서를 조금씩 섞어 한 칸씩 켜지는 것이 보이게 했습니다.
  • 칸마다 켜지는 순간 0.26초 동안 밝게 번쩍였다가 제 색으로 가라앉습니다. 칸마다 유리 밝기를 조금씩 다르게, 돔 가장자리는 어둡게 칠해 공 모양 양감을 냈습니다.
  • 숫자는 켜진 칸의 비율만큼 0 → 10,000명으로 올라갑니다. 다 켜지면 멈추고, 동작 줄이기 설정이면 처음부터 완성 상태입니다.
  • 칸 찾기: 하늘색 픽셀 덩어리를 칸으로 보고, 건물에 가려 하늘이 안 보이는 곳은 격자 크기(46px)의 삼각형으로 나눕니다. 앞쪽 갈대는 돔이 아니므로 켜지 않습니다.

    for(const L of ids){ const v=.9+.2*rnd();
      for(let j=start[L];j<start[L+1];j++){const p=px[j], y=(p/W)|0, x=p-y*W;
        const r=d[p*4],g=d[p*4+1],b=d[p*4+2], l=.3*r+.59*g+.11*b, ex=(x-CX)/R, ey=(y-CY)/R;
        const z=Math.sqrt(Math.max(0,1-ex*ex-ey*ey)), sh=(.62+.5*z+.18*(-ex*.5-ey*.6))*v;
        gold[p*4]=(l*.55+150)*sh; gold[p*4+1]=(l*.5+88)*sh; gold[p*4+2]=(l*.2+10)*sh; gold[p*4+3]=255; }}
    return {W,H,ids,start,px,gold};
  }

  // canvas: 사진 위에 겹친 투명 캔버스. opts.dur(ms), opts.onProgress(0~1)
  window.DomeCells = function(img, canvas, opts){
    opts=opts||{};
    const M=segment(img), ctx=canvas.getContext('2d');
    canvas.width=M.W; canvas.height=M.H;
    const out=ctx.createImageData(M.W,M.H), o=out.data, POP=260;
    let raf=0, lit=0, pops=[];
    function paint(L, f){ for(let j=M.start[L];j<M.start[L+1];j++){const p=M.px[j]*4;
      o[p]=M.gold[p]*f; o[p+1]=M.gold[p+1]*f; o[p+2]=M.gold[p+2]*f+(f-1)*120; o[p+3]=255; }}
    function clear(){ o.fill(0); ctx.putImageData(out,0,0); lit=0; pops=[]; }
    function finish(){ for(let i=0;i<M.ids.length;i++) paint(M.ids[i],1); ctx.putImageData(out,0,0); lit=M.ids.length; opts.onProgress&&opts.onProgress(1); }
    function play(){
      cancelAnimationFrame(raf); clear(); opts.onProgress&&opts.onProgress(0);
      const dur=opts.dur||5200, t0=performance.now()+(opts.delay||0), total=M.ids.length;
      const ease=t=>t<.5?4*t*t*t:1-Math.pow(-2*t+2,3)/2;
      (function tick(now){
        const p=Math.min(Math.max((now-t0)/dur,0),1), want=Math.round(ease(p)*total);
        while(lit<want){ const L=M.ids[lit++]; pops.push([L,now]); }
        pops=pops.filter(([L,ts])=>{ const a=(now-ts)/POP; paint(L, a>=1?1:1.45-.45*a); return a<1; });
        ctx.putImageData(out,0,0); opts.onProgress&&opts.onProgress(p<1?want/total:1);
        if(p<1||pops.length) raf=requestAnimationFrame(tick);
      })(performance.now());
    }
    return {play, finish, cells:M.ids.length};
  };
})();
</script>
<script>
(function(){
  var b=document.getElementById('replay'), n=document.getElementById('n');
  var img=document.querySelector('.ibox > img'), cv=document.getElementById('cells'), GOAL=10000;
  var reduce=window.matchMedia && matchMedia('(prefers-reduced-motion: reduce)').matches;
  function show(p){ n.textContent=Math.round(GOAL*p).toLocaleString('ko-KR'); }
  function start(){
    var dome=DomeCells(img, cv, {dur:5600, delay:500, onProgress:show});   // 칸 수만큼 0→10,000명
    if(reduce){ dome.finish(); if(b) b.hidden=true; return; }
    dome.play();
    b.addEventListener('click', function(){ dome.play(); });
  }
  if(img.complete && img.naturalWidth) start(); else img.addEventListener('load', start);
})();
